Installation

Add funnel chart to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.1.0.tgz zeus-ui add funnel-chart-card

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — chart-card, chevrons, chip, cx, segmented-control, select, tokens, use-count-up, use-dismiss-on-outside-press. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

funnel-chart-card-demo.tsx
"use client";

import { FunnelChartCard } from "@/components/application/charts/funnel-chart-card";

export default function FunnelChartCardDemo() {
  return (
    <div className="w-full max-w-2xl">
      <FunnelChartCard />
    </div>
  );
}

components/application/charts/funnel-chart-card.tsx
"use client";
import { motion } from "motion/react";
import { type ReactNode, useState } from "react";
import { useHydratedReducedMotion } from "../../hooks/use-hydrated-reduced-motion";
import { formatNumber } from "./chart-card";
import {
  type ChartCommonProps,
  ChartFrame,
  type ChartRange,
  ink,
  PeriodSelect,
  StatTiles,
  safeRatio,
  useChartRange,
} from "./pro-chart";
export interface FunnelStage {
  label: string;
  value: number;
  color?: string;
  icon?: ReactNode;
}
export interface FunnelChartRange extends ChartRange {
  stages: FunnelStage[];
}
export interface FunnelChartCardProps extends ChartCommonProps {
  stages?: FunnelStage[];
  ranges?: FunnelChartRange[];
  shape?: "smooth" | "sharp";
  mono?: boolean;
  tiles?: boolean;
}
export const FUNNEL_STAGES: FunnelStage[] = [
  { label: "Link opened", value: 197 },
  { label: "Started", value: 110 },
  { label: "Completed", value: 77 },
  { label: "Converted", value: 38 },
];
const RANGES: FunnelChartRange[] = [
  { id: "7d", label: "Last 7 days", stages: FUNNEL_STAGES, delta: 0.052 },
  {
    id: "30d",
    label: "Last 30 days",
    stages: FUNNEL_STAGES.map((s, i) => ({ ...s, value: [842, 463, 301, 152][i]! })),
    delta: 0.034,
  },
  {
    id: "90d",
    label: "Last 90 days",
    stages: FUNNEL_STAGES.map((s, i) => ({ ...s, value: [2510, 1380, 902, 455][i]! })),
    delta: -0.018,
  },
];
export function FunnelChartCard({
  stages,
  ranges: supplied,
  title = "Sign-up funnel",
  shape = "smooth",
  mono = false,
  tiles = true,
  ...props
}: FunnelChartCardProps = {}) {
  const ranges = supplied ?? (stages ? undefined : RANGES),
    period = useChartRange(ranges, props.defaultRange, props.onRangeChange),
    data = period.current?.stages ?? stages ?? FUNNEL_STAGES,
    [active, setActive] = useState<number | null>(null),
    reduced = useHydratedReducedMotion(),
    format = props.format ?? formatNumber,
    first = data[0]?.value ?? 0;
  const colors = data.map((s, i) => (mono ? "var(--color-text-tertiary)" : (s.color ?? ink(i))));
  return (
    <ChartFrame
      title={active == null ? title : (data[active]?.label ?? title)}
      value={
        active == null
          ? (period.current?.headline ?? props.headline ?? first)
          : (data[active]?.value ?? 0)
      }
      format={format}
      delta={period.current?.delta ?? props.delta}
      className={props.className}
      action={
        <PeriodSelect
          ranges={ranges}
          selected={period.id}
          onChange={(id) => {
            setActive(null);
            period.select(id);
          }}
          label={period.current?.label ?? props.range ?? "Last 7 days"}
        />
      }
      footer={
        tiles && (
          <StatTiles
            columns={Math.max(1, data.length)}
            items={data.map((s, i) => ({
              label: s.label,
              value: format(s.value),
              color: colors[i],
            }))}
            active={active}
            onHover={setActive}
          />
        )
      }
    >
      <svg
        viewBox="0 0 640 160"
        preserveAspectRatio="none"
        className="h-40 w-full overflow-visible"
        role="img"
        aria-label={`${title}: ${data.map((s) => `${s.label} ${format(s.value)}`).join(", ")}`}
      >
        {data.map((stage, i) => {
          const width = 640 / Math.max(1, data.length),
            x = i * width,
            h = Math.max(4, safeRatio(stage.value, first) * 136),
            next = Math.max(4, safeRatio(data[i + 1]?.value ?? stage.value, first) * 136),
            top = 80 - h / 2,
            bottom = 80 + h / 2,
            nt = 80 - next / 2,
            nb = 80 + next / 2;
          const path =
            shape === "sharp"
              ? `M ${x} ${top} L ${x + width - 3} ${nt} L ${x + width - 3} ${nb} L ${x} ${bottom} Z`
              : `M ${x} ${top} H ${x + width * 0.46} C ${x + width * 0.7} ${top} ${x + width * 0.68} ${nt} ${x + width - 3} ${nt} L ${x + width - 3} ${nb} C ${x + width * 0.68} ${nb} ${x + width * 0.7} ${bottom} ${x + width * 0.46} ${bottom} H ${x} Z`;
          return (
            <motion.g
              key={`${period.id}-${stage.label}-${reduced ? "reduced" : "animated"}`}
              initial={reduced ? false : { opacity: 0 }}
              animate={{ opacity: active != null && active !== i ? 0.25 : 1 }}
              transition={{ duration: reduced ? 0 : 0.4, delay: reduced ? 0 : i * 0.06 }}
              onMouseEnter={() => setActive(i)}
              onMouseLeave={() => setActive(null)}
            >
              {shape !== "sharp" && (
                <>
                  <path
                    d={path}
                    fill={colors[i]}
                    opacity={0.12}
                    transform="translate(0 -12) scale(1 1.15)"
                  />
                  <path
                    d={path}
                    fill={colors[i]}
                    opacity={0.2}
                    transform="translate(0 -6) scale(1 1.075)"
                  />
                </>
              )}
              <path d={path} fill={colors[i]} />
              <foreignObject x={x} y={70} width={width} height={24}>
                <span className="mx-auto block w-fit rounded-full bg-background-inner-default px-1.5 text-caption-1-medium text-text-primary">
                  {Math.round(safeRatio(stage.value, first) * 100)}%
                </span>
              </foreignObject>
            </motion.g>
          );
        })}
      </svg>
    </ChartFrame>
  );
}
