Checkbox

Hand-drawn tick animation, indeterminate state, two sizes and a card variant.
Preview
Code

Installation

Add checkbox to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.2.1.tgz zeus-ui add checkbox

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — cx, tokens. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

checkbox-demo.tsx
"use client";

import { Checkbox } from "@/components/base/checkbox/checkbox";
import { CheckboxCard } from "@/components/base/checkbox/checkbox-card";

export default function CheckboxDemo() {
  return (
    <div className="flex flex-col gap-6">
      <div className="flex flex-wrap items-center gap-6">
        <Checkbox aria-label="Unchecked" />
        <Checkbox defaultSelected aria-label="Checked" />
        <Checkbox isIndeterminate aria-label="Indeterminate" />
        <Checkbox isDisabled aria-label="Disabled" />
        <Checkbox size="sm" defaultSelected aria-label="Small" />
      </div>

      <Checkbox defaultSelected>Remember me</Checkbox>

      <div className="flex w-full max-w-sm flex-col gap-3">
        <CheckboxCard
          defaultSelected
          title="Weekly digest"
          description="A summary of what changed, every Monday."
        />
        <CheckboxCard title="Product updates" description="Only when something ships." />
      </div>
    </div>
  );
}

Source

3 files. Source code you can read, customize and keep in your project.

components/base/checkbox/checkbox-card.tsx
"use client";

import type { ReactNode, Ref } from "react";
import { Checkbox as AriaCheckbox } from "react-aria-components";
import type { CheckboxProps as AriaCheckboxProps } from "react-aria-components";
import { cx } from "../../utils/cx";
import { CheckboxGlyph } from "./checkbox-glyph";

/**
 * Figma source: Board UI → Checkbox items (node 3793:2966).
 *
 * Selectable card: title + description on the left, checkbox on the right.
 *   card    radius/2lg (10px), 1px border/button/default, pl 16 pr 20 py 12
 *   hover   bg background/primary/hover (#f7f7f7)
 *   title   Body 1/Medium, text/primary
 *   desc    Body 1/Regular, text/secondary
 *
 * The whole card toggles the checkbox (react-aria Checkbox renders the label),
 * so a click anywhere flips the state.
 */

export interface CheckboxCardProps extends Omit<AriaCheckboxProps, "children"> {
  title: ReactNode;
  description?: ReactNode;
  ref?: Ref<HTMLLabelElement>;
}

export function CheckboxCard({
  className,
  title,
  description,
  ref,
  ...props
}: CheckboxCardProps) {
  return (
    <AriaCheckbox
      ref={ref}
      {...props}
      className={(state) =>
        cx(
          "group flex w-full items-center justify-between gap-3 rounded-2lg border border-border-button-default py-3 pr-5 pl-4 select-none",
          "transition-colors duration-150 ease",
          state.isHovered && !state.isDisabled
            ? "bg-background-primary-hover"
            : "bg-background-primary-default",
          state.isDisabled ? "cursor-not-allowed opacity-50" : "cursor-pointer",
          typeof className === "function" ? className(state) : className,
        )
      }
    >
      {(state) => (
        <>
          <span className="flex min-w-0 flex-col gap-0.5">
            <span className="truncate text-body-medium text-text-primary">{title}</span>
            {description !== undefined && description !== null && (
              <span className="truncate text-body-regular text-text-secondary">{description}</span>
            )}
          </span>
          <span className="flex shrink-0 items-center py-1">
            <CheckboxGlyph state={state} />
          </span>
        </>
      )}
    </AriaCheckbox>
  );
}
