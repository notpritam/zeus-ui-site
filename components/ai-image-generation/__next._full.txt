function GenerationRipple() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const reduced = useReducedMotion();
  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas?.getContext("2d");
    if (!canvas || !ctx) return;
    let frame = 0;
    const start = performance.now();
    const ratio = Math.min(devicePixelRatio || 1, 2);
    canvas.width = 200 * ratio;
    canvas.height = 250 * ratio;
    ctx.scale(ratio, ratio);
    const draw = (now: number) => {
      const time = reduced ? 1 : (now - start) / 1000;
      ctx.clearRect(0, 0, 200, 250);
      for (let index = 0; index < 24; index++) {
        const radius = ((index * 7 + time * 13) % 168) + 3;
        const alpha = Math.sin(Math.min(radius / 168, 1) * Math.PI) * 0.24;
        ctx.beginPath();
        ctx.ellipse(100, 125, radius * 0.74, radius, Math.sin(time * 0.6) * 0.12, 0, Math.PI * 2);
        ctx.strokeStyle = `rgba(122,123,136,${alpha})`;
        ctx.lineWidth = 0.7;
        ctx.stroke();
      }
      if (!reduced) frame = requestAnimationFrame(draw);
    };
    draw(start);
    return () => cancelAnimationFrame(frame);
  }, [reduced]);
  return <canvas ref={canvasRef} aria-hidden className="absolute inset-0 size-full" />;
}

        input={input}
        onInputChange={setInput}

        onNew={reset}
        busy={busy}
        contained={contained}
        className={className}
        panel={panel}

          <div className="flex flex-1 flex-col justify-center py-20 text-center">
            <h1 className="text-title-2-medium text-text-primary">Bring your ideas to life.</h1>
            <p className="pt-2 text-body-regular text-text-secondary">
              Describe an image to get started.
            </p>
          </div>
        )}

          <motion.p
            key={prompt}
            initial={run === 0 || reduced ? false : { opacity: 0, y: 8 }}

            transition={{ duration: reduced ? 0 : 0.3 }}
            className="ml-auto max-w-md rounded-2xl bg-background-tertiary-default px-4 py-3 text-body-regular text-text-primary"
          >
            {prompt}
          </motion.p>
        )}
        {prompt && !onGenerate && (
          <p className="text-body-2-regular text-text-tertiary">
            Example image generation · No generation service connected
          </p>
        )}
        {(busy || result) && (
          <GenerationImage
            key={run}
            busy={busy}
            seconds={onGenerate ? null : seconds}
            animateEntry={run > 0}
            result={result}
            onOpen={() => result && setOpened({ src: result.src, alt: result.alt ?? prompt })}
          />
        )}
        {result && (
          <div className="flex items-center gap-3">
            <ConversationFeedback text={prompt} onRetry={() => void generate(prompt)} />
            <a
              href={result.src}
              download
              className="flex items-center gap-1 text-body-2-medium text-text-secondary hover:text-text-primary"
            >
              <RiDownloadLine className="size-4" />
              Download
            </a>
          </div>
        )}
        {error && (
          <p role="alert" className="text-body-regular text-status-rose-text">
            {error}
          </p>
        )}
        {prompt && !busy && !result && !error && (
          <p role="status" className="text-body-regular text-text-secondary">
            Generation stopped.
          </p>
        )}
        {prompt && !busy && !result && (
          <Button
            size="small"
            variant="secondary"
            className="self-start"
            onClick={() => void generate(prompt)}
          >
            Try again
          </Button>
        )}
      </AgentWorkspace>
      <ModalOverlay
        isOpen={!!opened}
        onOpenChange={(open) => {
          if (!open) setOpened(null);
        }}
        isDismissable
        className="fixed inset-0 z-50 flex items-center justify-center bg-background-modal-overlay p-4 backdrop-blur-lg"
      >
        <Modal className="max-h-[95dvh] max-w-4xl overflow-auto rounded-3xl border border-border-button-default bg-background-full p-3 shadow-dropdown">
          <Dialog className="outline-none">
            <div className="mb-3 flex items-center justify-between gap-4">
              <Heading slot="title" className="truncate text-body-medium text-text-primary">
                {opened?.alt}
              </Heading>
              <IconButton
                icon={RiCloseLine}
                size="small"
                aria-label="Close image"
                onClick={() => setOpened(null)}
              />
            </div>
            {opened && (
              <img
                src={opened.src}
                alt={opened.alt}
                className="max-h-[75dvh] w-full rounded-xl object-contain"
              />
            )}
            <div className="mt-3 flex justify-end">
              <a
                href={opened?.src}
                download
                className="flex items-center gap-1 rounded-full bg-button-primary px-3 py-2 text-body-medium text-button-primary-foreground"
              >
                <RiDownloadLine className="size-4" />
                Download image
              </a>
            </div>
          </Dialog>
        </Modal>
      </ModalOverlay>
    </>
  );
