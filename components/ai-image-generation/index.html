Installation

Add ai image generation to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.2.2.tgz zeus-ui add ai-image-generation

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — agent-log, agent-progress, agent-thinking, ai-chat, avatar, button, checkbox, chevrons, chip, composer, composer-loader, cx, date-picker, dropdown, file-upload, icon-button, input, pagination, radio, select, settings-modal, sidebar, switch, tabs, task-list, theme-toggle, tokens, tooltip, use-dismiss-on-outside-press. The component source is yours; imported npm packages are installed as needed.

function AgentRepositorySidebar({
  selected,
  onNew,
  onClose,
  collapsed,
  onCollapse,
  className,
}: {
  selected: string;
  onNew: () => void;
  onClose?: () => void;
  collapsed: boolean;
  onCollapse?: () => void;
  className?: string;
}) {
  const [expanded, setExpanded] = useState<Record<string, boolean>>({
    "vibl coding project": true,
  });
  const [query, setQuery] = useState("");
  const [searchOpen, setSearchOpen] = useState(false);
  const [settings, setSettings] = useState(false);
  const [notice, setNotice] = useState<string | null>(null);
  const inputRef = useRef<HTMLInputElement>(null);
  useEffect(() => {
    if (searchOpen) inputRef.current?.focus();
  }, [searchOpen]);
  useEffect(() => {
    const listener = (event: KeyboardEvent) => {
      if ((event.metaKey || event.ctrlKey) && event.key.toLowerCase() === "l") {
        event.preventDefault();
        setSearchOpen(true);
      }
    };
    window.addEventListener("keydown", listener);
    return () => window.removeEventListener("keydown", listener);
  }, []);
  const rows = REPOSITORIES.filter(
    (repo) =>
      !query ||
      `${repo.name} ${repo.chats.map((chat) => chat.name).join(" ")}`
        .toLowerCase()
        .includes(query.toLowerCase()),
  );
  return (
    <aside
      className={cx(
        "flex h-full shrink-0 flex-col justify-between overflow-hidden rounded-3xl border border-border-button-white bg-background-secondary-default p-3 shadow-sidebar transition-[width] duration-300 motion-reduce:transition-none",
        collapsed ? "w-[60px]" : "w-[260px]",
        className,
      )}
    >
      <div className="flex min-h-0 flex-col gap-3">
        <div className="flex items-center justify-between gap-1">
          <DashboardUserMenu collapsed={collapsed} />
          {!collapsed && (
            <IconButton
              size="small"
              icon={onClose ? RiCloseLine : RiSideBarLine}
              aria-label={onClose ? "Close navigation" : "Collapse sidebar"}
              onClick={onClose ?? onCollapse}
            />
          )}
        </div>
        {collapsed ? (
          <IconButton
            size="small"
            icon={RiMenuLine}
            aria-label="Expand sidebar"
            onClick={onCollapse}
          />
        ) : searchOpen ? (
          <InputBase
            ref={inputRef}
            aria-label="Search repositories"
            placeholder="Search repositories"
            value={query}
            onChange={(event) => setQuery(event.target.value)}
            onKeyDown={(event) => {
              if (event.key === "Escape") {
                setQuery("");
                setSearchOpen(false);
              }
            }}
            leadingIcon={RiSearchLine}
          />
        ) : (
          <button
            type="button"
            onClick={() => setSearchOpen(true)}
            className="flex h-9 cursor-pointer items-center gap-2 rounded-full bg-background-tertiary-default px-2.5 text-body-medium text-text-secondary"
          >
            <RiSearchLine className="size-5" />
            <span className="flex-1 text-left">Quick Search</span>
            <span className="rounded-md bg-background-secondary-hover px-1 text-caption-1-medium">
              ⌘L
            </span>
          </button>
        )}
        <div className="flex flex-col gap-1">
          {[
            { label: "New agent", Icon: RiAddLine, action: onNew },
            { label: "Automations", Icon: RiRobot2Line, action: () => setNotice("Automations") },
            { label: "Customize", Icon: RiSparklingLine, action: () => setSettings(true) },
          ].map(({ label, Icon, action }) => (
            <button
              type="button"
              key={label}
              onClick={action}
              aria-label={label}
              className="flex h-9 cursor-pointer items-center gap-2 overflow-hidden rounded-2lg px-2 text-body-medium whitespace-nowrap text-text-secondary hover:bg-background-secondary-hover"
            >
              <Icon className="size-5 shrink-0" />
              {!collapsed && label}
            </button>
          ))}
        </div>
        {!collapsed && (
          <div className="flex min-h-0 flex-col gap-2 overflow-y-auto pt-2">
            <p className="px-0 text-body-medium text-text-secondary">Repositories</p>
            {rows.map((repo) => {
              const open = !!expanded[repo.name] || !!query;
              return (
                <div key={repo.name}>
                  <button
                    type="button"
                    aria-expanded={open}
                    onClick={() => setExpanded((old) => ({ ...old, [repo.name]: !open }))}
                    className="flex h-9 w-full cursor-pointer items-center gap-2 rounded-2lg px-2 text-left text-body-medium text-text-secondary hover:bg-background-secondary-hover"
                  >
                    {open ? (
                      <RiFolderOpenLine className="size-5" />
                    ) : (
                      <RiFolder3Line className="size-5" />
                    )}
                    <span className="truncate">{repo.name}</span>
                  </button>
                  {open && (
                    <ul className="relative ml-4 border-l border-separator-border">
                      {repo.chats.map((chat) => (
                        <li key={chat.name} className="relative">
                          <span
                            aria-hidden
                            className="absolute top-0 -left-px h-4 w-3 rounded-bl-lg border-b border-l border-separator-border"
                          />
                          {chat.href ? (
                            <a
                              href={chat.href}
                              aria-current={selected === chat.name ? "page" : undefined}
                              className={cx(
                                "-ml-4 flex h-8 items-center gap-2 rounded-2lg pr-2 pl-9 text-body-medium text-text-secondary hover:bg-background-secondary-hover",
                                selected === chat.name && "bg-background-tertiary-default",
                              )}
                            >
                              <span className="truncate">{chat.name}</span>
                              <span className="ml-auto rounded-sm bg-background-tertiary-default px-1 text-caption-1-regular">
                                {chat.time}
                              </span>
                            </a>
                          ) : (
                            <button
                              type="button"
                              onClick={() => setNotice(chat.name)}
                              className="-ml-4 flex h-8 w-[calc(100%+16px)] cursor-pointer items-center gap-2 rounded-2lg pr-2 pl-9 text-left text-body-medium text-text-secondary hover:bg-background-secondary-hover"
                            >
                              <span className="truncate">{chat.name}</span>
                              <span className="ml-auto rounded-sm bg-background-tertiary-default px-1 text-caption-1-regular">
                                {chat.time}
                              </span>
                            </button>
                          )}
                        </li>
                      ))}
                    </ul>
                  )}
                </div>
              );
            })}
            {rows.length === 0 && (
              <p className="p-2 text-body-regular text-text-tertiary">No repositories found.</p>
            )}
          </div>
        )}
      </div>
      <div className="flex flex-col gap-2 pt-4">
        {!collapsed && <ThemeToggle />}
        <a
          href="/foundations"
          aria-label="Support"
          className="flex h-8 items-center gap-2 overflow-hidden px-2 text-body-medium text-text-secondary"
        >
          <RiCustomerServiceLine className="size-4 shrink-0" />
          {!collapsed && "Support"}
        </a>
        <button
          type="button"
          onClick={() => setSettings(true)}
          aria-label="Settings"
          className="flex h-8 cursor-pointer items-center gap-2 overflow-hidden px-2 text-body-medium text-text-secondary"
        >
          <RiSettings4Line className="size-4 shrink-0" />
          {!collapsed && "Settings"}
        </button>
        <div className="flex items-center gap-2 rounded-2xl bg-background-tertiary-default p-2">
          <Avatar initials="B" color="blue" size="sm" />
          {!collapsed && (
            <>
              <div className="min-w-0 flex-1">
                <p className="text-body-medium text-text-primary">Board team</p>
                <p className="text-body-regular text-text-secondary">Pro Plan</p>
              </div>
              <Button variant="secondary" size="small" onClick={() => setSettings(true)}>
                Upgrade
              </Button>
            </>
          )}
        </div>
      </div>
      <SettingsModal isOpen={settings} onClose={() => setSettings(false)} />
      {notice && (
        <div className="absolute inset-3 z-30 flex flex-col justify-center gap-3 rounded-3xl border border-border-button-default bg-background-primary-default p-5 shadow-dropdown">
          <p className="text-headline-medium text-text-primary">{notice}</p>
          <p className="text-body-regular text-text-secondary">
            This workspace is ready for a new conversation.
          </p>
          <Button
            onClick={() => {
              setNotice(null);
              onNew();
            }}
          >
            Start a new agent
          </Button>
          <Button variant="secondary" onClick={() => setNotice(null)}>
            Back to workspace
          </Button>
        </div>
      )}
    </aside>
  );
}

export interface AgentWorkspaceProps {
  kind: "chat" | "image";
  children: ReactNode;
  panel: ReactNode;
  input: string;
  onInputChange: (value: string) => void;
  onSubmit: (value: string) => void;
  onStop: () => void;
  onNew: () => void;
  busy?: boolean;
  contained?: boolean;
  className?: string;
}

export function AgentWorkspace({
  kind,
  children,
  panel,
  input,
  onInputChange,
  onSubmit,
  onStop,
  onNew,
  busy,
  contained,
  className,
}: AgentWorkspaceProps) {
  const [collapsed, setCollapsed] = useState(false);
  const [navOpen, setNavOpen] = useState(false);
  const [panelOpen, setPanelOpen] = useState(false);
  const [copied, setCopied] = useState(false);
  const reduced = useReducedMotion();
  const scrollRef = useRef<HTMLDivElement>(null);
  const selected = kind === "chat" ? "coding scenario" : "image generation";
  useEffect(() => {
    const el = scrollRef.current;
    if (!el) return;
    const observer = new ResizeObserver(() => {
      el.scrollTop = el.scrollHeight;
    });
    const content = el.firstElementChild;
    if (content) observer.observe(content);
    return () => observer.disconnect();
  }, []);
  useEffect(() => {
    const listener = (event: KeyboardEvent) => {
      if (event.key === "Escape") {
        setNavOpen(false);
        setPanelOpen(false);
      }
    };
    window.addEventListener("keydown", listener);
    return () => window.removeEventListener("keydown", listener);
  }, []);
  const copy = async () => {
    try {
      await navigator.clipboard.writeText(scrollRef.current?.innerText ?? "");
      setCopied(true);
    } catch {
      setCopied(false);
    }
  };
  return (
    <div
      className={cx(
        "relative flex w-full gap-3 overflow-hidden bg-background-full p-3",
        contained ? "h-full min-h-[640px]" : "h-dvh",
        className,
      )}
    >
      <AgentRepositorySidebar
        selected={selected}
        onNew={onNew}
        collapsed={collapsed}
        onCollapse={() => setCollapsed((value) => !value)}
        className="relative z-10 hidden lg:flex"
      />
      <div className="absolute inset-y-3 left-3 lg:hidden" inert={!navOpen}>
        <AgentRepositorySidebar
          selected={selected}
          onNew={() => {
            onNew();
            setNavOpen(false);
          }}
          collapsed={false}
          onClose={() => setNavOpen(false)}
        />
      </div>
      <motion.section
        animate={{ x: navOpen ? 272 : 0 }}
        transition={{ duration: reduced ? 0 : 0.4, ease: [0.22, 1, 0.36, 1] }}
        className="relative z-20 flex h-full min-w-0 flex-1 flex-col overflow-hidden rounded-3xl bg-background-secondary-default lg:!transform-none"
      >
        <header className="absolute inset-x-0 top-0 z-10 flex h-12 items-center justify-between gap-2 px-4 text-body-2-medium text-text-secondary">
          <div className="flex min-w-0 items-center gap-2">
            <IconButton
              icon={navOpen ? RiCloseLine : RiMenuLine}
              size="small"
              aria-label="Open navigation"
              onClick={() => setNavOpen((value) => !value)}
              className="lg:hidden"
            />
            <RiFolder3Line className="hidden size-4 shrink-0 text-text-tertiary sm:block" />
            <span className="hidden truncate text-text-tertiary sm:inline">
              vibl coding project
            </span>
            <RiArrowRightSLine className="hidden size-4 shrink-0 sm:block" />
            <span className="truncate">{selected}</span>
          </div>
          <div className="flex shrink-0 items-center gap-1">
            <IconButton
              icon={copied ? RiCheckLine : RiShareBoxLine}
              size="small"
              aria-label="Copy conversation"
              onClick={copy}
            />
            <Dropdown>
              <DropdownTrigger
                aria-label="Conversation actions"
                className="flex size-7 items-center justify-center rounded-md hover:bg-background-secondary-hover"
              >
                <RiMoreFill className="size-4" />
              </DropdownTrigger>
              <DropdownPopover aria-label="Conversation actions">
                <DropdownItem onSelect={copy}>
                  <RiFileCopyLine className="size-4" />
                  Copy conversation
                </DropdownItem>
                <DropdownItem onSelect={onNew}>
                  <RiAddLine className="size-4" />
                  New agent
                </DropdownItem>
              </DropdownPopover>
            </Dropdown>
            <IconButton
              icon={kind === "chat" ? RiCodeBoxLine : RiGalleryLine}
              size="small"
              aria-label="Open side panel"
              className="xl:hidden"
              onClick={() => setPanelOpen(true)}
            />
          </div>
        </header>
        <div ref={scrollRef} className="min-h-0 flex-1 overflow-y-auto">
          <div className="flex min-h-full flex-col justify-end gap-4 px-4 pt-16 pb-3">
            {children}
          </div>
        </div>
        <div className="flex shrink-0 flex-col gap-2.5 px-2 pb-2">
          <Composer
            value={input}
            onValueChange={onInputChange}
            onSubmit={onSubmit}
            onStop={onStop}
            busy={busy}
          />
          <StatusBar className="px-1" />
        </div>
        {navOpen && (
          <button
            type="button"
            onClick={() => setNavOpen(false)}
            aria-label="Close navigation overlay"
            className="absolute inset-0 z-20 lg:hidden"
          />
        )}
      </motion.section>
      <aside className="hidden h-full w-[410px] shrink-0 flex-col overflow-hidden pt-2 xl:flex">
        {panel}
      </aside>
      {panelOpen && (
        <div className="absolute inset-0 z-40 xl:hidden">
          <button
            type="button"
            aria-label="Close panel backdrop"
            className="absolute inset-0 bg-black/40"
            onClick={() => setPanelOpen(false)}
          />
          <div className="absolute inset-y-0 right-0 flex w-[min(410px,calc(100%_-_12px))] flex-col bg-background-full p-3 shadow-sidebar">
            <div className="flex h-10 shrink-0 items-center justify-between px-1">
              <span className="text-headline-medium text-text-primary">
                {kind === "chat" ? "Code" : "Gallery"}
              </span>
              <IconButton
                icon={RiCloseLine}
                aria-label="Close panel"
                onClick={() => setPanelOpen(false)}
              />
            </div>
            <div className="flex min-h-0 flex-1 flex-col overflow-hidden pt-2">{panel}</div>
          </div>
        </div>
      )}
    </div>
  );
}
