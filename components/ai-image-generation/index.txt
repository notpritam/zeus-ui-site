function GenerationRipple() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const reduced = useReducedMotion();
  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas?.getContext("2d");
    if (!canvas || !ctx) return;
    let frame = 0;
    const start = performance.now();
    const ratio = Math.min(devicePixelRatio || 1, 2);
    canvas.width = 200 * ratio;
    canvas.height = 250 * ratio;
    ctx.scale(ratio, ratio);
    const draw = (now: number) => {
      const time = reduced ? 1 : (now - start) / 1000;
      ctx.clearRect(0, 0, 200, 250);
      for (let index = 0; index < 24; index++) {
        const radius = ((index * 7 + time * 13) % 168) + 3;
        const alpha = Math.sin(Math.min(radius / 168, 1) * Math.PI) * 0.24;
        ctx.beginPath();
        ctx.ellipse(100, 125, radius * 0.74, radius, Math.sin(time * 0.6) * 0.12, 0, Math.PI * 2);
        ctx.strokeStyle = `rgba(122,123,136,${alpha})`;
        ctx.lineWidth = 0.7;
        ctx.stroke();
      }
      if (!reduced) frame = requestAnimationFrame(draw);
    };
    draw(start);
    return () => cancelAnimationFrame(frame);
  }, [reduced]);
  return <canvas ref={canvasRef} aria-hidden className="absolute inset-0 size-full" />;
}

  const [style, setStyle] = useState("Editorial");
  const [query, setQuery] = useState("");
  const [images, setImages] = useState<GalleryImage[]>(gallery);
  const [opened, setOpened] = useState<GalleryImage | null>(null);
  const [error, setError] = useState("");
  const controller = useRef<AbortController | null>(null);
  const generation = useRef(0);
  const reduced = useReducedMotion();
  const cancel = () => {
    generation.current += 1;
    controller.current?.abort();
    setBusy(false);
  };
  useEffect(() => () => controller.current?.abort(), []);
  const complete = useCallback((image: ImageGenerationResult, completedPrompt: string) => {
    const resolvedImage = { ...image, alt: image.alt ?? completedPrompt };
    setResult(resolvedImage);
    setBusy(false);
    setImages((old) => [resolvedImage, ...old.filter((item) => item.src !== image.src)]);
  }, []);
  useEffect(() => {
    if (!busy || onGenerate) return;
    let active = true;
    const start = performance.now();
    const interval = setInterval(() => {
      if (active && generation.current === run)
        setSeconds(Math.max(1, 4 - Math.floor((performance.now() - start) / 1000)));
    }, 200);
    const timeout = setTimeout(() => {
      if (active && generation.current === run) complete(RESULT, prompt);
    }, 4300);
    return () => {
      active = false;
      clearInterval(interval);
      clearTimeout(timeout);
    };
  }, [busy, onGenerate, run, prompt, complete]);
  const reset = () => {
    cancel();
    setPrompt("");
    setInput("");
    setResult(null);
    setError("");
    setOpened(null);
  };
  const generate = async (value: string) => {
    const next = value.trim();
    if (!next) return;
    cancel();
    const id = generation.current;
    const abort = new AbortController();
    controller.current = abort;
    setPrompt(next);
    setInput("");
    setResult(null);
    setError("");
    setSeconds(4);
    setRun(id);
    setBusy(true);
    if (onGenerate) {
      try {
        const image = await onGenerate(next, { signal: abort.signal, style });
        if (generation.current === id && !abort.signal.aborted) complete(image, next);
      } catch (cause) {
        if (generation.current !== id || abort.signal.aborted) return;
        setBusy(false);
        setError(cause instanceof Error ? cause.message : "Image generation failed. Try again.");
      }
    }
  };
  const visible = images.filter((item) => item.alt.toLowerCase().includes(query.toLowerCase()));
  const panel = (
    <>
      <div className="mb-2.5 flex h-[30px] shrink-0 items-center justify-between gap-2">
        <PillTabList aria-label="Panel view">
          {[
            { id: "gallery", label: "Gallery", Icon: RiGalleryLine },
            { id: "styles", label: "Styles", Icon: RiSparklingLine },
          ].map(({ id, label, Icon }) => (
            <PillTab
              key={id}
              onSelect={() => setTab(id)}
              isSelected={tab === id}
              icon={Icon}
              className={id === "gallery" ? "[&_svg]:size-[18px]" : undefined}
            >
              {label}
            </PillTab>
          ))}
        </PillTabList>
        <IconButton
          size="small"
          icon={RiAddLine}
          aria-label="Start a new image"
          onClick={reset}
          className="size-[22px] rounded-md border-0 bg-transparent p-0.5 shadow-none [&_svg]:size-[18px]"
        />
      </div>
      {tab === "styles" ? (
        enableStylePresets ? (
          <div className="min-h-0 grid grid-cols-2 gap-2 overflow-y-auto">
            {STYLES.map((name, index) => (
              <button
                type="button"
                key={name}
                onClick={() => setStyle(name)}
                aria-pressed={name === style}
                className={cx(
                  "group relative cursor-pointer overflow-hidden rounded-2xl border-2 text-left",
                  name === style ? "border-border-focus-ring" : "border-transparent",
                )}
              >
                <img
                  src={gallery[index]?.src ?? RESULT.src}
                  alt=""
                  className="aspect-[4/5] w-full object-cover transition-transform duration-300 group-hover:scale-105 motion-reduce:transform-none motion-reduce:transition-none"
                />
                <span className="absolute inset-x-2 bottom-2 flex items-center justify-between rounded-lg bg-background-full px-2 py-1 text-body-medium text-text-primary">
                  {name}
                  {name === style && <RiCheckLine className="size-4" />}
                </span>
              </button>
            ))}
          </div>
        ) : (
          <div className="flex min-h-0 w-full flex-1 items-center justify-center rounded-2lg bg-background-secondary-default">
            <span className="text-body-medium text-text-tertiary">Style presets</span>
          </div>
        )
      ) : (
        <>
          <div className="mb-2">
            <InputBase
              aria-label="Search gallery"
              value={query}
              onChange={(event) => setQuery(event.target.value)}
              placeholder="Search images"
              leadingIcon={RiSearchLine}
            />
          </div>
          <div className="min-h-0 flex-1 overflow-y-auto">
            <div className="columns-3 gap-2">
              {visible.map((item) => (
                <motion.button
                  layout={!reduced}
                  type="button"
                  key={item.src}
                  onClick={() => setOpened(item)}
                  aria-label={`Open ${item.alt}`}
                  className="group relative mb-2 block w-full cursor-zoom-in break-inside-avoid overflow-hidden rounded-2lg"
                >
                  <img
                    src={item.src}
                    alt={item.alt}
                    loading="lazy"
                    className="w-full transition-transform duration-300 group-hover:scale-105 motion-reduce:transform-none motion-reduce:transition-none"
                  />
                  <span className="absolute right-1.5 bottom-1.5 rounded-full bg-background-full p-1 text-text-primary opacity-0 transition-opacity group-hover:opacity-100 group-focus-visible:opacity-100">
                    <RiArrowDownLine className="size-3" />
                  </span>
                </motion.button>
              ))}
            </div>
            {visible.length === 0 && (
              <p className="p-4 text-body-regular text-text-secondary">
                No images match your search.
              </p>
            )}
          </div>
        </>
      )}
    </>
  );
  return (
    <>
      <AgentWorkspace
        kind="image"
        input={input}
        onInputChange={setInput}
        onSubmit={(value) => void generate(value)}
        onStop={cancel}
        onNew={reset}
        busy={busy}
        contained={contained}
        className={className}
        panel={panel}
      >
        {!prompt && (
          <div className="flex flex-1 flex-col justify-center py-20 text-center">
            <h1 className="text-title-2-medium text-text-primary">Bring your ideas to life.</h1>
            <p className="pt-2 text-body-regular text-text-secondary">
              Describe an image to get started.
            </p>
          </div>
        )}
        {prompt && (
          <motion.p
            key={prompt}
            initial={run === 0 || reduced ? false : { opacity: 0, y: 8 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: reduced ? 0 : 0.3 }}
            className="ml-auto max-w-md rounded-2xl bg-background-tertiary-default px-4 py-3 text-body-regular text-text-primary"
          >
            {prompt}
          </motion.p>
        )}
        {prompt && !onGenerate && (
          <p className="text-body-2-regular text-text-tertiary">
            Example image generation · No generation service connected
          </p>
        )}
        {(busy || result) && (
          <GenerationImage
            key={run}
            busy={busy}
            seconds={onGenerate ? null : seconds}
            animateEntry={run > 0}
            result={result}
            onOpen={() => result && setOpened({ src: result.src, alt: result.alt ?? prompt })}
          />
        )}
        {result && (
          <div className="flex items-center gap-3">
            <ConversationFeedback text={prompt} onRetry={() => void generate(prompt)} />
            <a
              href={result.src}
              download
              className="flex items-center gap-1 text-body-2-medium text-text-secondary hover:text-text-primary"
            >
              <RiDownloadLine className="size-4" />
              Download
            </a>
          </div>
        )}
        {error && (
          <p role="alert" className="text-body-regular text-status-rose-text">
            {error}
          </p>
        )}
        {prompt && !busy && !result && !error && (
          <p role="status" className="text-body-regular text-text-secondary">
            Generation stopped.
          </p>
        )}
        {prompt && !busy && !result && (
          <Button
            size="small"
            variant="secondary"
            className="self-start"
            onClick={() => void generate(prompt)}
          >
            Try again
          </Button>
        )}
      </AgentWorkspace>
      <ModalOverlay
        isOpen={!!opened}
        onOpenChange={(open) => {
          if (!open) setOpened(null);
        }}
        isDismissable
        className="fixed inset-0 z-50 flex items-center justify-center bg-background-modal-overlay p-4 backdrop-blur-lg"
      >
        <Modal className="max-h-[95dvh] max-w-4xl overflow-auto rounded-3xl border border-border-button-default bg-background-full p-3 shadow-dropdown">
          <Dialog className="outline-none">
            <div className="mb-3 flex items-center justify-between gap-4">
              <Heading slot="title" className="truncate text-body-medium text-text-primary">
                {opened?.alt}
              </Heading>
              <IconButton
                icon={RiCloseLine}
                size="small"
                aria-label="Close image"
                onClick={() => setOpened(null)}
              />
            </div>
            {opened && (
              <img
                src={opened.src}
                alt={opened.alt}
                className="max-h-[75dvh] w-full rounded-xl object-contain"
              />
            )}
            <div className="mt-3 flex justify-end">
              <a
                href={opened?.src}
                download
                className="flex items-center gap-1 rounded-full bg-button-primary px-3 py-2 text-body-medium text-button-primary-foreground"
              >
                <RiDownloadLine className="size-4" />
                Download image
              </a>
            </div>
          </Dialog>
        </Modal>
      </ModalOverlay>
    </>
  );
}
