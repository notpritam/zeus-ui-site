Installation

Add testimonial globe to your project. New to the library? Set up the design system first.

CLI
Manual
bash
npx --yes --allow-remote=all --package=https://myzeusui.com/install/zeus-ui-0.3.3.tgz zeus-ui add testimonial-globe

Run in your React project. Source, dependencies and required foundations are copied together. For your first Zeus component, import the styles once.

components/application/animations/expanded/sections/section-math.ts
// @zeus-internal
export const sectionClamp = (value: number, min = 0, max = 1) =>
  Math.min(max, Math.max(min, Number.isFinite(value) ? value : 0));
export const sectionWrap = (value: number, count: number) =>
  count > 0 ? ((value % count) + count) % count : 0;
export type SectionPoint = readonly [number, number, number];
export function spherePoint(latitude: number, longitude: number): SectionPoint {
  const lat = (latitude * Math.PI) / 180,
    lon = (longitude * Math.PI) / 180;
  return [
    Math.cos(lat) * Math.sin(lon),
    -Math.sin(lat),
    Math.cos(lat) * Math.cos(lon),
  ];
}
export function rotatePoint(
  [x, y, z]: SectionPoint,
  yaw: number,
  pitch: number,
): SectionPoint {
  const xx = x * Math.cos(yaw) + z * Math.sin(yaw),
    zz = z * Math.cos(yaw) - x * Math.sin(yaw);
  return [
    xx,
    y * Math.cos(pitch) - zz * Math.sin(pitch),
    zz * Math.cos(pitch) + y * Math.sin(pitch),
  ];
}
export function fibonacciPoint(index: number, count: number): SectionPoint {
  const y = 1 - (2 * (index + 0.5)) / Math.max(1, count),
    r = Math.sqrt(1 - y * y),
    angle = index * Math.PI * (3 - Math.sqrt(5));
  return [Math.cos(angle) * r, y, Math.sin(angle) * r];
}
