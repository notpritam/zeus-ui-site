export const sectionClamp = (value: number, min = 0, max = 1) =>
  Math.min(max, Math.max(min, Number.isFinite(value) ? value : 0));
export const sectionWrap = (value: number, count: number) =>
  count > 0 ? ((value % count) + count) % count : 0;
export type SectionPoint = readonly [number, number, number];
export function spherePoint(latitude: number, longitude: number): SectionPoint {
  const lat = (latitude * Math.PI) / 180,
    lon = (longitude * Math.PI) / 180;
  return [
    Math.cos(lat) * Math.sin(lon),
    -Math.sin(lat),
    Math.cos(lat) * Math.cos(lon),
  ];
}
export function rotatePoint(
  [x, y, z]: SectionPoint,
  yaw: number,
  pitch: number,
): SectionPoint {
  const xx = x * Math.cos(yaw) + z * Math.sin(yaw),
    zz = z * Math.cos(yaw) - x * Math.sin(yaw);
  return [
    xx,
    y * Math.cos(pitch) - zz * Math.sin(pitch),
    zz * Math.cos(pitch) + y * Math.sin(pitch),
  ];
}
export function fibonacciPoint(index: number, count: number): SectionPoint {
  const y = 1 - (2 * (index + 0.5)) / Math.max(1, count),
    r = Math.sqrt(1 - y * y),
    angle = index * Math.PI * (3 - Math.sqrt(5));
  return [Math.cos(angle) * r, y, Math.sin(angle) * r];
}

export function useSectionEnvironment() {
  const ref = useRef<HTMLDivElement>(null),
    [size, setSize] = useState({ width: 1000, height: 640 }),
    [active, setActive] = useState(false),
    [reduced, setReduced] = useState(false);
  useEffect(() => {
    const node = ref.current;
    if (!node) return;
    let visible = true;
    const media = window.matchMedia("(prefers-reduced-motion: reduce)");
    const update = () => {
      setActive(visible && !document.hidden);
      setReduced(media.matches);
    };
    const resize = () =>
      setSize({
        width: node.clientWidth || 1000,
        height: node.clientHeight || 640,
      });
    const ro =
      typeof ResizeObserver === "undefined" ? null : new ResizeObserver(resize);
    ro?.observe(node);
    const io =
      typeof IntersectionObserver === "undefined"
        ? null
        : new IntersectionObserver((entries) => {
            visible = entries[0]?.isIntersecting ?? true;
            update();
          });
    io?.observe(node);
    media.addEventListener("change", update);
    document.addEventListener("visibilitychange", update);
    resize();
    update();
    return () => {
      ro?.disconnect();
      io?.disconnect();
      media.removeEventListener("change", update);
      document.removeEventListener("visibilitychange", update);
    };
  }, []);
  return { ref, size, active, reduced };
}
export function useSectionLoop(
  active: boolean,
  callback: (delta: number) => void,
) {
  const latest = useRef(callback);
  latest.current = callback;
  useEffect(() => {
    if (!active) return;
    let frame = 0,
      last = 0;
    const tick = (now: number) => {
      const delta = last ? Math.min(0.04, (now - last) / 1000) : 0;
      last = now;
      latest.current(delta);
      frame = requestAnimationFrame(tick);
    };
    frame = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(frame);
  }, [active]);
}
export function useSectionSelection(
  count: number,
  index: number | undefined,
  defaultIndex: number,
  onChange?: (index: number) => void,
) {
  const [local, setLocal] = useState(defaultIndex);
  const selected = sectionWrap(Math.round(index ?? local), count),
    current = useRef(selected);
  current.current = selected;
  const select = useCallback(
    (next: number) => {
      next = sectionWrap(Math.round(next), count);
      if (next === current.current) return;
      if (index === undefined) {
        current.current = next;
        setLocal(next);
      }
      onChange?.(next);
    },
    [count, index, onChange],
  );
  return { selected, select, current };
}
export function useSectionScroll(
  ref: RefObject<HTMLElement | null>,
  scrollContainer?: RefObject<HTMLElement | null>,
  onProgress?: (progress: number) => void,
) {
  const [progress, setProgress] = useState(0),
    latest = useRef(onProgress),
    previous = useRef(-1);
  latest.current = onProgress;

    const node = ref.current;
    if (!node) return;

    const measure = () => {
      frame = 0;

        top = container?.getBoundingClientRect().top ?? 0,
        view = container?.clientHeight ?? innerHeight;
      const next = sectionClamp(

      );
      if (Math.abs(previous.current - next) >= 0.0001) {
        previous.current = next;
        setProgress(next);
        latest.current?.(next);

    };
    const request = () => {
      if (!frame) frame = requestAnimationFrame(measure);

    const target = container ?? window;

      cancelAnimationFrame(frame);
      target.removeEventListener("scroll", request);
      window.removeEventListener("resize", request);
      ro?.disconnect();
    };
  }, [ref, scrollContainer]);
  return progress;
}
export function useSphereRotation(
  active: boolean,
  reduced: boolean,
  autoplay: boolean,
  speed: number,
  drag: boolean,
) {
  const state = useRef({
      yaw: 0,
      pitch: 0,
      vx: 0,
      vy: 0,
      target: null as null | { yaw: number; pitch: number },
    }),
    [rotation, setRotation] = useState({ yaw: 0, pitch: 0 }),
    pointer = useRef<null | { id: number; x: number; y: number; time: number }>(
      null,
    );
  const publish = () =>
    setRotation({ yaw: state.current.yaw, pitch: state.current.pitch });
  useSectionLoop(active && !reduced, (delta) => {
    const s = state.current;
    if (pointer.current) return;
    if (s.target) {
      const ease = 1 - Math.exp(-delta * 6);
      s.yaw += (s.target.yaw - s.yaw) * ease;
      s.pitch += (s.target.pitch - s.pitch) * ease;
      if (
        Math.abs(s.yaw - s.target.yaw) + Math.abs(s.pitch - s.target.pitch) <
        0.001
      )
        s.target = null;
    } else {
      s.yaw += (s.vx + (autoplay ? speed : 0)) * delta;
      s.pitch = sectionClamp(s.pitch + s.vy * delta, -1.2, 1.2);
      s.vx *= Math.exp(-delta * 5);
      s.vy *= Math.exp(-delta * 5);
    }
    publish();
  });
  const orient = useCallback(
    (yaw: number, pitch: number) => {
      const s = state.current;
      const shortest =
        ((((yaw - s.yaw + Math.PI) % (Math.PI * 2)) + Math.PI * 2) %
          (Math.PI * 2)) -
        Math.PI;
      s.vx = 0;
      s.vy = 0;
      if (reduced) {
        s.yaw += shortest;
        s.pitch = pitch;
        s.target = null;
        setRotation({ yaw: s.yaw, pitch: s.pitch });
      } else s.target = { yaw: s.yaw + shortest, pitch };
    },
    [reduced],
  );
  const end = (event: PointerEvent<HTMLElement>) => {
    if (pointer.current?.id !== event.pointerId) return;
    pointer.current = null;
    if (reduced) {
      state.current.vx = 0;
      state.current.vy = 0;
    }
    event.currentTarget.releasePointerCapture?.(event.pointerId);
  };
  return {
    rotation,
    orient,
    handlers: {
      onPointerDown: (event: PointerEvent<HTMLElement>) => {
        if (
          !drag ||
          event.button !== 0 ||
          (event.target as HTMLElement).closest("button,a,input")
        )
          return;
        pointer.current = {
          id: event.pointerId,
          x: event.clientX,
          y: event.clientY,
          time: event.timeStamp,
        };
        state.current.target = null;
        state.current.vx = state.current.vy = 0;
        event.currentTarget.setPointerCapture?.(event.pointerId);
      },
      onPointerMove: (event: PointerEvent<HTMLElement>) => {
        const p = pointer.current;
        if (!p || p.id !== event.pointerId) return;
        const dx = (event.clientX - p.x) * 0.006,
          dy = (event.clientY - p.y) * 0.006,
          dt = Math.max(0.008, (event.timeStamp - p.time) / 1000);
        state.current.yaw += dx;
        state.current.pitch = sectionClamp(state.current.pitch + dy, -1.2, 1.2);
        state.current.vx = sectionClamp(dx / dt, -5, 5);
        state.current.vy = sectionClamp(dy / dt, -3, 3);
        pointer.current = {
          id: p.id,
          x: event.clientX,
          y: event.clientY,
          time: event.timeStamp,
        };
        publish();
      },
      onPointerUp: end,
      onPointerCancel: end,
      onLostPointerCapture: () => {
        pointer.current = null;
      },
    },
  };
}

          aria-label="Previous testimonial"

              );
          }}
