Installation

Add most active days to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.2.1.tgz zeus-ui add most-active-days-card

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — activity-rings-card, chart-card, chevrons, chip, cx, segmented-control, select, tokens, use-count-up, use-dismiss-on-outside-press. The component source is yours; imported npm packages are installed as needed.

The complete example running in the preview. Adjust the props and content for your interface.

components/application/charts/most-active-days-card.tsx
"use client";
import { useEffect, useRef, useState } from "react";
import { cx } from "../../utils/cx";
import {
  ACTIVITY_RINGS,
  type ActivityRing,
  ActivityRings,
  ActivityRingsCard,
} from "./activity-rings-card";
import { ChartFrame, MONTHS, WeekSwitcher } from "./pro-chart";
export interface ActivityDay {
  date: string;
  steps: number;
  rings?: ActivityRing[];
}
export interface MostActiveDaysCardProps {
  title?: string;
  data?: ActivityDay[];
  selectedDate?: Date;
  defaultDate?: Date;
  onDateChange?: (date: Date) => void;
  showActivity?: boolean;
  className?: string;
  headline?: number;
}
const INITIAL_DATE = new Date(2026, 6, 10);
const dayKey = (date: Date) =>
  `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}-${String(date.getDate()).padStart(2, "0")}`;
export function MostActiveDaysCard({
  title = "Most active days",
  data,
  selectedDate,
  defaultDate = INITIAL_DATE,
  onDateChange,
  showActivity = true,
  className,
  headline = 32459,
}: MostActiveDaysCardProps = {}) {
  const [internal, setInternal] = useState(defaultDate),
    selected = selectedDate ?? internal,
    [month, setMonth] = useState(selected.getMonth()),
    scrollRef = useRef<HTMLDivElement>(null),
    year = selected.getFullYear();
  useEffect(() => {
    setMonth(selected.getMonth());
  }, [selected]);
  useEffect(() => {
    const target = scrollRef.current?.querySelector<HTMLElement>(`[data-month="${month}"]`);
    if (target && scrollRef.current) scrollRef.current.scrollTop = target.offsetTop;
  }, [month]);
  const select = (date: Date) => {
    setInternal(date);
    onDateChange?.(date);
  };
  const currentDay = data?.find((d) => d.date === dayKey(selected));
  return (
    <div className={cx("grid w-full min-w-0 gap-4", showActivity && "md:grid-cols-2", className)}>
      <ChartFrame
        title={title}
        value={currentDay?.steps ?? headline}
        suffix="total steps"
        height={330}
        className="rounded-[20px]"
        action={
          <WeekSwitcher
            label={new Date(year, month).toLocaleDateString("en-US", { month: "long" })}
            onPrevious={() => setMonth((m) => (m + 11) % 12)}
            onNext={() => setMonth((m) => (m + 1) % 12)}
          />
        }
      >
        <div
          ref={scrollRef}
          className="relative -mx-1.5 -mb-0.5 min-h-0 flex-1 overflow-y-auto rounded-2lg bg-background-inner-default px-2.5 [scrollbar-width:none]"
        >
          {MONTHS.map((label, m) => {
            const count = new Date(year, m + 1, 0).getDate(),
              offset = 0;
            return (
              <div key={label} data-month={m} className="pb-4">
                <p className="px-1 pt-2.5 pb-3 text-headline-medium text-text-primary">{label}</p>
                <div className="grid grid-cols-7 gap-0.5">
                  {Array.from({ length: offset }, (_, i) => (
                    <span key={`blank-${i}`} />
                  ))}
                  {Array.from({ length: count }, (_, i) => {
                    const date = new Date(year, m, i + 1),
                      key = dayKey(date),
                      entry = data?.find((d) => d.date === key),
                      isSelected = dayKey(selected) === key,
                      isFuture = date > selected,
                      rings =
                        entry?.rings ??
                        ACTIVITY_RINGS.map((r, n) => ({
                          ...r,
                          progress: isFuture
                            ? 0.03
                            : Math.max(0.12, ((i * 17 + n * 11 + m * 7) % 90) / 100),
                        }));
                    return (
                      <button
                        type="button"
                        key={key}
                        aria-label={date.toLocaleDateString("en-US", { dateStyle: "full" })}
                        aria-pressed={isSelected}
                        onClick={() => select(date)}
                        className={cx(
                          "flex h-[76px] min-w-0 flex-col items-center justify-between rounded-2lg border-2 py-2 text-body-regular text-text-primary outline-none focus-visible:ring-2 focus-visible:ring-border-focus-ring",
                          isSelected
                            ? "border-chart-cursor"
                            : "border-transparent hover:bg-background-secondary-default",
                        )}
                      >
                        <span>{i + 1}</span>
                        <ActivityRings rings={rings} className="size-7 max-h-7" animate={false} />
                      </button>
                    );
                  })}
                </div>
              </div>
            );
          })}
        </div>
      </ChartFrame>
      {showActivity && (
        <ActivityRingsCard
          title={`Activity for ${selected.toLocaleDateString("en-US", { month: "long", day: "numeric", year: "numeric" })}`}
          rings={currentDay?.rings}
          selectedDate={selected}
        />
      )}
    </div>
  );
}
