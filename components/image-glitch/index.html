Image Glitch

Analog scan lines, separated color channels and digital block tearing on hover

Installation

Add image glitch to your project. New to the library? Set up the design system first.

CLI
Manual
bash
npx --yes --allow-remote=all --package=https://myzeusui.com/install/zeus-ui-0.3.3.tgz zeus-ui add image-glitch

Run in your React project. Source, dependencies and required foundations are copied together. For your first Zeus component, import the styles once.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

image-glitch-demo.tsx
"use client";

import { ImageGlitch } from "@/components/application/animations/expanded/shaders/image-glitch";

// This example uses Zeus demo images. Install the component's assets
// or pass your own image URLs. Setup: https://myzeusui.com/components/image-glitch/#installation

export default function Example() {
  return (
    <ImageGlitch
      image={"/assets/annnimate/action_cyclist-climb_02.jpg"}
      variant={"analog"}
      intensity={1}
      speed={1}
      scale={3}
      duration={0.5}
    />
  );
}

components/application/animations/expanded/shaders/field-shaders.ts
// @zeus-internal
/** Independent effect algorithms; the renderer owns sizing, textures and lifecycle. */
export const fieldVertex = `attribute vec2 position;varying vec2 uv;void main(){uv=position*.5+.5;gl_Position=vec4(position,0.,1.);}`;
const common = `precision highp float;
varying vec2 uv;uniform vec2 resolution,imageSize,mouse;uniform sampler2D media;
uniform float time,intensity,scale,hover,progress,radius,mouseStrength,variant;
uniform vec3 colorA,colorB,colorC,colorD;uniform vec3 trail[24];
float hash(vec2 p){return fract(sin(dot(p,vec2(127.1,311.7)))*43758.5453);}
float noise(vec2 p){vec2 i=floor(p),f=fract(p);f=f*f*(3.-2.*f);return mix(mix(hash(i),hash(i+vec2(1.,0.)),f.x),mix(hash(i+vec2(0.,1.)),hash(i+vec2(1.,1.)),f.x),f.y);}
float fbm(vec2 p){float n=0.;float a=.5;for(int i=0;i<5;i++){n+=a*noise(p);p=p*2.03+7.1;a*=.5;}return n;}
vec2 cover(vec2 p){float a=resolution.x/resolution.y,b=imageSize.x/imageSize.y;return (p-.5)*vec2(min(a/b,1.),min(b/a,1.))+.5;}
vec4 photo(vec2 p){return texture2D(media,clamp(cover(p),vec2(.001),vec2(.999)));}
vec2 metric(vec2 p){return p*resolution/min(resolution.x,resolution.y);}
float bayer(vec2 p){vec2 a=mod(floor(p),4.);float x=a.x,y=a.y;float b=mod(x,2.)*2.+mod(y,2.)*3.;b=mod(b,4.);float c=mod(floor(x/2.),2.)*2.+mod(floor(y/2.),2.)*3.;return (b*4.+mod(c,4.)+.5)/16.;}
`;
export const fieldFragments = {
  mesh:
    common +
    `void main(){vec2 p=metric(uv)*scale;vec2 m=metric(mouse);p+=(p-m)*exp(-length(p-m)*2.)*hover*mouseStrength*.22;float t=time*.3;vec2 a=vec2(.2+sin(t)*.3,.3+cos(t*.8)*.4);vec2 b=vec2(.8+cos(t*.7)*.4,.1+sin(t*.5)*.4);vec2 c=vec2(.2+sin(t*.6)*.4,.9+cos(t*.9)*.3);vec2 d=vec2(.9+cos(t*.4)*.3,.8+sin(t)*.3);p+=vec2(sin(p.y*3.+t),cos(p.x*2.-t))*.12*intensity;vec4 logWeight=-3.*vec4(dot(p-a,p-a),dot(p-b,p-b),dot(p-c,p-c),dot(p-d,p-d));float strongest=max(max(logWeight.x,logWeight.y),max(logWeight.z,logWeight.w));vec4 weights=exp(logWeight-strongest);float wa=weights.x,wb=weights.y,wc=weights.z,wd=weights.w;vec3 col=(colorA*wa+colorB*wb+colorC*wc+colorD*wd)/(wa+wb+wc+wd);gl_FragColor=vec4(col,1.);}`,
  dither:
    common +
    `void main(){vec2 cells=floor(uv*resolution/max(scale,1.));vec2 sampleUv=(cells+.5)*max(scale,1.)/resolution;vec3 rgb=photo(sampleUv).rgb;float luminance=dot(rgb,vec3(.299,.587,.114));float threshold=bayer(cells);vec3 dots=mix(colorA,colorB,step(threshold,luminance));float distance=length(metric(uv-mouse));float lens=(1.-smoothstep(radius*.82,radius,distance))*hover;gl_FragColor=vec4(mix(dots,photo(uv).rgb,lens),1.);}`,
  glitch:
    common +
    `void main(){float t=floor(time*18.);float amount=intensity*(.045+hover);vec2 p=uv;float line=noise(vec2(floor(p.y*scale*18.),t));if(variant<.5){p.x+=sin(p.y*90.+time*10.)*.002*amount;p.x+=(line-.5)*.045*amount*step(.65,line);}else{float block=hash(vec2(floor(p.y*scale*9.),t));p.x+=(block-.5)*.18*amount*step(.72,block);p.y+=.008*amount*(hash(vec2(t,1.))-.5);}float shift=.009*amount;vec3 col=vec3(photo(p+vec2(shift,0.)).r,photo(p).g,photo(p-vec2(shift,0.)).b);float scan=.98+.02*sin(uv.y*resolution.y*1.4);float vignette=1.-.32*dot((uv-.5)*1.4,(uv-.5)*1.4);col*=mix(1.,scan*vignette,min(intensity,1.));gl_FragColor=vec4(col,1.);}`,
  dissolve:
    common +
    `void main(){float n=fbm(metric(uv)*scale);float p=clamp(progress,0.,1.)*1.16-.08;float edgeWidth=max(.003,.035*intensity);float visible=smoothstep(p,p+.008,n);float edge=visible*(1.-smoothstep(p+edgeWidth,p+edgeWidth*2.,n));vec3 col=mix(photo(uv).rgb,colorA,edge);gl_FragColor=vec4(col,visible);}`,
  gooey:
    common +
    `void main(){float field=0.;for(int i=0;i<24;i++){vec3 mark=trail[i];vec2 distance=metric(uv-mark.xy);field+=exp(-dot(distance,distance)/max(.0001,radius*radius)) * mark.z;}float revealed=smoothstep(.45,.8,field*intensity);revealed=max(revealed,progress);gl_FragColor=vec4(colorA,1.-revealed);}`,
  trail:
    common +
    `void main(){vec2 cells=floor(uv*resolution/max(scale,1.));vec2 p=(cells+.5)*max(scale,1.)/resolution;float field=0.;for(int i=0;i<24;i++){vec3 mark=trail[i];vec2 d=metric(p-mark.xy);field+=exp(-dot(d,d)/max(.0001,radius*radius))*mark.z*.28;}float alpha=step(bayer(cells),clamp(field*intensity,0.,1.));vec2 dotUv=fract(uv*resolution/max(scale,1.))-.5;alpha*=1.-smoothstep(.31,.43,length(dotUv));gl_FragColor=vec4(colorA,alpha);}`,
} as const;
