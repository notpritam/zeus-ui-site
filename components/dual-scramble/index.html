Installation

Add dual scramble to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.2.1.tgz zeus-ui add dual-scramble

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — tokens. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

dual-scramble-demo.tsx
"use client";

import { DualScramble } from "@/components/application/animations/dual-scramble";

export default function Demo() {
  return (
    <div className="w-full rounded-2xl bg-[#101010] p-8 text-[#fffaf2] sm:p-14">
      <p className="mb-8 text-xs uppercase tracking-widest text-white/50">Kinetic Athletics</p>
      <h2 className="text-4xl font-medium tracking-tight sm:text-6xl">
        <DualScramble text="Built for speed." trigger="load" duration={0.8} />
      </h2>
      <p className="mt-4 text-sm text-white/65">
        Carbon-plate running shoes engineered for race day.
      </p>
      <div className="mt-14 flex flex-wrap gap-6 text-lg">
        {["Running", "Cycling", "Training", "Apparel"].map((text) => (
          <DualScramble key={text} text={text} trigger="hover" />
        ))}
      </div>
      <div className="mt-12 border-t border-white/15 pt-8 font-mono text-3xl">
        <DualScramble text="42.2KM" trigger="click" />
        <p className="mt-2 text-xs text-white/45">Click to scramble</p>
      </div>
    </div>
  );
}

Source

1 file. Source code you can read, customize and keep in your project.

components/application/animations/dual-scramble.tsx
"use client";

// Adapted from the user-supplied Annnimate Starter Pack by Good Fella.
import { useRef, type CSSProperties } from "react";
import { gsap } from "gsap";
import { useGSAP } from "@gsap/react";
import { ScrollTrigger } from "gsap/dist/ScrollTrigger";
import { ScrambleTextPlugin } from "gsap/dist/ScrambleTextPlugin";

gsap.registerPlugin(useGSAP, ScrollTrigger, ScrambleTextPlugin);

export interface DualScrambleProps {
  text: string;
  className?: string;
  trigger?: "scroll" | "hover" | "click" | "load";
  duration?: number;
  chars?: string;
  accentColor?: string;
  finalColor?: string;
  start?: string;
  onClick?: () => void;
}

const css = `.z-scramble{position:relative;display:inline-grid;max-width:100%;border:0;background:none;padding:0;color:inherit;font:inherit;text-align:inherit;white-space:pre-wrap;overflow-wrap:anywhere}.z-scramble>span{grid-area:1/1}.z-scramble-spacer{visibility:hidden}.z-scramble-accent{color:var(--z-scramble-accent,#a4fb81)}.z-scramble-final{color:var(--z-scramble-final,currentColor)}button.z-scramble{cursor:pointer}button.z-scramble:focus-visible{outline:2px solid currentColor;outline-offset:5px}`;

export function DualScramble({
  text,
  className = "",
  trigger = "scroll",
  duration = 0.6,
  chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789#@$%&",
  accentColor = "#a4fb81",
  finalColor = "currentColor",
  start = "top 80%",
  onClick,
}: DualScrambleProps) {
  const root = useRef<HTMLSpanElement & HTMLButtonElement>(null);
  const layer = useRef<HTMLSpanElement>(null);
  useGSAP(
    (_, contextSafe) => {
      const el = root.current,
        target = layer.current;
      if (!el || !target) return;
      const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
      let timeline: gsap.core.Timeline | undefined;
      const play = contextSafe!(() => {
        timeline?.kill();
        target.textContent = text;
        if (mq.matches || !chars.length) return;
        const random = Array.from(text, (c) =>
          /\s/.test(c) ? c : (chars[Math.floor(Math.random() * chars.length)] ?? c),
        ).join("");
        timeline = gsap
          .timeline()
          .to(target, {
            duration,
            scrambleText: {
              text: random,
              chars,
              speed: 1,
              revealDelay: 0.1,
              oldClass: "z-scramble-final",
              newClass: "z-scramble-accent",
            },
            ease: "none",
          })
          .to(
            target,
            {
              duration,
              scrambleText: {
                text,
                chars,
                speed: 1,
                revealDelay: 0.1,
                oldClass: "z-scramble-accent",
                newClass: "z-scramble-final",
              },
              ease: "none",
            },
            duration / Math.max(1, text.replace(/\s/g, "").length),
          );
      });
      if (trigger === "load") play();
      const scroll =
        trigger === "scroll"
          ? ScrollTrigger.create({ trigger: el, start, once: true, onEnter: play })
          : undefined;
      const events =
        trigger === "hover" ? ["pointerenter", "focus"] : trigger === "click" ? ["click"] : [];
      events.forEach((event) => el.addEventListener(event, play));
      return () => {
        timeline?.kill();
        scroll?.kill();
        events.forEach((event) => el.removeEventListener(event, play));
        target.textContent = text;
      };
    },
    { scope: root, dependencies: [text, trigger, duration, chars, start], revertOnUpdate: true },
  );
  const Tag = trigger === "hover" || trigger === "click" ? "button" : "span";
  return (
    <>
      <style>{css}</style>
      <Tag
        ref={root}
        type={Tag === "button" ? "button" : undefined}
        aria-label={text}
        onClick={onClick}
        className={`z-scramble ${className}`}
        style={
          { "--z-scramble-accent": accentColor, "--z-scramble-final": finalColor } as CSSProperties
        }
      >
        <span className="z-scramble-spacer" aria-hidden="true">
          {text}
        </span>
        <span ref={layer} aria-hidden="true">
          {text}
        </span>
      </Tag>
    </>
  );
}
