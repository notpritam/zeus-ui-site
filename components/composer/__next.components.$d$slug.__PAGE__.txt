                      className="w-0.5 rounded-full bg-foreground-icon-primary"
                      animate={{ height: reduce ? height : [height, 5, height] }}
                      transition={{ repeat: Infinity, duration: 0.6 + index * 0.1 }}
                    />
                  ))}
                </span>
              ) : (
                <RiMic2Line className="size-5 text-foreground-icon-primary" aria-hidden />
              )}
            </button>
          </GlassSurface>
          <button
            type="button"
            aria-label={busy ? "Stop generating" : "Send message"}
            onClick={busy ? onStop : submit}
            className={cx(
              "flex size-9 cursor-pointer items-center justify-center rounded-full bg-button-primary p-2 outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-border-focus-ring",
              glass && "bg-transparent! text-foreground-icon-primary",
            )}
          >
            {busy ? (
              <RiStopFill className="size-5" aria-hidden />
            ) : (
              <RiArrowUpLine className="size-5 text-text-white" aria-hidden />
            )}
          </button>
        </div>
      </div>
    </div>
  );
}

        <span
          aria-hidden
          className="pointer-events-none absolute inset-0 rounded-full bg-text-primary opacity-10 blur-[50px] dark:hidden"
        />
        <ComposerLoader active={active} surface={active}>

            {...props}
            busy={active}

              props.onSubmit?.(value);
              if (controlledBusy === undefined) setBusy(true);
            }}
            onStop={() => {
              props.onStop?.();
              setBusy(false);
            }}
          />
        </ComposerLoader>
      </div>
      <StatusBar />
    </div>
  );
