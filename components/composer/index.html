Composer

The AI chat composer for React + Tailwind CSS: attachment and model menus, voice and send controls, status bar, and a liquid-glass working state
AI & agentsZeusBlocks
Preview
Code
57%

Installation

Add composer to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.2.2.tgz zeus-ui add composer

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — agent-thinking, composer-loader, cx, dropdown, input, radio, select, tokens, use-dismiss-on-outside-press. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

composer-demo.tsx
"use client";
import { AiChatComposerPreview } from "@/components/application/composer/composer";
export default function ComposerDemo() {
  return (
    <div className="flex min-h-[260px] w-full items-center justify-center">
      <AiChatComposerPreview />
    </div>
  );
}

  children,
  className,
}: {

function ComposerPill({
  value: controlled,
  onValueChange,
  onSubmit,
  onStop,
  busy = false,
  model: controlledModel,
  models = COMPOSER_MODELS,
  onModelChange,
  onFilesChange,
  onToolSelect,
  showToolSelection = false,
  onVoiceToggle,
  placeholder,
  className,
  glass = false,
}: ComposerProps & { glass?: boolean }) {
  const [input, setInput] = useState("");
  const [model, setModel] = useState("fable-5");
  const [attachmentOpen, setAttachmentOpen] = useState(false);
  const [modelOpen, setModelOpen] = useState(false);
  const [effort, setEffort] = useState(1);
  const [listening, setListening] = useState(false);
  const [attachments, setAttachments] = useState<string[]>([]);
  const [tool, setTool] = useState<string | null>(null);
  const [compactPlaceholder, setCompactPlaceholder] = useState(false);
  const fileRef = useRef<HTMLInputElement>(null);
  const reduce = useReducedMotion() === true;
  const inputId = useId();
  const value = controlled ?? input;
  const selectedModel = controlledModel ?? model;
  const modelLabel = models.find((item) => item.id === selectedModel)?.label ?? selectedModel;
  useEffect(() => {
    if (typeof window.matchMedia !== "function") return;
    const query = window.matchMedia("(max-width: 639px)");
    const updatePlaceholder = () => setCompactPlaceholder(query.matches);
    updatePlaceholder();
    query.addEventListener("change", updatePlaceholder);
    return () => query.removeEventListener("change", updatePlaceholder);
  }, []);
  function update(value: string) {
    if (controlled === undefined) setInput(value);
    onValueChange?.(value);
  }
  function submit() {
    if (!value.trim() || busy) return;
    onSubmit?.(value.trim());
    if (controlled === undefined) setInput("");
  }
  function chooseTool(id: string) {
    setTool(id);
    setAttachmentOpen(false);
    onToolSelect?.(id);
  }
  const glassControl = glass && "border-transparent! bg-transparent! shadow-none!";
  return (
    <div className={cx("relative w-full", className)}>
      <AnimatePresence>
        {((showToolSelection && tool) || attachments.length > 0) && (
          <motion.div
            initial={{ opacity: 0, height: 0 }}
            animate={{ opacity: 1, height: "auto" }}
            exit={{ opacity: 0, height: 0 }}
            className="flex flex-wrap gap-1.5 overflow-hidden pb-2"
          >
            {[
              ...(showToolSelection && tool ? [{ name: tool, isTool: true }] : []),
              ...attachments.map((name) => ({ name, isTool: false })),
            ].map(({ name, isTool }, index) => (
              <button
                type="button"
                key={`${index}-${name}`}
                onClick={() => {
                  if (isTool) setTool(null);
                  else setAttachments((names) => names.filter((item) => item !== name));
                }}
                aria-label={`Remove ${name}`}
                className="rounded-md bg-background-secondary-default px-2 py-1 text-caption-1-medium text-text-secondary"
              >
                {name} <span aria-hidden>×</span>
              </button>
            ))}
          </motion.div>
        )}
      </AnimatePresence>
      <div
        className={cx(
          "relative flex h-[52px] w-full items-center gap-2.5 rounded-full p-2",
          glass ? "bg-transparent" : "bg-background-primary-default shadow-xs",
        )}
      >
        <input
          ref={fileRef}
          type="file"
          multiple
          className="hidden"
          aria-label="Attach files"
          onChange={(event) => {
            const files = Array.from(event.target.files ?? []);
            setAttachments((names) => [...names, ...files.map((file) => file.name)]);
            onFilesChange?.(files);
            event.target.value = "";
          }}
        />
        <Dropdown isOpen={attachmentOpen} onOpenChange={setAttachmentOpen}>
          <GlassSurface active={glass}>
            <DropdownTrigger
              aria-label="Add attachment"
              className={cx(
                "flex size-9 items-center justify-center rounded-full bg-ai-chat-composer-add-background p-2 transition-colors duration-150 hover:bg-ai-chat-composer-add-hover-background",
                glassControl,
              )}
            >
              <RiAddLine
                aria-hidden
                className={cx(
                  "size-5 text-foreground-icon-primary transition-transform duration-200",
                  attachmentOpen && "rotate-45",
                )}
              />
            </DropdownTrigger>
          </GlassSurface>
          <DropdownPopover
            aria-label="Add attachment or tool"
            placement="top start"
            offset={8}
            className="w-[361px] p-2"
            dialogClassName="gap-2"
          >
            <DropdownGroup label="Add">
              <DropdownItem
                className="px-2 py-1.5"
                onSelect={() => {
                  setAttachmentOpen(false);
                  fileRef.current?.click();
                }}
              >
                <RiAttachment2 className="size-5 shrink-0 text-foreground-icon-secondary" />
                <span className="truncate text-body-medium whitespace-nowrap">
                  Files and folders
                </span>
              </DropdownItem>
              <DropdownItem className="px-2 py-1.5" onSelect={() => chooseTool("Goal")}>
                <RiFocus3Line className="size-5 shrink-0 text-foreground-icon-secondary" />
                <span className="truncate text-body-medium whitespace-nowrap">
                  <span>Goal</span>
                  <span className="ml-1.5 text-text-secondary">Set a goal for faster results</span>
                </span>
              </DropdownItem>
              <DropdownItem className="px-2 py-1.5" onSelect={() => chooseTool("Plan mode")}>
                <RiListCheck3 className="size-5 shrink-0 text-foreground-icon-secondary" />
                <span className="truncate text-body-medium whitespace-nowrap">
                  <span>Plan mode</span>
                  <span className="ml-1.5 text-text-secondary">Manage complex tasks</span>
                </span>
              </DropdownItem>
            </DropdownGroup>
            <DropdownGroup label="Plugins">
              {PLUGINS.map((plugin) => (
                <DropdownItem
                  key={plugin.id}
                  onSelect={() => chooseTool(plugin.label)}
                  className="px-2 py-1.5"
                >
                  <img
                    src={plugin.asset}
                    alt=""
                    aria-hidden
                    className={cx("size-6 shrink-0", plugin.id === "documents" && "dark:hidden")}
                  />
                  {plugin.id === "documents" && (
                    <img
                      src="/ai-chat/plugin-documents-dark.svg"
                      alt=""
                      aria-hidden
                      className="hidden size-6 shrink-0 dark:block"
                    />
                  )}
                  <span className="truncate text-body-medium whitespace-nowrap text-text-primary">
                    {plugin.label}
                    <span className="ml-1.5 text-text-secondary">{plugin.note}</span>
                  </span>
                </DropdownItem>
              ))}
            </DropdownGroup>
          </DropdownPopover>
        </Dropdown>
        <TextField
          aria-label="Message"
          value={value}
          onChange={update}
          className="min-w-0 flex-1"
          onKeyDown={(event) => {
            if (event.key === "Enter" && !event.shiftKey && !event.nativeEvent.isComposing) {
              event.preventDefault();
              submit();
            }
          }}
        >
          <InputBase
            id={inputId}
            placeholder={
              listening
                ? "Listening…"
                : (placeholder ?? (compactPlaceholder ? "Ask me" : "Ask me anything"))
            }
            fieldClassName="rounded-none bg-transparent p-0 ring-0"
            className="h-5 pl-0 caret-accent-500 focus:placeholder:text-text-tertiary [&:not(:focus)]:[mask-image:linear-gradient(to_right,#000_calc(100%-32px),transparent_100%)]"
          />
        </TextField>
        <Dropdown isOpen={modelOpen} onOpenChange={setModelOpen}>
          <GlassSurface active={glass} className="rounded-xl">
            <DropdownTrigger
              aria-label={`Select model: ${modelLabel}`}
              className={cx(
                "flex h-8 items-center justify-center gap-0.5 rounded-xl bg-background-primary-default py-1.5 pr-1 pl-2 hover:bg-background-primary-hover",
                glassControl,
              )}
            >
              <span className="max-w-[110px] truncate px-0.5 text-body-medium text-text-secondary">
                {modelLabel}
              </span>
              <RiArrowDownSLine
                aria-hidden
                className={cx(
                  "size-[18px] text-foreground-icon-secondary transition-transform duration-200",
                  modelOpen && "rotate-180",
                )}
              />
            </DropdownTrigger>
          </GlassSurface>
          <DropdownPopover
            aria-label="Models and effort"
            placement="top start"
            offset={8}
            className="w-[266px]"
            dialogClassName="gap-0"
          >
            <DropdownGroup label="Models">
              <div role="radiogroup" aria-label="Model" className="flex flex-col gap-1">
                {models.map((item) => (
                  <button
                    type="button"
                    role="radio"
                    key={item.id}
                    aria-checked={item.id === selectedModel}
                    className={cx(
                      "flex w-full cursor-pointer items-center justify-between gap-2.5 rounded-2lg p-2 outline-none transition-colors hover:bg-background-primary-hover focus-visible:bg-background-primary-hover",
                      item.id === selectedModel && "bg-background-primary-hover",
                    )}
                    onClick={() => {
                      setModel(item.id);
                      onModelChange?.(item.id);
                      setModelOpen(false);
                    }}
                  >
                    <span className="truncate text-body-medium whitespace-nowrap text-text-primary">
                      {item.label}
                    </span>
                    <RadioDot selected={item.id === selectedModel} />
                  </button>
                ))}
              </div>
            </DropdownGroup>
            <DropdownDivider className="mt-[7px] mb-3" />
            <div className="flex w-full flex-col">
              <div className="pl-2 text-body-medium text-text-secondary">
                Effort{" "}
                <motion.span
                  key={effort}
                  initial={{ opacity: 0, filter: "blur(4px)" }}
                  animate={{ opacity: 1, filter: "blur(0px)" }}
                  transition={{ duration: 0.35, ease: "easeOut" }}
                  className="inline-block text-text-primary"
                >
                  {EFFORT_LEVELS[effort]}
                </motion.span>
              </div>
              <div className="flex justify-between px-2 pt-2 pb-[3px] text-body-2-medium text-text-secondary">
                <span>Faster</span>
                <span>Smarter</span>
              </div>
              <div className="mt-1 w-full px-2 pb-2">
                <AriaSlider
                  aria-label="Effort"
                  value={effort}
                  onChange={setEffort}
                  minValue={0}
                  maxValue={5}
                  step={1}
                  className="w-full"
                >
                  <div className="relative h-[27px] w-full overflow-hidden rounded-lg bg-background-secondary-default">
                    <div
                      aria-hidden
                      className="absolute inset-y-0 left-0 rounded-lg bg-background-tertiary-hover transition-[width] duration-150 ease-out"
                      style={{ width: `calc(${effort * 20}% + ${21 * (1 - effort / 5)}px)` }}
                    />
                    <div
                      aria-hidden
                      className="absolute inset-x-[9px] top-[7px] flex h-[13px] items-center justify-between"
                    >
                      {EFFORT_LEVELS.map((level, index) => (
                        <span
                          key={level}
                          className="h-full w-[3px] rounded-[2px] bg-foreground-icon-tertiary"
                          style={{ opacity: index <= effort ? 1 : 0.3 }}
                        />
                      ))}
                    </div>
                    <div className="absolute inset-x-[10.5px] inset-y-0">
                      <SliderTrack className="h-full w-full">
                        <SliderThumb className="top-1/2 h-[27px] w-[21px] cursor-grab rounded-[7px] border border-border-checkbox-default bg-background-primary-default shadow-xs outline-none transition-shadow data-[dragging]:cursor-grabbing data-[focus-visible]:ring-2 data-[focus-visible]:ring-border-focus-ring" />
                      </SliderTrack>
                    </div>
                  </div>
                </AriaSlider>
              </div>
            </div>
          </DropdownPopover>
        </Dropdown>
        <div className="flex shrink-0 items-center gap-2 pl-1.5">
          <GlassSurface active={glass}>
            <button
              type="button"
              aria-label={listening ? "Stop voice input" : "Voice input"}
              aria-pressed={listening}
              onClick={() => {
                setListening(!listening);
                onVoiceToggle?.(!listening);
              }}
              className={cx(
                "flex size-9 cursor-pointer items-center justify-center rounded-full border border-border-button-default bg-background-primary-default p-2 shadow-xs outline-none hover:bg-background-primary-hover focus-visible:ring-2 focus-visible:ring-border-focus-ring",
                glassControl,
              )}
            >
              {listening ? (
                <span className="flex h-5 items-center gap-0.5" aria-hidden>
                  {[6, 13, 19, 11, 7].map((height, index) => (
                    <motion.span
                      key={index}
                      className="w-0.5 rounded-full bg-foreground-icon-primary"
                      animate={{ height: reduce ? height : [height, 5, height] }}
                      transition={{ repeat: Infinity, duration: 0.6 + index * 0.1 }}
                    />
                  ))}
                </span>
              ) : (
                <RiMic2Line className="size-5 text-foreground-icon-primary" aria-hidden />
              )}
            </button>
          </GlassSurface>
          <button
            type="button"
            aria-label={busy ? "Stop generating" : "Send message"}
            onClick={busy ? onStop : submit}
            className={cx(
              "flex size-9 cursor-pointer items-center justify-center rounded-full bg-button-primary p-2 outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-border-focus-ring",
              glass && "bg-transparent! text-foreground-icon-primary",
            )}
          >
            {busy ? (
              <RiStopFill className="size-5" aria-hidden />
            ) : (
              <RiArrowUpLine className="size-5 text-text-white" aria-hidden />
            )}
          </button>
        </div>
      </div>
    </div>
  );
}

              >
                <span className="flex-1">{value}</span>
                {value === activeMode && <RiCheckLine className="size-4" />}
              </DropdownItem>
            ))}
          </DropdownPopover>
        </Dropdown>
        <span
          aria-label={`Context window ${percent}% used`}
          title={`${percent}% of the context window used`}
          className="flex items-center gap-1 rounded-full bg-background-tertiary-default py-1 pr-2 pl-1.5"
        >
          <svg
            width="16"
            height="16"
            viewBox="0 0 16 16"
            fill="none"
            className="shrink-0 -rotate-90"
            aria-hidden
          >
            <circle cx="8" cy="8" r="6" stroke="var(--color-neutral-300)" strokeWidth="2.5" />
            <circle
              cx="8"
              cy="8"
              r="6"
              stroke="var(--color-neutral-500)"
              strokeWidth="2.5"
              strokeDasharray={`${(percent / 100) * contextCircumference} ${contextCircumference}`}
              strokeLinecap="round"
            />
          </svg>
          <span className="text-body-2-medium whitespace-nowrap text-text-secondary">
            {`${Math.round(percent)}%`}
          </span>
        </span>
      </div>
    </div>
  );
}

export function AiChatComposerPreview({ busy: controlledBusy, ...props }: ComposerProps = {}) {
  const [busy, setBusy] = useState(false);
  const active = controlledBusy ?? busy;
  useEffect(() => {
    if (!busy || controlledBusy !== undefined) return;
    const timer = window.setTimeout(() => setBusy(false), 4500);
    return () => window.clearTimeout(timer);
  }, [busy, controlledBusy]);
  return (
    <div className="flex w-full max-w-[560px] flex-col gap-2.5">
      <AnimatePresence>
        {active && (
          <motion.div
            initial={{ opacity: 0, height: 0 }}
            animate={{ opacity: 1, height: "auto" }}
            exit={{ opacity: 0, height: 0 }}
          >
            <AgentThinking variant="infinity" />
          </motion.div>
        )}
      </AnimatePresence>
      <div className="relative">
        <span
          aria-hidden
          className="pointer-events-none absolute inset-0 rounded-full bg-text-primary opacity-10 blur-[50px] dark:hidden"
        />
        <ComposerLoader active={active} surface={active}>
          <GlassComposer
            {...props}
            busy={active}
            glass={active}
            onSubmit={(value) => {
              props.onSubmit?.(value);
              if (controlledBusy === undefined) setBusy(true);
            }}
            onStop={() => {
              props.onStop?.();
              setBusy(false);
            }}
          />
        </ComposerLoader>
      </div>
      <StatusBar />
    </div>
  );
}
