import { type CSSProperties, type ReactNode, useEffect, useRef } from "react";

export interface CardFlip3DProps {
  children?: ReactNode;
  image?: string;
  alt?: string;
  maxTilt?: number;
  perspective?: number;
  scale?: number;
  duration?: number;
  glare?: boolean;
  shadow?: boolean;
  inverse?: boolean;
  className?: string;
  style?: CSSProperties;
  cardStyle?: CSSProperties;
}
/** A perspective card whose tilt, glare and shadow follow the pointer with a damped return. */
export function CardFlip3D({
  children,
  image,
  alt = "",
  maxTilt = 20,
  perspective = 1000,
  scale = 1.03,
  duration = 0.6,
  glare = true,
  shadow = true,
  inverse = false,
  className,
  style,
  cardStyle,
}: CardFlip3DProps) {
  const host = useRef<HTMLDivElement>(null),
    card = useRef<HTMLDivElement>(null),
    shine = useRef<HTMLDivElement>(null);
  const latest = useRef({ maxTilt, scale, duration, shadow, inverse });
  latest.current = { maxTilt, scale, duration, shadow, inverse };
  useEffect(() => {
    const root = host.current,
      face = card.current,
      reflection = shine.current;
    if (!root || !face || !reflection) return;
    const reduced = matchMedia("(prefers-reduced-motion: reduce)"),
      coarse = matchMedia("(pointer: coarse)");
    let frame = 0,
      last = 0,
      x = 0,
      y = 0,
      zoom = 1,
      hover = 0,
      tx = 0,
      ty = 0,
      targetHover = 0,
      gx = 50,
      gy = 50;
    const draw = (now: number) => {
      frame = 0;
      const p = latest.current,
        dt = Math.min(64, last ? now - last : 16);
      last = now;
      const ease = reduced.matches ? 1 : 1 - Math.exp(-dt / Math.max(1, p.duration * 220));
      x += (tx - x) * ease;
      y += (ty - y) * ease;
      hover += (targetHover - hover) * ease;
      const targetZoom = 1 + (Math.max(0.5, p.scale) - 1) * targetHover;
      zoom += (targetZoom - zoom) * ease;
      face.style.transform = `rotateX(${x}deg) rotateY(${y}deg) scale(${zoom})`;
      face.style.boxShadow = p.shadow
        ? `${-y * 0.5}px ${18 + x * 0.5}px ${28 + hover * 20}px rgba(0,0,0,${0.08 + hover * 0.1})`
        : "none";
      reflection.style.opacity = `${hover * 0.45}`;
      reflection.style.background = `radial-gradient(circle at ${gx}% ${gy}%, #fff8, transparent 65%)`;
      if (
        Math.abs(tx - x) +
          Math.abs(ty - y) +
          Math.abs(targetZoom - zoom) +
          Math.abs(targetHover - hover) >
        0.005
      )

    };

      if (!frame) frame = requestAnimationFrame(draw);
    };
    const move = (event: PointerEvent) => {
      if (reduced.matches || coarse.matches || event.pointerType === "touch") return;
      const rect = root.getBoundingClientRect(),
        p = latest.current;
      const px = Math.max(-1, Math.min(1, ((event.clientX - rect.left) / rect.width) * 2 - 1)),
        py = Math.max(-1, Math.min(1, ((event.clientY - rect.top) / rect.height) * 2 - 1));
      const tilt = Math.max(0, Math.min(60, p.maxTilt)) * (p.inverse ? -1 : 1);
      tx = -py * tilt;
      ty = px * tilt;
      gx = (px + 1) * 50;
      gy = (py + 1) * 50;
      targetHover = 1;
      schedule();
    };
    const leave = () => {
      tx = ty = targetHover = 0;
      schedule();
    };
    const reset = () => {
      cancelAnimationFrame(frame);
      frame = 0;
      last = 0;
      tx = ty = x = y = hover = targetHover = 0;
      zoom = 1;
      face.style.transform = "rotateX(0deg) rotateY(0deg) scale(1)";
      reflection.style.opacity = "0";
      face.style.boxShadow = latest.current.shadow ? "0 18px 28px #0001" : "none";
    };
    const visibility = () => {
      if (document.hidden) reset();
    };
    const observer = new IntersectionObserver((entries) => {
      if (!entries[0]?.isIntersecting) reset();
    });
    observer.observe(root);
    root.addEventListener("pointermove", move);
    root.addEventListener("pointerleave", leave);
    reduced.addEventListener("change", reset);
    coarse.addEventListener("change", reset);
    document.addEventListener("visibilitychange", visibility);
    return () => {
      cancelAnimationFrame(frame);
      observer.disconnect();
      root.removeEventListener("pointermove", move);
      root.removeEventListener("pointerleave", leave);
      reduced.removeEventListener("change", reset);
      coarse.removeEventListener("change", reset);
      document.removeEventListener("visibilitychange", visibility);
    };
  }, []);
  return (
    <div
      ref={host}
      data-card-flip-3d
      className={className}
      style={{ perspective, width: "min(400px, 80%)", aspectRatio: "3 / 4", ...style }}
    >
      <div
        ref={card}
        data-tilt-face
        style={{
          position: "relative",
          width: "100%",
          height: "100%",
          borderRadius: 8,
          transformStyle: "preserve-3d",
          background: "#161616",
          color: "#fff",
          boxShadow: shadow ? "0 18px 28px #0001" : "none",
          ...cardStyle,
        }}
      >
        {image && (
          <img
            src={image}
            alt={alt}
            draggable={false}
            style={{
              width: "100%",
              height: "100%",
              position: "absolute",
              inset: 0,
              objectFit: "cover",
              borderRadius: "inherit",
            }}
          />
        )}
        {children && (
          <div style={{ position: "relative", transform: "translateZ(24px)", height: "100%" }}>
            {children}
          </div>
        )}
        <div
          ref={shine}
          aria-hidden
          style={{
            position: "absolute",
            inset: 0,
            borderRadius: "inherit",
            pointerEvents: "none",
            opacity: 0,
            display: glare ? "block" : "none",
          }}
        />
      </div>
    </div>
  );
