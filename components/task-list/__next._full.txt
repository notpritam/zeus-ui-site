/** `revealed` counts one unit for each task header and each child step. */
export function TaskList({
  tasks = DEFAULT_AGENT_TASKS,
  revealed: controlled,
  run = true,
  startDelay = 320,
  stepInterval = 850,
  collapseOnComplete = false,
  onComplete,
  className,
}: TaskListProps = {}) {
  const reduce = useLogMotion();
  const total = tasks.reduce((sum, task) => sum + 1 + (task.steps?.length ?? 0), 0);
  const timedTotal = total + (controlled === undefined && total > 0 ? 1 : 0);
  const revealed = useRevealTicker({
    total: timedTotal,
    run,
    startDelay,
    stepInterval,
    revealed: controlled,
    onComplete,
  });
  const complete = revealed >= timedTotal;
  let offset = 0;
  return (
    <div className={cx("flex w-full flex-col gap-[5px]", className)} aria-label="Agent task log">
      {tasks.map((task, index) => {
        const start = offset;
        const units = 1 + (task.steps?.length ?? 0);
        offset += units;
        if (revealed <= start) return null;
        const done =
          controlled === undefined ? revealed > start + units : revealed >= start + units;
        return (
          <TaskGroup
            key={task.id ?? `${index}-${task.label}`}
            task={task}
            shown={Math.max(0, revealed - start - 1)}
            done={done}
            reduce={reduce}
            collapse={collapseOnComplete === "all" ? complete : collapseOnComplete && done}
          />
        );
      })}
      {!complete && <WorkingRow label="Working" reduce={reduce} />}
    </div>
  );
}
