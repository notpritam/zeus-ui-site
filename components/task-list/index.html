Task List

Streaming agent task log for React and Tailwind CSS. Tasks reveal step by step with soft height, blur, and shimmer, and drop straight into a chat thread.
Preview
Code
Working0.0s

Installation

Add task list to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.1.0.tgz zeus-ui add task-list

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — agent-log, agent-thinking, cx, tokens. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

task-list-demo.tsx
"use client";
import { TaskList } from "@/components/application/task-list/task-list";
import { RiRefreshLine } from "@remixicon/react";
import { useState } from "react";
export default function TaskListDemo() {
  const [turn, setTurn] = useState(0);
  const [complete, setComplete] = useState(false);
  return (
    <div className="flex min-h-[320px] w-full items-center justify-center p-2">
      <div className="flex w-full max-w-[420px] flex-col gap-4">
        <TaskList key={turn} onComplete={() => setComplete(true)} />
        {complete && (
          <button
            type="button"
            className="flex cursor-pointer items-center gap-1.5 self-start rounded-lg border border-border-button-default bg-background-primary-default px-2.5 py-1.5 text-body-medium text-text-secondary shadow-xs transition-colors duration-150 hover:bg-background-primary-hover hover:text-text-primary focus-visible:ring-2 focus-visible:ring-border-focus-ring focus-visible:outline-none"
            onClick={() => {
              setComplete(false);
              setTurn((value) => value + 1);
            }}
          >
            <RiRefreshLine className="size-4" aria-hidden />
            Run again
          </button>
        )}
      </div>
    </div>
  );
}

components/application/task-list/task-list.tsx
"use client";

import { RiArrowDownSLine, RiFileCodeLine, RiReactjsLine, RiSearchLine } from "@remixicon/react";
import { AnimatePresence, motion } from "motion/react";
import { type ComponentType, type ReactNode, useId, useState } from "react";
import { cx } from "../../utils/cx";
import {
  LogRow,
  ShimmerText,
  SOFT_EASE,
  UNIT_ANIMATE,
  UNIT_INITIAL,
  UNIT_TRANSITION,
  useLogMotion,
  useRevealMask,
  useRevealTicker,
  WorkingRow,
} from "../agent-log/agent-log";

export interface TaskResourceChip {
  label: string;
  icon?: ReactNode;
  href?: string;
}
export interface AgentTaskStep {
  label: string;
  chips?: TaskResourceChip[];
}
export interface AgentTask {
  id?: string;
  label: string;
  completedLabel?: string;
  icon?: ComponentType<{ className?: string; "aria-hidden"?: boolean | "true" | "false" }>;
  steps?: AgentTaskStep[];
}
export interface TaskListProps {
  tasks?: AgentTask[];
  revealed?: number;
  run?: boolean;
  startDelay?: number;
  stepInterval?: number;
  collapseOnComplete?: boolean | "all";
  onComplete?: () => void;
  className?: string;
}
const reactIcon = <RiReactjsLine className="size-3.5 text-[#61dafb]" />;
export const DEFAULT_AGENT_TASKS: AgentTask[] = [
  {
    id: "search",
    label: "Searching project files",
    completedLabel: "Found project files",
    icon: RiSearchLine,
    steps: [
      { label: 'Searching "app/page.tsx, components structure"' },
      { label: "Read", chips: [{ label: "page.tsx", icon: reactIcon }] },
      { label: "Scanning 52 files" },
      { label: "Reading files", chips: [{ label: "layout.tsx", icon: reactIcon }] },
    ],
  },
  {
    id: "edit",
    label: "Editing files",
    completedLabel: "Registered the theme toggle",
    icon: RiFileCodeLine,
    steps: [
      { label: "Created", chips: [{ label: "theme-toggle.tsx", icon: reactIcon }] },
      { label: "Wired it into", chips: [{ label: "layout.tsx", icon: reactIcon }] },
      { label: "Ran lint and type-check, 0 errors" },
    ],
  },
];

export function TaskChip({ label, icon, href }: TaskResourceChip) {
  const classes =
    "mr-0.5 ml-[3px] inline-flex max-w-full -translate-y-px items-center gap-1 rounded-md border border-border-button-default/50 bg-background-secondary-default py-0.5 pr-1.5 pl-1 align-middle transition-colors duration-150 hover:border-border-button-hover hover:bg-background-tertiary-default";
  const content = (
    <>
      {icon}
      <span className="truncate text-caption-2-medium text-text-primary">{label}</span>
    </>
  );
  return href ? (
    <a
      href={href}
      className={cx(
        classes,
        "outline-none hover:bg-background-secondary-hover focus-visible:ring-2 focus-visible:ring-border-focus-ring",
      )}
    >
      {content}
    </a>
  ) : (
    <span className={classes}>{content}</span>
  );
}
function TaskGroup({
  task,
  shown,
  done,
  collapse,
  reduce,
}: {
  task: AgentTask;
  shown: number;
  done: boolean;
  collapse: boolean;
  reduce: boolean;
}) {
  const [manualOpen, setManualOpen] = useState<boolean | null>(null);
  const open = manualOpen ?? !collapse;
  const id = useId();
  const mask = useRevealMask(reduce);
  const steps = task.steps ?? [];
  const Icon = task.icon ?? RiSearchLine;
  const label = done ? (task.completedLabel ?? task.label) : task.label;
  return (
    <motion.section
      initial={reduce ? false : UNIT_INITIAL}
      animate={UNIT_ANIMATE}
      transition={UNIT_TRANSITION}
      {...mask}
      className="overflow-hidden"
    >
      <button
        type="button"
        aria-label={label}
        aria-expanded={open}
        aria-controls={id}
        onClick={() => setManualOpen(!open)}
        className="group flex w-full cursor-pointer items-center gap-2 rounded-md py-0.5 text-left outline-none focus-visible:ring-2 focus-visible:ring-border-focus-ring"
      >
        <Icon aria-hidden className="size-4 shrink-0 text-foreground-icon-secondary" />
        <span className="min-w-0 flex-1 text-body-medium text-text-secondary">
          <AnimatePresence mode="popLayout" initial={false}>
            <motion.span
              key={label}
              initial={reduce ? false : { opacity: 0, y: 3, filter: "blur(3px)" }}
              animate={{ opacity: 1, y: 0, filter: "blur(0px)" }}
              exit={{ opacity: 0, y: -3, filter: "blur(3px)" }}
              transition={{ duration: reduce ? 0 : 0.3, ease: SOFT_EASE }}
              className="block truncate"
            >
              {done ? label : <ShimmerText>{label}</ShimmerText>}
            </motion.span>
          </AnimatePresence>
        </span>
        <RiArrowDownSLine
          aria-hidden
          className={cx(
            "size-4 shrink-0 text-foreground-icon-tertiary transition-transform duration-300 group-hover:text-foreground-icon-secondary motion-reduce:transition-none",
            open && "rotate-180",
          )}
        />
      </button>
      <motion.div
        id={id}
        aria-hidden={!open}
        inert={!open}
        initial={false}
        animate={{ height: open ? "auto" : 0, opacity: open ? 1 : 0 }}
        transition={{
          height: { duration: reduce ? 0 : 0.3, ease: SOFT_EASE },
          opacity: { duration: reduce ? 0 : 0.22, ease: "easeOut" },
        }}
        className="overflow-hidden"
      >
        <ul className="mt-0.5 ml-2 flex flex-col">
          {steps.slice(0, shown).map((step, index) => (
            <LogRow
              key={`${index}-${step.label}`}
              first={index === 0}
              last={index === Math.min(steps.length, shown) - 1}
              reduce={reduce}
            >
              <span className="block py-1 text-body-regular text-text-secondary">
                {step.label}
                {step.chips?.map((chip, index) => (
                  <TaskChip key={`${index}-${chip.label}`} {...chip} />
                ))}
              </span>
            </LogRow>
          ))}
        </ul>
      </motion.div>
    </motion.section>
  );
}

/** `revealed` counts one unit for each task header and each child step. */
export function TaskList({
  tasks = DEFAULT_AGENT_TASKS,
  revealed: controlled,
  run = true,
  startDelay = 320,
  stepInterval = 850,
  collapseOnComplete = false,
  onComplete,
  className,
}: TaskListProps = {}) {
  const reduce = useLogMotion();
  const total = tasks.reduce((sum, task) => sum + 1 + (task.steps?.length ?? 0), 0);
  const timedTotal = total + (controlled === undefined && total > 0 ? 1 : 0);
  const revealed = useRevealTicker({
    total: timedTotal,
    run,
    startDelay,
    stepInterval,
    revealed: controlled,
    onComplete,
  });
  const complete = revealed >= timedTotal;
  let offset = 0;
  return (
    <div className={cx("flex w-full flex-col gap-[5px]", className)} aria-label="Agent task log">
      {tasks.map((task, index) => {
        const start = offset;
        const units = 1 + (task.steps?.length ?? 0);
        offset += units;
        if (revealed <= start) return null;
        const done =
          controlled === undefined ? revealed > start + units : revealed >= start + units;
        return (
          <TaskGroup
            key={task.id ?? `${index}-${task.label}`}
            task={task}
            shown={Math.max(0, revealed - start - 1)}
            done={done}
            reduce={reduce}
            collapse={collapseOnComplete === "all" ? complete : collapseOnComplete && done}
          />
        );
      })}
      {!complete && <WorkingRow label="Working" reduce={reduce} />}
    </div>
  );
}
