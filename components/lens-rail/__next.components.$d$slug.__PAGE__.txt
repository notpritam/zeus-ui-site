export function useGalleryEnvironment() {
  const ref = useRef<HTMLDivElement>(null);
  const [size, setSize] = useState({ width: 1000, height: 640 });
  const [active, setActive] = useState(false);
  const [reduced, setReduced] = useState(false);
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    let visible = true;
    const media = window.matchMedia("(prefers-reduced-motion: reduce)");
    const update = () => {
      setReduced(media.matches);
      setActive(visible && !document.hidden);
    };
    const resize = () => setSize({ width: el.clientWidth || 1000, height: el.clientHeight || 640 });
    resize();
    const ro = typeof ResizeObserver === "undefined" ? null : new ResizeObserver(resize);
    ro?.observe(el);
    const io =
      typeof IntersectionObserver === "undefined"
        ? null
        : new IntersectionObserver((entries) => {
            visible = entries[0]?.isIntersecting ?? true;
            update();
          });
    io?.observe(el);
    media.addEventListener("change", update);
    document.addEventListener("visibilitychange", update);
    update();
    return () => {
      ro?.disconnect();
      io?.disconnect();
      media.removeEventListener("change", update);
      document.removeEventListener("visibilitychange", update);
    };
  }, []);
  return { ref, size, active, reduced };
}
export function useGalleryIndex({
  items,
  index,
  defaultIndex = 0,
  onChange,
}: GalleryInternalProps) {
  const [value, setValue] = useState(defaultIndex);
  const current = galleryWrap(index ?? value, items.length);
  const choose = useCallback(
    (next: number) => {
      next = galleryWrap(next, items.length);
      if (index === undefined) setValue(next);
      if (next !== current) onChange?.(next);
    },
    [current, index, items.length, onChange],
  );
  return { current, choose };
}
export function useGalleryMotion(
  count: number,
  options: {
    index?: number;
    defaultIndex?: number;
    onChange?: (index: number) => void;
    drag?: boolean;
    autoplay?: boolean;
    speed?: number;
    sensitivity?: number;
    snap?: boolean;
    vertical?: boolean;
    pauseOnHover?: boolean;
  },
  environment: ReturnType<typeof useGalleryEnvironment>,
) {
  const {
    index,
    defaultIndex = 0,
    onChange,
    drag = true,
    autoplay = false,
    speed = 0.15,
    sensitivity = 220,
    snap = false,
    vertical = false,
    pauseOnHover = false,
  } = options;
  const [phase, setPhase] = useState(defaultIndex);
  const value = useRef(defaultIndex);
  const velocity = useRef(0);
  const target = useRef<number | null>(null);
  const pointer = useRef<{ id: number; position: number; time: number; distance: number } | null>(
    null,
  );
  const moved = useRef(false);
  const [hovered, setHovered] = useState(false);
  const [focused, setFocused] = useState(false);
  const [paused, setPaused] = useState(false);
  const [dragging, setDragging] = useState(false);
  const lastIndex = useRef(galleryWrap(Math.round(defaultIndex), count));
  const changeRef = useRef(onChange);
  changeRef.current = onChange;
  const commit = useCallback(
    (next: number) => {
      if (!Number.isFinite(next)) return;
      value.current = next;
      setPhase(next);
      const selected = galleryWrap(Math.round(next), count);
      if (lastIndex.current !== selected) {
        lastIndex.current = selected;
        changeRef.current?.(selected);
      }
    },
    [count],
  );
  const choose = useCallback(
    (next: number) => {
      if (index !== undefined) {
        changeRef.current?.(galleryWrap(Math.round(next), count));
        return;
      }
      velocity.current = 0;
      if (environment.reduced) {
        target.current = null;
        commit(next);
      } else target.current = next;
    },
    [commit, environment.reduced, index, count],
  );
  useEffect(() => {
    if (index !== undefined) {
      target.current = null;
      velocity.current = 0;
      value.current = index;
      setPhase(index);
      lastIndex.current = galleryWrap(Math.round(index), count);
    }
  }, [index, count]);
  useEffect(() => {
    if (!environment.active || environment.reduced || count < 1) return;
    let frame = 0,
      last = 0;
    const tick = (time: number) => {
      const dt = Math.min((time - (last || time)) / 1000, 0.04);
      last = time;
      if (!pointer.current) {
        if (target.current !== null) {
          const next = value.current + (target.current - value.current) * (1 - Math.exp(-dt * 12));
          if (Math.abs(next - target.current) < 0.0005) {
            commit(target.current);
            target.current = null;
          } else commit(next);
        } else {
          velocity.current *= Math.exp(-dt * 5);
          const drift = autoplay && !paused && !focused && !(pauseOnHover && hovered) ? speed : 0;
          if (Math.abs(velocity.current) > 0.0001 || drift)
            commit(value.current + (velocity.current + drift) * dt);
        }
      }
      frame = requestAnimationFrame(tick);
    };
    frame = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(frame);
  }, [
    environment.active,
    environment.reduced,
    count,
    autoplay,
    speed,
    paused,
    hovered,
    focused,
    pauseOnHover,
    commit,
  ]);
  const release = (event: PointerEvent<HTMLDivElement>) => {
    const p = pointer.current;
    if (!p || p.id !== event.pointerId) return;
    pointer.current = null;
    setDragging(false);
    if (event.currentTarget.hasPointerCapture?.(event.pointerId))
      event.currentTarget.releasePointerCapture(event.pointerId);
    if (environment.reduced) velocity.current = 0;
    if (snap) {
      const next = Math.round(value.current + velocity.current * 0.1);
      velocity.current = 0;
      choose(next);
    }
  };
  return {
    phase: index ?? phase,
    current: galleryWrap(Math.round(index ?? phase), count),
    choose,
    dragging,
    paused,
    setPaused: (next: boolean) => {
      setPaused(next);
      if (next) {
        velocity.current = 0;
        target.current = null;
      }
    },
    moved,
    handlers: {
      onPointerEnter: () => setHovered(true),
      onPointerLeave: () => setHovered(false),
      onFocus: (event: React.FocusEvent<HTMLDivElement>) =>
        setFocused(!galleryControlTarget(event.target as Element)),
      onBlur: (event: React.FocusEvent<HTMLDivElement>) => {
        if (!event.currentTarget.contains(event.relatedTarget)) setFocused(false);
      },
      onPointerDown: (event: PointerEvent<HTMLDivElement>) => {
        if (!drag || event.button !== 0 || galleryControlTarget(event.target as Element)) return;
        target.current = null;
        velocity.current = 0;
        moved.current = false;
        pointer.current = {
          id: event.pointerId,
          position: vertical ? event.clientY : event.clientX,
          time: performance.now(),
          distance: 0,
        };
        setDragging(true);
      },
      onPointerMove: (event: PointerEvent<HTMLDivElement>) => {
        const p = pointer.current;
        if (!p || p.id !== event.pointerId) return;
        const next = vertical ? event.clientY : event.clientX;
        const now = performance.now();
        const delta = -(next - p.position) / Math.max(1, sensitivity);
        p.distance += Math.abs(next - p.position);
        moved.current = p.distance > 5;
        if (moved.current && !event.currentTarget.hasPointerCapture(event.pointerId))
          event.currentTarget.setPointerCapture(event.pointerId);
        velocity.current = delta / Math.max(0.008, (now - p.time) / 1000);
        commit(value.current + delta);
        p.position = next;
        p.time = now;
      },
      onPointerUp: release,
      onPointerCancel: (event: PointerEvent<HTMLDivElement>) => {
        velocity.current = 0;
        release(event);
      },
      onKeyDown: (event: KeyboardEvent<HTMLDivElement>) => {
        if (event.key === "ArrowRight" || event.key === "ArrowDown") {
          event.preventDefault();
          choose(Math.round(value.current) + 1);
        } else if (event.key === "ArrowLeft" || event.key === "ArrowUp") {
          event.preventDefault();
          choose(Math.round(value.current) - 1);
        } else if (event.key === "Home") {
          event.preventDefault();
          choose(0);
        } else if (event.key === "End") {
          event.preventDefault();
          choose(count - 1);
        } else if (event.key === " " && event.target === event.currentTarget) {
          event.preventDefault();
          setPaused((v) => !v);
          setFocused(false);
        }
      },
    },
  };
}

export interface LensRailProps extends GalleryInternalProps {

/** Gaussian lens magnification changes both card scale and spacing around the focal point. */
export function LensRail({
  items,
  bellWidth = 1.8,
  maxScale = 1,
  minScale = 0.38,
  gap = 32,
  spread = 1,
  drag = true,
  index,
  defaultIndex = 0,
  onChange,
  onSelect,
  height = 640,
  label = "Lens rail",
  className = "",
  style,
  renderCaption,
}: LensRailProps) {
  const env = useGalleryEnvironment();
  const width = Math.min(290, env.size.width * 0.53),
    cardHeight = width * 1.33;
  const motion = useGalleryMotion(
    items.length,
    { index, defaultIndex, onChange, drag, snap: true, sensitivity: width + gap },
    env,
  );
  return (
    <div
      ref={env.ref}
      {...motion.handlers}
      className={`z-ag ${className}`}
      data-gallery="lens-rail"
      data-index={motion.current}
      data-phase={motion.phase.toFixed(4)}
      tabIndex={0}
      role="region"
      aria-label={label}
      style={galleryStyle(height, {
        userSelect: drag ? "none" : undefined,
        touchAction: drag ? "pan-y" : undefined,
        cursor: drag ? "grab" : undefined,
        ...style,
      })}
    >
      <GalleryInternalStyles />
      <div
        style={{
          position: "absolute",
          top: 0,
          bottom: 0,
          left: "50%",
          width: 1,
          background: "#deddd9",
        }}
      />
      {items.map((item, i) => {
        const d = galleryDistance(i, motion.phase, items.length);
        const scale =
          minScale + (maxScale - minScale) * Math.exp((-d * d) / (2 * bellWidth * bellWidth));
        const x =
          Math.sign(d) *
          (Math.abs(d) * (width * minScale + gap) +
            width *
              (maxScale - minScale) *
              bellWidth *
              Math.sqrt(Math.PI / 2) *
              Math.tanh(Math.abs(d) / bellWidth)) *
          spread;
        return (
          <button
            key={item.id}
            type="button"
            className="z-ag-card"
            aria-label={`View ${item.title ?? `image ${i + 1}`}`}
            aria-pressed={motion.current === i}
            data-gallery-card={i}
            onClick={(event) => {
              if (event.detail === 0 || !motion.moved.current) {
                motion.choose(motion.phase + d);
                onSelect?.(item, i);
              }
            }}
            style={{
              width,
              height: cardHeight,
              borderRadius: 8,
              transform: `translate(-50%,-50%) translateX(${x}px) scale(${scale})`,
              zIndex: 100 - Math.round(Math.abs(d) * 10),
            }}
          >
            <GalleryInternalImage item={item} />
            {renderCaption && <span className="z-ag-caption">{renderCaption(item, i)}</span>}
          </button>
        );
      })}
      <GalleryInternalControls
        current={motion.current}
        count={items.length}
        onPrevious={() => motion.choose(Math.round(motion.phase) - 1)}
        onNext={() => motion.choose(Math.round(motion.phase) + 1)}
      />
    </div>
  );
}
