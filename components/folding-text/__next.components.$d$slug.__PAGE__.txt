/** Grapheme splitting keeps accents and composed emoji in one animated unit. */
export function typeGraphemes(text: string): string[] {
  return typeof Intl.Segmenter === "function"
    ? Array.from(
        new Intl.Segmenter(undefined, { granularity: "grapheme" }).segment(text),
        (item) => item.segment,
      )
    : Array.from(text);
}
export const typeClamp = (n: number) => Math.max(0, Math.min(1, n));
export function typeStaggerProgress(progress: number, index: number, count: number, spread = 0.55) {
  const stagger = Math.max(0, Math.min(0.95, spread));
  return typeClamp((progress - (index / Math.max(1, count - 1)) * stagger) / (1 - stagger));
}
export const typeEaseOut = (p: number) => 1 - Math.pow(1 - typeClamp(p), 3);
export interface TypeGlyphIdentity {
  id: number;
  char: string;
}
/** Retain the nearest unused copy of each glyph, including repeated characters. */
export function typeMatchGlyphs(previous: TypeGlyphIdentity[], text: string): TypeGlyphIdentity[] {
  const used = new Set<number>();
  let nextId = Math.max(-1, ...previous.map((glyph) => glyph.id)) + 1;
  return typeGraphemes(text).map((char, index) => {
    const matches = previous
      .map((glyph, oldIndex) => ({ glyph, oldIndex }))
      .filter(({ glyph }) => glyph.char === char && !used.has(glyph.id))
      .sort((a, b) => Math.abs(a.oldIndex - index) - Math.abs(b.oldIndex - index));
    const match = matches[0]?.glyph;
    if (match) {
      used.add(match.id);
      return match;
    }
    return { char, id: nextId++ };
  });
}

  split = "chars",

}) {

              whiteSpace: "pre",
              transformOrigin: "50% 100%",

export function typeScrollParent(element: HTMLElement): HTMLElement | null {
  let parent = element.parentElement;
  while (parent) {
    if (
      /(auto|scroll)/.test(getComputedStyle(parent).overflowY) &&
      parent.scrollHeight > parent.clientHeight
    )
      return parent;
    parent = parent.parentElement;
  }
  return null;
}
/** Instance-owned scheduling: hidden/offscreen work stops, font promises cannot revive an unmount. */
export function useTypeLoop<T extends HTMLElement>(
  root: RefObject<T | null>,
  draw: (time: number, delta: number, reduced: boolean) => void | boolean,
  key: string | number = "",
  enabled = true,
) {
  const latest = useRef(draw);
  latest.current = draw;
  useEffect(() => {
    const element = root.current;
    if (!element || !enabled) return;
    let disposed = false,
      frame = 0,
      visible = true,
      ready = false,
      elapsed = 0,
      last = 0;
    const media = matchMedia("(prefers-reduced-motion: reduce)");
    const ownsRoot = () => !disposed && root.current === element;
    const tick = (now: number) => {
      frame = 0;
      if (!ownsRoot() || !visible || document.hidden || !ready || media.matches) return;
      const delta = last ? Math.min(64, now - last) : 0;
      last = now;
      elapsed += delta;
      if (latest.current(elapsed, delta, false) !== false && ownsRoot())
        frame = requestAnimationFrame(tick);
    };
    const sync = () => {
      cancelAnimationFrame(frame);
      frame = 0;
      last = 0;
      if (!ownsRoot() || !ready) return;
      if (media.matches) latest.current(0, 0, true);
      else if (visible && !document.hidden) frame = requestAnimationFrame(tick);
    };
    const observer =
      typeof IntersectionObserver !== "undefined"
        ? new IntersectionObserver((entries) => {
            visible = entries[0]?.isIntersecting ?? true;
            sync();
          })
        : null;
    observer?.observe(element);
    media.addEventListener("change", sync);
    document.addEventListener("visibilitychange", sync);
    (document.fonts?.ready ?? Promise.resolve()).then(() => {
      if (ownsRoot()) {
        ready = true;
        sync();
      }
    });
    return () => {
      disposed = true;
      cancelAnimationFrame(frame);
      observer?.disconnect();
      media.removeEventListener("change", sync);
      document.removeEventListener("visibilitychange", sync);
    };
  }, [root, key, enabled]);
}

      }

    };

      typeof IntersectionObserver !== "undefined"
        ? new IntersectionObserver((entries) => {
            active = entries[0]?.isIntersecting ?? true;
            sync();

    document.addEventListener("visibilitychange", sync);

      element.addEventListener("pointerenter", start);
      element.addEventListener("focus", start);
    }
    if (trigger === "click") {
      element.addEventListener("click", start);
      element.addEventListener("keydown", keydown);
    }
    if (trigger === "scroll") {
      window.addEventListener("scroll", scroll, { passive: true });
      scrollParent?.addEventListener("scroll", scroll, { passive: true });
      window.addEventListener("resize", scroll);
    }
    (document.fonts?.ready ?? Promise.resolve()).then(() => {
      if (!ownsRoot()) return;
      ready = true;
      if (controlled.current !== undefined) render(controlled.current);
      else if (trigger === "load") start();
      else if (trigger === "scroll") scroll();
      else render(1);
    });
    return () => {
      disposed = true;
      currentRender.current = null;
      cancelAnimationFrame(frame);
      observer?.disconnect();
      media.removeEventListener("change", sync);
      document.removeEventListener("visibilitychange", sync);
      element.removeEventListener("pointerenter", start);
      element.removeEventListener("focus", start);
      element.removeEventListener("click", start);
      element.removeEventListener("keydown", keydown);
      window.removeEventListener("scroll", scroll);
      scrollParent?.removeEventListener("scroll", scroll);
      window.removeEventListener("resize", scroll);
    };
  }, [root, trigger, duration, delay, once, replayKey, scrollDistance, contentKey]);
  useEffect(() => {
    if (progress !== undefined) currentRender.current?.(progress);
  }, [progress]);
}
