function contains(lon: number, lat: number, polygon: number[][]) {
  let inside = false;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    const a = polygon[i],
      b = polygon[j];
    if (
      a[1] > lat !== b[1] > lat &&
      lon < ((b[0] - a[0]) * (lat - a[1])) / (b[1] - a[1]) + a[0]
    )
      inside = !inside;
  }
  return inside;
}

export function sphereRoute(
  from: SectionPoint,
  to: SectionPoint,
  yaw: number,
  pitch: number,
  radius: number,
  cx: number,
  cy: number,
) {
  const dot = Math.max(
      -0.9999,
      Math.min(
        0.9999,
        from.reduce((sum, n, i) => sum + n * to[i], 0),
      ),
    ),
    angle = Math.acos(dot);
  let path = "",
    pen = false;
  for (let i = 0; i <= 40; i++) {
    const t = i / 40,
      a = Math.sin((1 - t) * angle) / Math.sin(angle),
      b = Math.sin(t * angle) / Math.sin(angle),
      lift = 1 + 0.19 * Math.sin(t * Math.PI);
    const p = rotatePoint(
      [
        (from[0] * a + to[0] * b) * lift,
        (from[1] * a + to[1] * b) * lift,
        (from[2] * a + to[2] * b) * lift,
      ],
      yaw,
      pitch,
    );
    if (p[2] < 0) {
      pen = false;
      continue;
    }
    path += `${pen ? "L" : "M"}${(cx + p[0] * radius).toFixed(2)},${(cy + p[1] * radius).toFixed(2)} `;
    pen = true;
  }
  return path;
}
