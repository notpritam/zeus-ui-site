export const scrollClamp = (n: number) => (Number.isFinite(n) ? Math.max(0, Math.min(1, n)) : 0);
export const scrollLerp = (a: number, b: number, p: number) => a + (b - a) * scrollClamp(p);
export const scrollRange = (value: number, start: number, end: number) =>
  start === end ? Number(value >= end) : scrollClamp((value - start) / (end - start));
export const scrollEase = (p: number) => {
  const q = scrollClamp(p);
  return q * q * (3 - 2 * q);
};
export const scrollStagger = (p: number, index: number, count: number, spread = 0.4) => {
  const s = Math.min(0.95, Math.max(0, spread));
  return scrollClamp((p - (index / Math.max(1, count - 1)) * s) / (1 - s));
};
export const scrollRandom = (index: number, seed = 1) => {
  const x = Math.sin((index + 1) * 127.1 + seed * 311.7) * 43758.5453;
  return x - Math.floor(x);
};
export interface ScrollFrame {
  x: number;
  y: number;
  width: number;
  height: number;
  rotation?: number;
  radius?: number;
}
export function scrollFrame(
  frames: readonly ScrollFrame[],
  progress: number,
): Required<ScrollFrame> {
  const p = scrollClamp(progress) * Math.max(0, frames.length - 1),
    index = Math.floor(p);
  const a = frames[index] ?? { x: 0, y: 0, width: 100, height: 100 },
    b = frames[Math.min(index + 1, frames.length - 1)] ?? a;
  return {
    x: scrollLerp(a.x, b.x, p - index),
    y: scrollLerp(a.y, b.y, p - index),
    width: scrollLerp(a.width, b.width, p - index),
    height: scrollLerp(a.height, b.height, p - index),
    rotation: scrollLerp(a.rotation ?? 0, b.rotation ?? 0, p - index),
    radius: scrollLerp(a.radius ?? 0, b.radius ?? 0, p - index),
  };
}

export function scrollMixColor(a: string, b: string, p: number) {
  const parse = (s: string): number[] | null => {
    const hex = s.replace(/^#/, "");
    if (/^[0-9a-f]{3}$/i.test(hex)) return [...hex].map((x) => parseInt(x + x, 16));
    if (/^[0-9a-f]{6}$/i.test(hex)) return [0, 2, 4].map((i) => parseInt(hex.slice(i, i + 2), 16));
    const parts = s
      .match(/^rgba?\(([^)]+)\)$/)?.[1]
      ?.match(/[\d.]+/g)
      ?.map(Number);
    return parts && parts.length >= 3 ? parts : null;
  };
  const ca = parse(a),
    cb = parse(b);
  if (!ca || !cb) return p < 1 ? a : b;
  const channels = [0, 1, 2].map((i) => Math.round(scrollLerp(ca[i]!, cb[i]!, p))).join(", ");

export function scrollParent(element: HTMLElement): HTMLElement | null {
  let p = element.parentElement;
  while (p) {
    if (/(auto|scroll)/.test(getComputedStyle(p).overflowY) && p.scrollHeight > p.clientHeight)
      return p;
    p = p.parentElement;

  return null;

import { useRef, type ReactNode } from "react";
import { useScrollMotion, scrollStage, type ScrollMotionProps } from "./scroll-shared";
import { scrollClamp } from "./scroll-math";
export interface VelocityClipProps extends ScrollMotionProps {
  children: ReactNode;
  intensity?: number;
  maxInset?: number;
  maxVelocity?: number;
  scaleIntensity?: number;
  axis?: "x" | "y" | "both";
  borderRadius?: number;
}
export function VelocityClip({
  children,
  intensity = 1,
  maxInset = 24,
  maxVelocity = 2400,
  scaleIntensity = 0.1,
  axis = "x",
  borderRadius = 10,
  ...props
}: VelocityClipProps) {
  const root = useRef<HTMLDivElement>(null);
  useScrollMotion(
    root,
    props,
    ({ velocity, reduced }) => {
      const q = reduced
          ? 0
          : scrollClamp((Math.abs(velocity) / Math.max(1, maxVelocity)) * intensity),
        inset = q * Math.min(49, Math.max(0, maxInset)),
        media = root.current!.firstElementChild as HTMLElement;
      media.style.clipPath = `inset(${axis === "x" ? 0 : inset}% ${axis === "y" ? 0 : inset}% round ${borderRadius}px)`;
      media.style.transform = `scale(${1 + q * scaleIntensity})`;
      root.current!.dataset.clip = String(inset);
    },
    `${intensity}:${maxInset}:${maxVelocity}:${scaleIntensity}:${axis}:${borderRadius}`,
  );
  return (
    <div
      ref={root}
      data-scroll-animation="velocity-clip"
      className={props.className}
      style={{ ...scrollStage, ...props.style }}
    >
      <div style={{ width: "100%", height: "100%", overflow: "hidden", borderRadius }}>
        {children}
      </div>
    </div>
  );
}
