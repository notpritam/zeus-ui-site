Installation

Add animated pricing to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.2.3.tgz zeus-ui add animated-pricing

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — tokens. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

animated-pricing-demo.tsx
"use client";

import { AnimatedPricing } from "@/components/application/animations/animated-pricing";
import { useState } from "react";

export default function Demo() {
  const [selection, setSelection] = useState("");
  return (
    <div className="w-full">
      <AnimatedPricing
        onSelectPlan={(plan, period) => setSelection(`${plan.name} · ${period} selected`)}
      />
      {selection && (
        <p role="status" className="mt-3 text-center text-sm">
          {selection}
        </p>
      )}
    </div>
  );
}

components/application/animations/animated-pricing.tsx
"use client";

// Annnimate Starter Pack adaptation. See docs/ANNNIMATE.md for provenance.
import { forwardRef } from "react";
import Engine from "./pricing-engine.jsx";
import css from "./pricing-styles";

export interface AnimatedPricingPlan {
  name: string;
  nameClass?: "prc_name_silver" | "prc_name_champagne" | "prc_name_ice";
  monthly: number;
  yearly: number;
  popular?: boolean;
  features: string[];
  ctaLabel: string;
}
export interface AnimatedPricingHandle {
  getPeriod: () => "monthly" | "yearly";
  setPeriod: (period: "monthly" | "yearly") => void;
}
export interface AnimatedPricingProps {
  className?: string;
  title?: string;
  logoSrc?: string;
  logoAlt?: string;
  plans?: AnimatedPricingPlan[];
  rollDuration?: number;
  digitStagger?: number;
  toggleDuration?: number;
  shimmerDuration?: number;
  savingsLabel?: string;
  defaultPeriod?: "monthly" | "yearly";
  currency?: string;
  showSavingsChip?: boolean;
  ease?: string;
  disable?: string;
  onPeriodChange?: (period: "monthly" | "yearly") => void;
  onSelectPlan?: (plan: AnimatedPricingPlan, period: "monthly" | "yearly") => void;
}

export const AnimatedPricing = forwardRef<AnimatedPricingHandle, AnimatedPricingProps>(
  function AnimatedPricing(props, ref) {
    return (
      <>
        <style>{css}</style>
        <Engine {...props} ref={ref} />
      </>
    );
  },
);
