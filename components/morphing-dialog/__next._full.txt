export function useControlValue<T>(
  value: T | undefined,
  initial: T,
  onChange?: (value: T) => void,
): [T, (next: T) => void] {
  const [local, setLocal] = useState(initial);
  return [
    value === undefined ? local : value,
    (next) => {
      if (value === undefined) setLocal(next);
      onChange?.(next);
    },
  ];
}

export function ControlLiquid({
  root,
  revision,
  duration = 0.65,
  goo = 18,
}: {
  root: RefObject<HTMLElement | null>;
  revision: unknown;
  duration?: number;
  goo?: number;
}) {
  const id = `z-mc-goo-${useId().replace(/:/g, "")}`;
  const transition = useControlTransition(duration);
  const [shapes, setShapes] = useState<
    { id: string; x: number; y: number; w: number; h: number; r: number }[]
  >([]);
  useLayoutEffect(() => {
    const el = root.current;
    if (!el) return;
    const measure = () => {
      const next = [...el.querySelectorAll<HTMLElement>("[data-z-mc-liquid]")].map((node) => {
        let x = 0,
          y = 0;
        let parent: HTMLElement | null = node;
        while (parent && parent !== el) {
          x += parent.offsetLeft;
          y += parent.offsetTop;
          parent = parent.offsetParent as HTMLElement | null;
        }
        return {
          id: node.dataset.zMcLiquid!,
          x,
          y,
          w: node.offsetWidth,
          h: node.offsetHeight,
          r: parseFloat(getComputedStyle(node).borderRadius) || 24,
        };
      });
      setShapes((old) => (JSON.stringify(old) === JSON.stringify(next) ? old : next));
    };
    measure();
    const ro = typeof ResizeObserver !== "undefined" ? new ResizeObserver(measure) : null;
    ro?.observe(el);
    el.querySelectorAll("[data-z-mc-liquid]").forEach((e) => ro?.observe(e));
    let live = true;
    document.fonts?.ready.then(() => {
      if (live) measure();
    });
    return () => {
      live = false;
      ro?.disconnect();
    };
  }, [root, revision]);
  return (
    <>
      <ControlGoo id={id} goo={goo} />
      <div className="z-mc-liquid" aria-hidden="true" style={{ filter: `url(#${id})` }}>
        {shapes.map((s) => (

            initial={false}

          />
        ))}
      </div>
    </>
  );
}

/** Native top-layer semantics, Escape, modal focus containment, and focus return. Closing waits for the visual reverse. */
export function useControlDialog(
  open: boolean,
  onOpenChange: (open: boolean) => void,
  duration = 0.65,
  restoreFocus?: RefObject<HTMLElement | null>,
) {
  const ref = useRef<HTMLDialogElement>(null);
  const trigger = useRef<HTMLElement | null>(null);
  const owned = useRef(false);
  const unlock = useRef<(() => void) | null>(null);
  const reduce = useReducedMotion();
  const change = useRef(onOpenChange);
  change.current = onOpenChange;
  useLayoutEffect(() => {
    const node = ref.current;
    if (!node) return;
    let timer: ReturnType<typeof setTimeout> | undefined;
    if (open) {
      if (!owned.current) {
        trigger.current =
          restoreFocus?.current ??
          (document.activeElement instanceof HTMLElement ? document.activeElement : null);
        // Promote an SSR-readable, default-open surface into the modal top layer.
        if (node.open) {
          node.close?.();
          node.removeAttribute("open");
        }
        if (node.showModal) node.showModal();
        else node.setAttribute("open", "");
        owned.current = true;
        unlock.current = controlLockScroll();
      }
    } else if (node.open)
      timer = setTimeout(
        () => {
          node.close?.();
          node.removeAttribute("open");
          owned.current = false;
          unlock.current?.();
          unlock.current = null;
          trigger.current?.focus();
        },
        reduce ? 0 : Math.max(0, duration) * 1000,
      );
    const cancel = (event: Event) => {
      event.preventDefault();
      change.current(false);
    };
    node.addEventListener("cancel", cancel);
    return () => {
      clearTimeout(timer);
      node.removeEventListener("cancel", cancel);
    };
  }, [open, duration, reduce]);
  useEffect(
    () => () => {
      ref.current?.close?.();
      owned.current = false;
      unlock.current?.();
      unlock.current = null;
      trigger.current?.focus();
    },
    [],
  );
  return ref;
}

export function ControlTextLines({
  text,
  open,
  duration = 0.8,
  stagger = 0.045,
  delay = 0,
}: {
  text: string;
  open: boolean;
  duration?: number;
  stagger?: number;
  delay?: number;
}) {
  const paragraph = useRef<HTMLParagraphElement>(null);
  const mounted = useRef(false);
  useEffect(() => {
    mounted.current = true;
  }, []);
  const [lines, setLines] = useState<string[] | null>(null);
  const transition = useControlTransition(duration);
  useLayoutEffect(() => {
    const node = paragraph.current;
    if (!node) return;
    let live = true;
    const measure = () => {
      if (!node.clientWidth) return;
      const probe = document.createElement("span");
      probe.textContent = text;
      Object.assign(probe.style, {
        position: "absolute",
        display: "block",
        visibility: "hidden",
        pointerEvents: "none",
        width: `${node.clientWidth}px`,
        inset: "0 auto auto 0",
      });
      node.appendChild(probe);
      const range = document.createRange();
      const textNode = probe.firstChild;
      if (!textNode || typeof range.getBoundingClientRect !== "function") {
        probe.remove();
        return;
      }
      const measured: string[] = [];
      let previousTop = Number.NaN;
      for (const match of text.matchAll(/\S+\s*/g)) {
        range.setStart(textNode, match.index!);
        range.setEnd(textNode, match.index! + match[0].trimEnd().length);
        const top = range.getBoundingClientRect().top;
        if (!measured.length || Math.abs(top - previousTop) > 2) {
          measured.push(match[0]);
          previousTop = top;
        } else measured[measured.length - 1] += match[0];
      }
      probe.remove();
      if (live)
        setLines((old) => (JSON.stringify(old) === JSON.stringify(measured) ? old : measured));
    };
    measure();
    const observer = typeof ResizeObserver !== "undefined" ? new ResizeObserver(measure) : null;
    observer?.observe(node);
    document.fonts?.ready.then(() => live && measure());
    return () => {
      live = false;
      observer?.disconnect();
    };
  }, [text]);
  return (
    <p ref={paragraph} className="z-mc-text-lines">
      {lines
        ? lines.map((line, index) => (
            <span className="z-mc-text-line" key={`${index}-${line}`}>
              <motion.span
                initial={mounted.current && open ? { y: "110%", opacity: 0 } : false}
                animate={{ y: open ? "0%" : "110%", opacity: open ? 1 : 0 }}
                transition={{
                  ...transition,
                  delay: open && transition.duration ? delay + index * stagger : 0,
                }}
              >
                {line}
              </motion.span>
            </span>
          ))
        : text}
    </p>
  );
}
