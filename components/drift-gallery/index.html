Installation

Add drift gallery to your project. New to the library? Set up the design system first.

CLI
Manual
bash
npx --yes --allow-remote=all --package=https://myzeusui.com/install/zeus-ui-0.3.1.tgz zeus-ui add drift-gallery

Run in your React project. Source, dependencies and required foundations are copied together. For your first Zeus component, import the styles once.

components/application/animations/expanded/sections/drift-gallery.tsx
"use client";
import {
  useEffect,
  useId,
  useRef,
  useState,
  type CSSProperties,
  type ReactNode,
  type PointerEvent,
} from "react";
import {
  SectionImage,
  sectionStyles,
  useSectionEnvironment,
  useSectionLoop,
  useSectionSelection,
} from "./section-shared";
import { sectionClamp, sectionWrap } from "./section-math";
export interface DriftGalleryItem {
  id: string;
  title: string;
  src?: string;
  alt?: string;
  description?: string;
  category?: string;
  content?: ReactNode;
}
export interface DriftGalleryProps {
  items: readonly DriftGalleryItem[];
  title?: ReactNode;
  description?: ReactNode;
  watermark?: string;
  action?: { label: string; href: string };
  index?: number;
  defaultIndex?: number;
  onChange?: (index: number) => void;
  onSelect?: (item: DriftGalleryItem) => void;
  drag?: boolean;
  autoplay?: boolean;
  speed?: number;
  cardWidth?: number;
  gap?: number;
  separation?: number;
  friction?: number;
  height?: number;
  className?: string;
  style?: CSSProperties;
  label?: string;
}
/** Lateral inertial product cards with grab separation and velocity-driven grain/color. */
export function DriftGallery({
  items,
  title = "Made for the move.",
  description,
  watermark = "FIELD 26",
  action,
  index,
  defaultIndex = 0,
  onChange,
  onSelect,
  drag = true,
  autoplay = false,
  speed = 0.12,
  cardWidth = 250,
  gap = 35,
  separation = 45,
  friction = 4.5,
  height = 720,
  className = "",
  style,
  label = "Drift gallery",
}: DriftGalleryProps) {
  const env = useSectionEnvironment(),
    selection = useSectionSelection(
      items.length,
      index,
      defaultIndex,
      onChange,
    ),
    [phase, setPhase] = useState(defaultIndex),
    [velocity, setVelocity] = useState(0),
    [grabbed, setGrabbed] = useState(false),
    motion = useRef({
      phase: defaultIndex,
      velocity: 0,
      target: null as number | null,
    }),
    pointer = useRef<null | {
      id: number;
      x: number;
      time: number;
      distance: number;
    }>(null),
    noise = useId().replaceAll(":", "");
  const width = Math.min(cardWidth, env.size.width * 0.55),
    step = width + Math.max(0, gap),
    spread = grabbed ? Math.max(0, separation) : 0;
  const publish = () => {
    setPhase(motion.current.phase);
    setVelocity(motion.current.velocity);
    selection.select(Math.round(motion.current.phase));
  };
  useEffect(() => {
    if (index !== undefined) {
      motion.current.phase = index;
      motion.current.velocity = 0;
      setPhase(index);
    }
  }, [index]);
  useSectionLoop(env.active && !env.reduced, (delta) => {
    const m = motion.current;
    if (pointer.current) return;
    if (m.target !== null) {
      m.phase += (m.target - m.phase) * (1 - Math.exp(-delta * 9));
      if (Math.abs(m.target - m.phase) < 0.0005) {
        m.phase = m.target;
        m.target = null;
      }
    } else {
      if (Math.abs(m.velocity) < 0.002 && !autoplay) return;
      m.phase += (m.velocity + (autoplay ? speed : 0)) * delta;
      m.velocity *= Math.exp(-Math.max(0.5, friction) * delta);
    }
    publish();
  });
  const go = (next: number) => {
    const m = motion.current;
    m.velocity = 0;
    if (env.reduced) {
      m.phase = next;
      publish();
    } else m.target = next;
    selection.select(next);
  };
  const end = (event: PointerEvent<HTMLDivElement>) => {
    if (pointer.current?.id !== event.pointerId) return;
    pointer.current = null;
    setGrabbed(false);
    if (env.reduced) {
      motion.current.velocity = 0;
      setVelocity(0);
    }
    event.currentTarget.releasePointerCapture?.(event.pointerId);
  };
  const chosen = items[selection.selected];
  return (
    <div
      ref={env.ref}
      role="region"
      aria-label={label}
      tabIndex={0}
      className={`z-as z-as-drift ${className}`}
      data-section="drift-gallery"
      data-index={selection.selected}
      data-phase={phase.toFixed(5)}
      data-velocity={velocity.toFixed(4)}
      data-grabbed={grabbed}
      data-separation={spread}
      style={{ height: `min(${height}px,100svh)`, minHeight: 540, ...style }}
      onKeyDown={(e) => {
        if (e.key === "ArrowLeft" || e.key === "ArrowRight") {
          e.preventDefault();
          go(
            Math.round(motion.current.phase) +
              (e.key === "ArrowRight" ? 1 : -1),
          );
        }
      }}
      onPointerDown={(event) => {
        if (
          !drag ||
          event.button !== 0 ||
          (event.target as HTMLElement).closest("nav, a, .z-as-drift-intro")
        )
          return;
        pointer.current = {
          id: event.pointerId,
          x: event.clientX,
          time: event.timeStamp,
          distance: 0,
        };
        motion.current.velocity = 0;
        motion.current.target = null;
        setGrabbed(true);
        event.currentTarget.setPointerCapture?.(event.pointerId);
      }}
      onPointerMove={(event) => {
        const p = pointer.current;
        if (!p || p.id !== event.pointerId) return;
        const delta = (p.x - event.clientX) / step,
          dt = Math.max(0.008, (event.timeStamp - p.time) / 1000);
        motion.current.phase += delta;
        motion.current.velocity = sectionClamp(delta / dt, -8, 8);
        pointer.current = {
          id: p.id,
          x: event.clientX,
          time: event.timeStamp,
          distance: p.distance + Math.abs(delta),
        };
        publish();
      }}
      onPointerUp={end}
      onPointerCancel={(event) => {
        motion.current.velocity = 0;
        end(event);
      }}
      onLostPointerCapture={() => {
        pointer.current = null;
        setGrabbed(false);
      }}
    >
      <style>
        {sectionStyles +
          `.z-as-drift{overflow:hidden;touch-action:pan-y;user-select:none;cursor:grab}.z-as-drift[data-grabbed=true]{cursor:grabbing}.z-as-drift .z-as-drift-intro{position:absolute;top:30px;left:4.4%;z-index:5;max-width:470px;cursor:auto}.z-as-drift .z-as-drift-intro h2{font-size:clamp(32px,5.4vw,64px);line-height:1.05;letter-spacing:-.05em;font-weight:600}.z-as-drift .z-as-drift-intro p{font-size:13px;color:#777;max-width:290px;margin:20px 0}.z-as-drift .z-as-drift-intro a{display:inline-block;background:var(--z-as-accent);color:#171717;border-radius:40px;text-decoration:none;font-size:11px;letter-spacing:.06em;text-transform:uppercase;padding:12px 22px}.z-as-drift .z-as-drift-watermark{position:absolute;top:30%;left:-2%;font-size:clamp(100px,34vw,440px);letter-spacing:-.07em;font-weight:600;line-height:1;white-space:nowrap;color:#11111108;pointer-events:none}.z-as-drift .z-as-drift-card{position:absolute;left:50%;top:46%;height:36%;min-height:180px;max-height:330px;will-change:transform;border:0;padding:0;background:#d7d7d2;border-radius:8px;overflow:hidden;cursor:inherit;transition:filter .15s}.z-as-drift .z-as-drift-card>span{position:absolute;bottom:0;left:0;right:0;padding:24px 15px 12px;background:linear-gradient(transparent,#0008);color:white;text-align:left;font-size:14px}.z-as-drift .z-as-drift-detail{position:absolute;top:36px;right:4.4%;max-width:180px;text-align:right;z-index:3;font-size:12px}.z-as-drift .z-as-drift-detail strong{display:block;font-size:22px;font-weight:400;letter-spacing:-.03em}.z-as-drift .z-as-drift-detail small{color:#999}.z-as-drift .z-as-controls{position:absolute;bottom:24px;left:0;right:0;z-index:6}.z-as-drift .z-as-controls button{background:#1111}.z-as-drift .z-as-controls span{font-size:12px}@media(max-width:600px){.z-as-drift .z-as-drift-intro{top:25px;right:20px}.z-as-drift .z-as-drift-intro p{margin:12px 0}.z-as-drift .z-as-drift-detail{display:none}.z-as-drift .z-as-drift-card{top:47%;height:35%;min-height:160px}.z-as-drift .z-as-drift-watermark{top:40%;font-size:38vw}}`}
      </style>
      <div className="z-as-drift-intro">
        <h2>{title}</h2>
        <p>{description}</p>
        {action && <a href={action.href}>{action.label}</a>}
      </div>
      <div className="z-as-drift-watermark" aria-hidden="true">
        {watermark}
      </div>
      {chosen && (
        <div className="z-as-drift-detail">
          <small>{String(selection.selected + 1).padStart(2, "0")}</small>
          <strong>{chosen.title}</strong>
          <span>{chosen.category}</span>
        </div>
      )}
      {items.map((item, i) => {
        const distance =
            sectionWrap(i - phase + items.length / 2, items.length) -
            items.length / 2,
          x = distance * step + Math.sign(distance) * spread,
          y = Math.sin(distance * 0.9) * 12;
        return (
          <button
            key={item.id}
            type="button"
            aria-label={`View ${item.title}`}
            className="z-as-drift-card"
            data-drift-card={i}
            style={{
              width,
              transform: `translate3d(calc(-50% + ${x}px),${y}px,0) rotate(${sectionClamp(velocity * distance * 0.35, -8, 8)}deg)`,
              filter: `saturate(${1 + Math.min(1.2, Math.abs(velocity) * 0.14)})`,
              transition:
                grabbed && !env.reduced ? "transform .16s ease-out" : "none",
            }}
            onClick={() => {
              if (Math.abs(velocity) < 0.3) {
                go(i);
                onSelect?.(item);
              }
            }}
          >
            <SectionImage item={item} />
            <span>{item.title}</span>
          </button>
        );
      })}
      <svg
        className="z-as-fill"
        aria-hidden="true"
        style={{
          width: "100%",
          height: "100%",
          pointerEvents: "none",
          opacity: Math.min(0.22, Math.abs(velocity) * 0.045),
          mixBlendMode: "multiply",
          zIndex: 4,
        }}
      >
        <filter id={noise}>
          <feTurbulence
            type="fractalNoise"
            baseFrequency=".65"
            numOctaves="3"
            stitchTiles="stitch"
          />
        </filter>
        <rect width="100%" height="100%" filter={`url(#${noise})`} />
      </svg>
      <nav className="z-as-controls" aria-label="Gallery navigation">
        <button
          type="button"
          aria-label="Previous product"
          onClick={() => go(Math.round(motion.current.phase) - 1)}
        >
          ‹
        </button>
        <span aria-live="polite">{chosen?.title ?? "No products"}</span>
        <button
          type="button"
          aria-label="Next product"
          onClick={() => go(Math.round(motion.current.phase) + 1)}
        >
          ›
        </button>
      </nav>
    </div>
  );
}
