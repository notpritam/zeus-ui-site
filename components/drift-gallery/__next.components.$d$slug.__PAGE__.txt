/** Lateral inertial product cards with grab separation and velocity-driven grain/color. */
export function DriftGallery({
  items,
  title = "Made for the move.",
  description,
  watermark = "FIELD 26",
  action,
  index,
  defaultIndex = 0,
  onChange,
  onSelect,
  drag = true,
  autoplay = false,
  speed = 0.12,
  cardWidth = 250,
  gap = 35,
  separation = 45,
  friction = 4.5,
  height = 720,
  className = "",

  const env = useSectionEnvironment(),
    selection = useSectionSelection(
      items.length,
      index,
      defaultIndex,
      onChange,
    ),
    [phase, setPhase] = useState(defaultIndex),
    [velocity, setVelocity] = useState(0),
    [grabbed, setGrabbed] = useState(false),
    motion = useRef({
      phase: defaultIndex,
      velocity: 0,
      target: null as number | null,
    }),
    pointer = useRef<null | {
      id: number;
      x: number;
      time: number;
      distance: number;

    noise = useId().replaceAll(":", "");

    setPhase(motion.current.phase);
    setVelocity(motion.current.velocity);
    selection.select(Math.round(motion.current.phase));

      motion.current.velocity = 0;

      if (Math.abs(m.target - m.phase) < 0.0005) {
        m.phase = m.target;
        m.target = null;
      }
    } else {
      if (Math.abs(m.velocity) < 0.002 && !autoplay) return;
      m.phase += (m.velocity + (autoplay ? speed : 0)) * delta;
      m.velocity *= Math.exp(-Math.max(0.5, friction) * delta);
    }
    publish();
  });

    const m = motion.current;
    m.velocity = 0;
    if (env.reduced) {
      m.phase = next;
      publish();

    selection.select(next);

    if (pointer.current?.id !== event.pointerId) return;
    pointer.current = null;
    setGrabbed(false);

          );
        }
      }}

          x: event.clientX,
          time: event.timeStamp,
          distance: p.distance + Math.abs(delta),
        };

  Math.min(max, Math.max(min, Number.isFinite(value) ? value : 0));

export type SectionPoint = readonly [number, number, number];
export function spherePoint(latitude: number, longitude: number): SectionPoint {
  const lat = (latitude * Math.PI) / 180,
    lon = (longitude * Math.PI) / 180;
  return [
    Math.cos(lat) * Math.sin(lon),
    -Math.sin(lat),
    Math.cos(lat) * Math.cos(lon),
  ];
}
export function rotatePoint(
  [x, y, z]: SectionPoint,
  yaw: number,
  pitch: number,
): SectionPoint {
  const xx = x * Math.cos(yaw) + z * Math.sin(yaw),
    zz = z * Math.cos(yaw) - x * Math.sin(yaw);
  return [
    xx,
    y * Math.cos(pitch) - zz * Math.sin(pitch),
    zz * Math.cos(pitch) + y * Math.sin(pitch),
  ];
}
export function fibonacciPoint(index: number, count: number): SectionPoint {
  const y = 1 - (2 * (index + 0.5)) / Math.max(1, count),
    r = Math.sqrt(1 - y * y),
    angle = index * Math.PI * (3 - Math.sqrt(5));
  return [Math.cos(angle) * r, y, Math.sin(angle) * r];
}
