Card Fan

Hovering a stacked group spreads images into an arc; targeting one card raises it and widens neighboring spacing before the stack closes again.

Installation

Add card fan to your project. New to the library? Set up the design system first.

CLI
Manual
bash
npx --yes --allow-remote=all --package=https://myzeusui.com/install/zeus-ui-0.3.3.tgz zeus-ui add card-fan

Run in your React project. Source, dependencies and required foundations are copied together. For your first Zeus component, import the styles once.

components/application/animations/expanded/galleries/card-fan.tsx
"use client";
import { useState } from "react";
import { motion } from "motion/react";
import {
  GalleryInternalImage,
  GalleryInternalStyles,
  galleryStyle,
  useGalleryEnvironment,
  useGalleryIndex,
  type GalleryInternalProps,
  type GalleryInternalItem,
} from "./gallery-shared";
export interface CardFanItem extends GalleryInternalItem {}
export interface CardFanProps extends GalleryInternalProps {
  items: readonly CardFanItem[];
  spread?: number;
  angle?: number;
  lift?: number;
  duration?: number;
}
export function CardFan(props: CardFanProps) {
  const {
    items,
    spread = 170,
    angle = 10,
    lift = 50,
    duration = 0.7,
    height = 640,
    label = "Card fan",
    className = "",
    style,
    renderCaption,
    onSelect,
  } = props;
  const env = useGalleryEnvironment();
  const { current, choose } = useGalleryIndex(props);
  const [open, setOpen] = useState(false);
  const [hovered, setHovered] = useState<number | null>(null);
  const width = Math.min(310, env.size.width * 0.47);
  const span = Math.min(spread, (env.size.width - width * 0.9) / Math.max(1, items.length - 1));
  return (
    <div
      ref={env.ref}
      className={`z-ag ${className}`}
      data-gallery="card-fan"
      data-open={open}
      data-index={current}
      role="region"
      aria-label={label}
      onPointerEnter={() => setOpen(true)}
      onPointerLeave={() => {
        setOpen(false);
        setHovered(null);
      }}
      onFocus={() => setOpen(true)}
      onBlur={(e) => {
        if (!e.currentTarget.contains(e.relatedTarget)) {
          setOpen(false);
          setHovered(null);
        }
      }}
      style={galleryStyle(height, { background: "#fff", ...style })}
    >
      <GalleryInternalStyles />
      {items.map((item, i) => {
        const d = i - (items.length - 1) / 2;
        const targeted = i === hovered;
        const x = open
          ? d * span + (hovered === null ? 0 : Math.sign(i - hovered) * Math.min(span * 0.15, 20))
          : d * 5;
        const y = open ? Math.abs(d) * Math.abs(d) * 12 - (targeted ? lift : 0) : Math.abs(d) * 2;
        const rotation = open ? d * angle : d * 3;
        return (
          <motion.button
            type="button"
            key={item.id}
            className="z-ag-card"
            data-gallery-card={i}
            aria-pressed={current === i}
            aria-label={item.title ?? `Image ${i + 1}`}
            onPointerEnter={() => {
              setHovered(i);
              choose(i);
            }}
            onFocus={() => {
              setHovered(i);
              choose(i);
            }}
            onClick={() => {
              setOpen(true);
              setHovered(i);
              choose(i);
              onSelect?.(item, i);
            }}
            animate={{ x, y, rotate: targeted ? rotation * 0.3 : rotation }}
            transition={{ duration: env.reduced ? 0 : duration, ease: [0.22, 1, 0.36, 1] }}
            style={{
              width,
              height: width * 1.28,
              marginLeft: -width / 2,
              marginTop: -width * 0.64,
              borderRadius: 16,
              transformOrigin: "50% 100%",
              zIndex: targeted ? 50 : items.length - Math.round(Math.abs(d)),
              cursor: "pointer",
            }}
          >
            <GalleryInternalImage item={item} />
            {renderCaption && <span className="z-ag-caption">{renderCaption(item, i)}</span>}
          </motion.button>
        );
      })}
    </div>
  );
}
