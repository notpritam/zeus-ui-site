import { type CSSProperties, type ReactNode, useId, useRef } from "react";
import { MenuInternalStyles, useMenuDismiss, useMenuState } from "./menu-shared";
export interface GooeyMenuAction {
  id: string;
  label: string;
  icon?: ReactNode;
  href?: string;
  onAction?: () => void;
}
export interface GooeyMenuProps {
  actions: readonly GooeyMenuAction[];
  arrangement?: "arc" | "radial" | "row" | "grid";
  radius?: number;
  arcAngle?: number;
  gridColumns?: number;
  blur?: number;
  duration?: number;
  size?: number;
  color?: string;
  iconColor?: string;
  open?: boolean;
  defaultOpen?: boolean;
  onOpenChange?: (open: boolean) => void;
  onAction?: (action: GooeyMenuAction) => void;
  className?: string;
  style?: CSSProperties;
}
export function GooeyMenu({
  actions,
  arrangement = "arc",
  radius = 78,
  arcAngle = 160,
  gridColumns = 3,
  blur = 7,
  duration = 0.55,
  size = 60,
  color = "#ff572b",
  iconColor = "#fff",
  open,
  defaultOpen = false,
  onOpenChange,
  onAction,
  className,
  style,
}: GooeyMenuProps) {
  const [expanded, setExpanded] = useMenuState(open, defaultOpen, onOpenChange),
    id = useId().replaceAll(":", "");
  const root = useRef<HTMLDivElement>(null),
    trigger = useRef<HTMLButtonElement>(null);
  useMenuDismiss(expanded, setExpanded, root, trigger);
  const positions = actions.map((_, i) => {
    const n = Math.max(1, actions.length - 1),
      angle =
        ((arrangement === "radial"
          ? -90 + (i * 360) / actions.length
          : -90 - arcAngle / 2 + (i * arcAngle) / n) *
          Math.PI) /
        180;
    if (arrangement === "row")
      return { x: (i - (actions.length - 1) / 2) * (size + 10), y: -radius };
    if (arrangement === "grid") {
      const cols = Math.max(1, Math.round(gridColumns));
      return {
        x: ((i % cols) - (Math.min(cols, actions.length) - 1) / 2) * (size + 10),
        y: -(Math.floor(i / cols) + 1) * (size + 10),
      };
    }
    return { x: Math.cos(angle) * radius, y: Math.sin(angle) * radius };
  });
  const transform = (i: number) =>
    expanded
      ? `translate(${positions[i].x}px,${positions[i].y}px) scale(1)`
      : "translate(0px,0px) scale(.75)";
  const itemStyle = (i: number): CSSProperties => ({
    position: "absolute",
    left: (size - 44) / 2,
    top: (size - 44) / 2,

export function useMenuState(
  value: boolean | undefined,
  initial: boolean,
  onChange?: (open: boolean) => void,
) {
  const [local, setLocal] = useState(initial);
  return [

export function useMenuDismiss(
  open: boolean,
  onChange: (open: boolean) => void,
  host: RefObject<HTMLElement | null>,
  trigger?: RefObject<HTMLElement | null>,
) {
  const latest = useRef(onChange);
  latest.current = onChange;
  useEffect(() => {
    if (!open) return;
    const pointer = (event: PointerEvent) => {
      if (!host.current?.contains(event.target as Node)) latest.current(false);
    };
    const key = (event: KeyboardEvent) => {
      if (event.key === "Escape") {
        event.preventDefault();
        latest.current(false);
        trigger?.current?.focus();
      }
    };
    document.addEventListener("pointerdown", pointer);
    document.addEventListener("keydown", key);
    return () => {
      document.removeEventListener("pointerdown", pointer);
      document.removeEventListener("keydown", key);
    };
  }, [open, host, trigger]);
}
