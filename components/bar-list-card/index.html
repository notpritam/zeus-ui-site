Bar List

A ranked list where each row's bar is its share of the top — no axis, no library.

Installation

Add bar list to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.2.1.tgz zeus-ui add bar-list-card

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — chart-card, chevrons, chip, cx, segmented-control, select, tokens, use-count-up, use-dismiss-on-outside-press. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

bar-list-card-demo.tsx
"use client";

import { BarListCard } from "@/components/application/charts/bar-list-card";

export default function BarListCardDemo() {
  return (
    <div className="w-full max-w-md">
      <BarListCard />
    </div>
  );
}

components/application/charts/bar-list-card.tsx
"use client";
import { RiComputerLine, RiSmartphoneLine, RiTabletLine } from "@remixicon/react";
import { motion } from "motion/react";
import { type ReactNode, useState } from "react";
import { Tab, TabList, TabPanel, Tabs } from "react-aria-components";
import { useHydratedReducedMotion } from "../../hooks/use-hydrated-reduced-motion";
import { cx } from "../../utils/cx";
import { formatNumber } from "./chart-card";
import { BAR_LIST_DEMO_TABS } from "./chart-demo-data";
import { safeRatio, sum } from "./pro-chart";
export interface BarListItem {
  label: string;
  value: number;
  icon?: ReactNode;
  tone?: number;
  color?: string;
  href?: string;
}
export interface BarListTab {
  id: string;
  label: string;
  items: BarListItem[];
}
export interface BarListCardProps {
  data?: BarListItem[];
  items?: BarListItem[];
  tabs?: BarListTab[];
  title?: string;
  unit?: string;
  metricLabel?: string;
  metric?: "value" | "share";
  tone?: number;
  limit?: number;
  defaultTab?: string;
  onTabChange?: (id: string) => void;
  mono?: boolean;
  className?: string;
  height?: number | "auto";
}
export const BAR_LIST_DATA: BarListItem[] = [
  { label: "Desktop", value: 5980, icon: <RiComputerLine className="size-4" /> },
  { label: "Mobile", value: 3020, icon: <RiSmartphoneLine className="size-4" /> },
  { label: "Tablet", value: 820, icon: <RiTabletLine className="size-4" /> },
];
const TABS: BarListTab[] = BAR_LIST_DEMO_TABS.map((tab) => ({
  ...tab,
  items: tab.id === "devices" ? BAR_LIST_DATA : tab.items,
}));

function formatShare(value: number, total: number) {
  const percentage = safeRatio(value, total) * 100;
  return percentage > 0 && percentage < 0.5 ? "<0.5%" : `${Math.round(percentage)}%`;
}

export function BarListCard({
  data,
  items,
  tabs: provided,
  title = "Pages",
  unit,
  metricLabel = "Visitors",
  metric = "share",
  tone = 6,
  limit = 5,
  defaultTab,
  onTabChange,
  mono = false,
  className,
  height = "auto",
}: BarListCardProps = {}) {
  const tabs = provided ?? (data || items ? undefined : TABS),
    [selected, setSelected] = useState(defaultTab ?? tabs?.[0]?.id),
    [expanded, setExpanded] = useState(false),
    [hover, setHover] = useState<string | null>(null),
    reduced = useHydratedReducedMotion();
  const rows = (tabItems: BarListItem[]) => {
    const sorted = [...tabItems].sort((a, b) => b.value - a.value),
      total = sum(sorted.map((r) => r.value)),
      maximum = Math.max(0, ...sorted.map((r) => r.value));
    return (
      <>
        <ul className="-mx-2 flex flex-col gap-1">
          {(expanded ? sorted : sorted.slice(0, Math.max(0, limit))).map((row) => (
            <li
              key={row.label}
              onMouseEnter={() => setHover(row.label)}
              onMouseLeave={() => setHover(null)}
              className="relative flex h-9 items-center overflow-hidden rounded-lg px-2.5"
            >
              <motion.div
                key={reduced ? "reduced" : "animated"}
                className="absolute inset-y-0 left-0 rounded-lg"
                initial={reduced ? false : { width: 0 }}
                animate={{ width: `${safeRatio(row.value, maximum) * 100}%` }}
                transition={{ duration: reduced ? 0 : 0.5, ease: "easeOut" }}
                style={{
                  background: mono
                    ? "var(--color-text-tertiary)"
                    : (row.color ?? `var(--color-chart-${row.tone ?? tone})`),
                  opacity: hover === row.label ? 0.3 : 0.14,
                }}
              />
              <span className="relative flex min-w-0 flex-1 items-center gap-2 text-body-regular text-text-primary">
                <span className="text-text-secondary">{row.icon}</span>
                {row.href ? (
                  <a
                    href={row.href}
                    className="truncate outline-none hover:underline focus-visible:ring-2 focus-visible:ring-border-focus-ring"
                  >
                    {row.label}
                  </a>
                ) : (
                  <span className="truncate">{row.label}</span>
                )}
              </span>
              <span className="relative text-body-medium text-text-primary tabular-nums">
                {metric === "value" ? formatNumber(row.value) : formatShare(row.value, total)}
              </span>
            </li>
          ))}
        </ul>
        {sorted.length > limit && (
          <button
            type="button"
            onClick={() => setExpanded(!expanded)}
            className="mx-auto mt-3 block rounded-full border border-border-button-default bg-background-primary-default px-3 py-1 text-body-2-medium text-text-secondary outline-none focus-visible:ring-2 focus-visible:ring-border-focus-ring"
          >
            {expanded ? "Show fewer" : `Show ${sorted.length - limit} more`}
          </button>
        )}
      </>
    );
  };
  const metricHeading = (
    <span className="shrink-0 pb-2.5 text-caption-1-medium tracking-[0.06em] text-text-tertiary uppercase">
      {unit ?? metricLabel}
    </span>
  );
  return (
    <section
      className={cx(
        "flex w-full min-w-0 flex-col rounded-2xl bg-background-secondary-default px-4 pt-1 pb-2",
        className,
      )}
      style={height === "auto" ? undefined : { height }}
    >
      {tabs ? (
        <Tabs
          selectedKey={selected}
          onSelectionChange={(key) => {
            setSelected(String(key));
            setExpanded(false);
            onTabChange?.(String(key));
          }}
          className="flex flex-col gap-3"
        >
          <div className="-mx-4 flex items-end gap-3 border-b border-separator-border px-4">
            <TabList
              aria-label="Breakdown"
              className="flex min-w-0 flex-1 overflow-x-auto [scrollbar-width:none]"
            >
              {tabs.map((tab) => (
                <Tab
                  key={tab.id}
                  id={tab.id}
                  className={({ isSelected }) =>
                    cx(
                      "cursor-pointer whitespace-nowrap border-b-2 px-2.5 py-[7px] text-body-regular outline-none focus-visible:ring-2 focus-visible:ring-border-focus-ring",
                      isSelected
                        ? "border-accent-600 text-accent-600"
                        : "border-transparent text-text-primary",
                    )
                  }
                >
                  {tab.label}
                </Tab>
              ))}
            </TabList>
            {metricHeading}
          </div>
          {tabs.map((tab) => (
            <TabPanel key={tab.id} id={tab.id} className="outline-none">
              {rows(tab.items)}
            </TabPanel>
          ))}
        </Tabs>
      ) : (
        <>
          <div className="-mx-4 mb-3 flex items-end justify-between border-b border-separator-border px-4">
            <p className="py-2 text-body-medium text-text-primary">{title}</p>
            {metricHeading}
          </div>
          {rows(items ?? data ?? [])}
        </>
      )}
    </section>
  );
}
