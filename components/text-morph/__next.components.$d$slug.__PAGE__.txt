import { useEffect, useLayoutEffect, useRef, useState, type CSSProperties } from "react";
import { typeMatchGlyphs } from "./typography-math";
export interface TextMorphProps {
  text?: string;
  duration?: number;
  className?: string;
  style?: CSSProperties;
  onComplete?: () => void;
  rapidInputThreshold?: number;
}
/** Matching characters retain their DOM identity and move from their actual previous positions. */
export function TextMorph({
  text = "Running Shoes",
  duration = 0.65,
  rapidInputThreshold = 90,
  className,
  style,
  onComplete,
}: TextMorphProps) {
  const root = useRef<HTMLSpanElement>(null),
    ghostLayer = useRef<HTMLSpanElement>(null);
  const [glyphs, setGlyphs] = useState(() => typeMatchGlyphs([], text));
  const pending = useRef(false);
  const rendered = useRef(text),
    oldBoxes = useRef(new Map<number, { box: DOMRect; char: string }>());
  const animations = useRef<Animation[]>([]),
    lastChange = useRef(0),
    rapid = useRef(false),
    generation = useRef(0),
    fontReady = useRef(false);
  const clear = () => {
    generation.current++;
    animations.current.forEach((animation) => animation.cancel());
    animations.current = [];
    ghostLayer.current?.replaceChildren();
  };
  useEffect(() => {
    let disposed = false;
    (document.fonts?.ready ?? Promise.resolve()).then(() => {
      if (!disposed) fontReady.current = true;
    });
    const mq = matchMedia("(prefers-reduced-motion: reduce)");
    const change = () => {
      if (mq.matches) clear();
    };
    mq.addEventListener("change", change);
    return () => {
      disposed = true;
      clear();
      mq.removeEventListener("change", change);
    };
  }, []);
  useLayoutEffect(() => {
    if (rendered.current === text) return;
    const now = performance.now();
    rapid.current = now - lastChange.current < rapidInputThreshold;
    lastChange.current = now;
    // Measure before cancelling an interrupted transition, preserving its current visual position.
    oldBoxes.current = new Map(
      Array.from(root.current!.querySelectorAll<HTMLElement>("[data-glyph-id]")).map((span) => [
        Number(span.dataset.glyphId),
        { box: span.getBoundingClientRect(), char: span.textContent ?? "" },
      ]),
    );
    clear();
    pending.current = true;
    rendered.current = text;
    setGlyphs((old) => typeMatchGlyphs(old, text));
  }, [text, rapidInputThreshold]);
  useLayoutEffect(() => {
    const element = root.current;
    if (!element || !pending.current) return;
    pending.current = false;
    const old = oldBoxes.current;
    oldBoxes.current = new Map();
    if (
      matchMedia("(prefers-reduced-motion: reduce)").matches ||
      !fontReady.current ||
      typeof element.animate !== "function"
    )
      return;
    const token = ++generation.current,
      ms = Math.max(1, (rapid.current ? Math.min(0.08, duration) : duration) * 1000);
    const current = Array.from(element.querySelectorAll<HTMLElement>("[data-glyph-id]"));
    const retained = new Set(current.map((span) => Number(span.dataset.glyphId)));
    current.forEach((span) => {
      const previous = old.get(Number(span.dataset.glyphId))?.box,
        box = span.getBoundingClientRect();
      const frames = previous
        ? [
            { transform: `translate(${previous.x - box.x}px,${previous.y - box.y}px)` },
            { transform: "translate(0,0)" },
          ]
        : [
            { transform: "translateY(.4em) scale(.75)", opacity: 0 },
            { transform: "translateY(0) scale(1)", opacity: 1 },
          ];
      animations.current.push(
        span.animate(frames, { duration: ms, easing: "cubic-bezier(.2,.75,.25,1)" }),
      );
    });
    const bounds = element.getBoundingClientRect();
    for (const [id, { box, char }] of old) {
      if (retained.has(id)) continue;
      const ghost = document.createElement("span");
      ghost.textContent = char;
      Object.assign(ghost.style, {
        position: "absolute",
        left: `${box.x - bounds.x}px`,
        top: `${box.y - bounds.y}px`,
        whiteSpace: "pre",
      });
      ghostLayer.current?.append(ghost);
      animations.current.push(
        ghost.animate(
          [
            { opacity: 1, transform: "translateY(0)" },
            { opacity: 0, transform: "translateY(-.3em)" },
          ],
          { duration: ms * 0.6, fill: "forwards", easing: "ease-out" },
        ),
      );
    }
    Promise.all(
      animations.current.map((animation) => animation.finished.catch(() => undefined)),
    ).then(() => {
      if (generation.current === token) {
        ghostLayer.current?.replaceChildren();
        onComplete?.();
      }
    });
  }, [glyphs, duration, onComplete]);
  return (
    <span
      ref={root}
      className={className}
      style={{ position: "relative", display: "inline-block", ...style }}
      aria-label={text}
      data-typography="text-morph"
    >
      <span aria-hidden="true">
        {glyphs.map((glyph) => (
          <span
            key={glyph.id}
            data-glyph-id={glyph.id}
            style={{ display: "inline-block", whiteSpace: "pre" }}
          >
            {glyph.char}
          </span>
        ))}
      </span>
      <span
        ref={ghostLayer}
        aria-hidden="true"
        style={{ position: "absolute", inset: 0, pointerEvents: "none" }}
      />
    </span>
  );
}

/** Grapheme splitting keeps accents and composed emoji in one animated unit. */
export function typeGraphemes(text: string): string[] {
  return typeof Intl.Segmenter === "function"
    ? Array.from(
        new Intl.Segmenter(undefined, { granularity: "grapheme" }).segment(text),
        (item) => item.segment,
      )
    : Array.from(text);
}
export const typeClamp = (n: number) => Math.max(0, Math.min(1, n));
export function typeStaggerProgress(progress: number, index: number, count: number, spread = 0.55) {
  const stagger = Math.max(0, Math.min(0.95, spread));
  return typeClamp((progress - (index / Math.max(1, count - 1)) * stagger) / (1 - stagger));
}

export interface TypeGlyphIdentity {
  id: number;
  char: string;
}
/** Retain the nearest unused copy of each glyph, including repeated characters. */
export function typeMatchGlyphs(previous: TypeGlyphIdentity[], text: string): TypeGlyphIdentity[] {
  const used = new Set<number>();
  let nextId = Math.max(-1, ...previous.map((glyph) => glyph.id)) + 1;
  return typeGraphemes(text).map((char, index) => {
    const matches = previous
      .map((glyph, oldIndex) => ({ glyph, oldIndex }))
      .filter(({ glyph }) => glyph.char === char && !used.has(glyph.id))
      .sort((a, b) => Math.abs(a.oldIndex - index) - Math.abs(b.oldIndex - index));
    const match = matches[0]?.glyph;
    if (match) {
      used.add(match.id);
      return match;
    }
    return { char, id: nextId++ };
  });
}
