/** Footer uncovered on scroll with a translucent expanding quick-navigation dock. */
export function DockFooter({
  brand,
  groups,
  headline,
  description,
  copyright,
  children,
  background = "#353710",
  backgroundImage,
  open,
  defaultOpen = false,
  onOpenChange,
  onNavigate,

  duration = 0.65,
  revealDistance = 620,
  scrollContainer,
  onProgress,
  height = 720,
  className = "",
  style,
}: DockFooterProps) {
  const env = useSectionEnvironment(),
    progress = useSectionScroll(env.ref, scrollContainer, onProgress),
    [local, setLocal] = useState(defaultOpen),
    bottom = progress >= 0.985,
    atBottom = useRef(false),
    expanded = open ?? local;
  const change = (next: boolean) => {
    if (open === undefined) setLocal(next);
    onOpenChange?.(next);
  };
  useEffect(() => {
    if (bottom && !atBottom.current && autoOpen) {
      if (open === undefined) setLocal(true);
      onOpenChange?.(true);
    }
    atBottom.current = bottom;
  }, [bottom, autoOpen, open, onOpenChange]);
  const t = {
      duration: env.reduced ? 0 : Math.max(0, duration),
      ease: [0.22, 0.8, 0.2, 1] as const,
    },
    links = groups.flatMap((group) => group.links).slice(0, 6);
  return (
    <div
      ref={env.ref}
      className={`z-as z-as-dock-footer ${className}`}
      data-section="dock-footer"
      data-progress={progress.toFixed(4)}
      data-open={expanded}

export const sectionClamp = (value: number, min = 0, max = 1) =>
  Math.min(max, Math.max(min, Number.isFinite(value) ? value : 0));
export const sectionWrap = (value: number, count: number) =>
  count > 0 ? ((value % count) + count) % count : 0;
export type SectionPoint = readonly [number, number, number];
export function spherePoint(latitude: number, longitude: number): SectionPoint {
  const lat = (latitude * Math.PI) / 180,
    lon = (longitude * Math.PI) / 180;
  return [
    Math.cos(lat) * Math.sin(lon),
    -Math.sin(lat),
    Math.cos(lat) * Math.cos(lon),
  ];
}
export function rotatePoint(
  [x, y, z]: SectionPoint,
  yaw: number,
  pitch: number,
): SectionPoint {
  const xx = x * Math.cos(yaw) + z * Math.sin(yaw),
    zz = z * Math.cos(yaw) - x * Math.sin(yaw);
  return [
    xx,
    y * Math.cos(pitch) - zz * Math.sin(pitch),
    zz * Math.cos(pitch) + y * Math.sin(pitch),
  ];
}
export function fibonacciPoint(index: number, count: number): SectionPoint {
  const y = 1 - (2 * (index + 0.5)) / Math.max(1, count),
    r = Math.sqrt(1 - y * y),
    angle = index * Math.PI * (3 - Math.sqrt(5));
  return [Math.cos(angle) * r, y, Math.sin(angle) * r];
}
