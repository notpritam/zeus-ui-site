import { type CSSProperties, type ReactNode, useEffect, useRef } from "react";

export interface ImageTrailProps {
  images: readonly (string | { src: string; alt?: string })[];
  children?: ReactNode;
  threshold?: number;
  delay?: number;
  imageWidth?: number;
  imageHeight?: number;
  entranceDuration?: number;
  exitDuration?: number;
  exitDelay?: number;
  rotation?: number;
  maxImages?: number;
  className?: string;
  style?: CSSProperties;
  height?: CSSProperties["height"];
}

/** Decorative, distance-gated image instances. Children retain normal pointer and keyboard input. */
export function ImageTrail({
  images,
  children,
  threshold = 50,
  delay = 40,
  imageWidth = 180,
  imageHeight = 240,
  entranceDuration = 0.35,
  exitDuration = 0.7,
  exitDelay = 0.55,
  rotation = 10,
  maxImages = 24,
  className,
  style,
  height,
}: ImageTrailProps) {
  const host = useRef<HTMLDivElement>(null),
    layer = useRef<HTMLDivElement>(null);
  const latest = useRef({
    images,
    threshold,
    delay,
    imageWidth,
    imageHeight,
    entranceDuration,
    exitDuration,
    exitDelay,
    rotation,
    maxImages,
  });
  latest.current = {
    images,
    threshold,
    delay,
    imageWidth,
    imageHeight,
    entranceDuration,
    exitDuration,
    exitDelay,
    rotation,
    maxImages,
  };
  useEffect(() => {
    const root = host.current,
      surface = layer.current;
    if (!root || !surface) return;
    const motion = matchMedia("(prefers-reduced-motion: reduce)"),
      touch = matchMedia("(pointer: coarse)");
    const active = new Map<HTMLImageElement, Animation>();
    let previous: { x: number; y: number } | null = null,
      emitted = -Infinity,
      index = 0,
      visible = true;
    const clear = () => {
      for (const [image, animation] of active) {
        animation.cancel();
        image.remove();
      }
      active.clear();
      previous = null;
    };
    const finite = (n: number, fallback: number, min = 0) =>
      Number.isFinite(n) ? Math.max(min, n) : fallback;
    const move = (event: PointerEvent) => {
      if (
        !visible ||
        document.hidden ||
        motion.matches ||
        touch.matches ||
        event.pointerType === "touch"
      )
        return;
      const p = latest.current,
        rect = root.getBoundingClientRect();
      if (!p.images.length) return;
      const point = { x: event.clientX - rect.left, y: event.clientY - rect.top },
        now = performance.now();
      if (
        previous &&
        (Math.hypot(point.x - previous.x, point.y - previous.y) < finite(p.threshold, 50, 1) ||
          now - emitted < finite(p.delay, 40))
      )
        return;
      const from = previous ?? point;
      previous = point;
      emitted = now;
      while (active.size >= Math.min(60, finite(p.maxImages, 24, 1))) {
        const first = active.keys().next().value!;
        active.get(first)?.cancel();
        first.remove();
        active.delete(first);
      }
      const media = p.images[index++ % p.images.length],
        image = document.createElement("img");
      image.src = typeof media === "string" ? media : media.src;
      image.alt = "";
      image.draggable = false;
      image.dataset.trailImage = "";
      Object.assign(image.style, {
        position: "absolute",
        left: "0",
        top: "0",
        width: `${Math.min(rect.width * 0.65, finite(p.imageWidth, 180, 1))}px`,
        height: `${finite(p.imageHeight, 240, 1)}px`,
        objectFit: "cover",
        borderRadius: "6px",
        boxShadow: "0 12px 30px #0002",
        willChange: "transform, opacity",
      });
      surface.append(image);
      const angle = Math.sin(index * 2.39996) * finite(p.rotation, 10);
      const transform = (x: number, y: number, scale: number) =>
        `translate(${x}px,${y}px) translate(-50%,-50%) rotate(${angle}deg) scale(${scale})`;
      const enter = finite(p.entranceDuration, 0.35),
        hold = finite(p.exitDelay, 0.55),
        exit = finite(p.exitDuration, 0.7),
        total = Math.max(0.01, enter + hold + exit);
      const animation = image.animate(
        [
          {
            transform: transform(
              from.x + (point.x - from.x) * 0.45,
              from.y + (point.y - from.y) * 0.45,
              0.75,
            ),
            opacity: 0,
            offset: 0,
            easing: "cubic-bezier(.16,1,.3,1)",
          },
          { transform: transform(point.x, point.y, 1), opacity: 1, offset: enter / total },
          {
            transform: transform(point.x, point.y, 1),
            opacity: 1,
            offset: (enter + hold) / total,
            easing: "cubic-bezier(.4,0,.7,1)",
          },
          { transform: transform(point.x, point.y + 24, 0), opacity: 0, offset: 1 },
        ],
        { duration: total * 1000, fill: "both" },
      );
      active.set(image, animation);
      animation.onfinish = () => {
        active.delete(image);
        image.remove();
      };
    };
    const leave = () => {
      previous = null;
    };
    const visibility = () => {
      if (document.hidden) clear();
    };
    const observer = new IntersectionObserver((entries) => {
      visible = !!entries[0]?.isIntersecting;
      if (!visible) clear();
    });
    observer.observe(root);
    root.addEventListener("pointermove", move);
    root.addEventListener("pointerleave", leave);
    motion.addEventListener("change", clear);
    touch.addEventListener("change", clear);
    document.addEventListener("visibilitychange", visibility);
    return () => {
      clear();
      observer.disconnect();
      root.removeEventListener("pointermove", move);
      root.removeEventListener("pointerleave", leave);
      motion.removeEventListener("change", clear);
      touch.removeEventListener("change", clear);
      document.removeEventListener("visibilitychange", visibility);
    };
  }, []);
  return (
    <div
      ref={host}
      data-image-trail
      className={className}
      style={{
        position: "relative",
        isolation: "isolate",
        width: "100%",
        height: height ?? "var(--zeus-animation-height, 480px)",
        overflow: "hidden",
        ...style,
      }}
    >
      {children}
      <div
        ref={layer}
        aria-hidden
        style={{ position: "absolute", inset: 0, zIndex: 3, pointerEvents: "none" }}
      />
    </div>
  );
}
