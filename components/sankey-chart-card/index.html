Installation

Add sankey chart to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.2.1.tgz zeus-ui add sankey-chart-card

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — chart-card, chevrons, chip, cx, segmented-control, select, tokens, use-count-up, use-dismiss-on-outside-press. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

"use client";

import { SankeyChartCard } from "@/components/application/charts/sankey-chart-card";

  return (
    <div className="w-full max-w-2xl">
      <SankeyChartCard />
    </div>
  );
}

export function useChartRange<T extends ChartRange>(
  ranges: T[] | undefined,
  defaultRange?: string,
  onChange?: (id: string) => void,
) {
  const [selected, setSelected] = useState(defaultRange ?? ranges?.[0]?.id);
  const current = ranges?.find((range) => range.id === selected) ?? ranges?.[0];
  const select = (id: string) => {
    setSelected(id);
    onChange?.(id);
  };
  return { current, id: current?.id, select };
}

export function AnimatedNumber({
  value,
  format = formatNumber,
}: {
  value: number;
  format?: (value: number) => string;
}) {
  const reduced = useHydratedReducedMotion();
  const [display, setDisplay] = useState(value);
  const current = useRef(value);
  useEffect(() => {
    if (reduced) {
      current.current = value;
      setDisplay(value);
      return;
    }
    const from = current.current,
      start = performance.now();
    let frame = 0;
    const tick = (time: number) => {
      const t = Math.min(1, (time - start) / 300);
      current.current = from + (value - from) * (1 - (1 - t) ** 3);
      setDisplay(current.current);
      if (t < 1) frame = requestAnimationFrame(tick);
    };
    frame = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(frame);
  }, [value, reduced]);
  return <>{format(reduced ? value : display)}</>;
}

  suffix,

                  delta >= 0

                    : "bg-status-rose-background text-status-rose-text",
                )}
              >
                {delta > 0 ? "+" : ""}
                {+(delta * 100).toFixed(1)}%
              </span>
            )}
          </div>
        </div>
        {action}
      </div>
      {children}
      {footer}
    </section>
  );
}

export function PulseDot({ cx: x, cy: y, stroke }: { cx?: number; cy?: number; stroke?: string }) {
  const reduced = useHydratedReducedMotion();
  if (x == null || y == null) return null;
  return (
    <g>
      <motion.circle
        key={reduced ? "reduced" : "animated"}
        cx={x}
        cy={y}
        fill={stroke}
        initial={false}
        animate={reduced ? { r: 7, opacity: 0.2 } : { r: [5, 11, 5], opacity: [0.25, 0, 0.25] }}
        transition={reduced ? { duration: 0 } : { duration: 1.6, repeat: Infinity }}
      />
      <circle
        cx={x}
        cy={y}
        r={3.5}
        fill={stroke}
        stroke="var(--color-background-inner-default)"
        strokeWidth={2}
      />
    </g>
  );
}
