import { useRef, type ReactNode } from "react";
import { useScrollMotion, scrollStage, type ScrollMotionProps } from "./scroll-shared";
import { scrollFrame, type ScrollFrame } from "./scroll-math";
export type FlipZoneFrame = ScrollFrame;
export interface FlipZoneProps extends ScrollMotionProps {
  children: ReactNode /** Percentage-based frames. The same media node travels through every frame. */;
  frames: readonly FlipZoneFrame[];
  background?: ReactNode;
  showFrames?: boolean;
}
export function FlipZone({
  children,
  frames,
  background,
  showFrames = false,
  ...props
}: FlipZoneProps) {
  const root = useRef<HTMLDivElement>(null);
  useScrollMotion(
    root,
    props,
    ({ progress: p }) => {
      const f = scrollFrame(frames, p),
        el = root.current!.querySelector<HTMLElement>(":scope > [data-scroll-media]")!;
      Object.assign(el.style, {
        left: `${f.x}%`,
        top: `${f.y}%`,
        width: `${f.width}%`,
        height: `${f.height}%`,
        transform: `rotate(${f.rotation}deg)`,
        borderRadius: `${f.radius}px`,
      });
      root.current!.dataset.zone = String(
        Math.min(frames.length - 1, Math.floor(p * Math.max(0, frames.length - 1))),
      );
    },
    JSON.stringify(frames),
  );
  const first = frames[0];
  return (
    <div
      ref={root}
      data-scroll-animation="flip-zone"
      className={props.className}
      style={{ ...scrollStage, ...props.style }}
    >
      {background}
      {showFrames &&
        frames.map((f, i) => (
          <div
            aria-hidden="true"
            key={i}
            style={{
              position: "absolute",
              left: `${f.x}%`,
              top: `${f.y}%`,
              width: `${f.width}%`,
              height: `${f.height}%`,
              border: "1px dashed #8885",
              borderRadius: f.radius,
            }}
          />
        ))}
      <div
        data-scroll-media=""
        style={{
          position: "absolute",
          left: `${first?.x ?? 0}%`,
          top: `${first?.y ?? 0}%`,
          width: `${first?.width ?? 100}%`,
          height: `${first?.height ?? 100}%`,
          overflow: "hidden",
        }}
      >
        {children}
      </div>
    </div>
  );
}

export function useScrollMotion<T extends HTMLElement>(

  options: ScrollMotionProps & ScrollEntranceOptions,
  draw: (metrics: ScrollMetrics) => void,
  key = "",
  preserveProgress = false,
  pageProgress = false,
) {

  latest.current = { options, draw };
  const refresh = useRef<(() => void) | null>(null);
  const {
    scrollContainer,
    scrollTarget,
    range,
    start = 0.9,
    end = 0.1,
    scrub = 0.12,
    trigger = "scroll",
    duration = 1.2,
    delay = 0,
    once = true,
    replayKey,
  } = options;

      if (!ownsRoot()) return;
      const reduced = media.matches,
        value = reduced && !preserveProgress ? 1 : scrollClamp(p);
      latest.current.draw({
        progress: value,
        raw,
        velocity: reduced ? 0 : velocity,

        width,
        height,
      });

      el.dataset.velocity = String(reduced ? 0 : velocity);
      el.dataset.reduced = String(reduced);

      frame = 0;

      if (!frame && ownsRoot() && !document.hidden && visible && !media.matches) {
        previousTime = 0;
        frame = requestAnimationFrame(tick);
      }
    };

      if (!ownsRoot()) return;

                frame = 0;

                  p = 0;
                }

            },

          )

      entryTime = performance.now();

        cancelAnimationFrame(frame);
        frame = 0;
        velocity = 0;
      } else measure();
    };

    media.addEventListener("change", measure);
    document.addEventListener("visibilitychange", visibility);

      disposed = true;
      refresh.current = null;

      resize?.disconnect();

      window.removeEventListener("resize", measure);

    scrollContainer,
    scrollTarget,
    range?.[0],
    range?.[1],
    start,
    end,
    scrub,
    trigger,
    duration,
    delay,
    once,
    replayKey,
    key,
    preserveProgress,
    pageProgress,
  ]);

    refresh.current?.();
  }, [options.progress]);
}
