Installation

Add design tokens to your project. New to the library? Set up the design system first.

CLI
Manual
bash
npx --yes --allow-remote=all --package=https://myzeusui.com/install/zeus-ui-0.3.0.tgz zeus-ui add tokens

Run in your React project. Source, dependencies and required foundations are copied together. For your first Zeus component, import the styles once.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

tokens-demo.tsx
"use client";

import { cx } from "@/utils/cx";

const SURFACES = [
  { label: "Primary", className: "bg-background-primary-default" },
  { label: "Secondary", className: "bg-background-secondary-default" },
  { label: "Inner", className: "bg-background-inner-default" },
];
const CHART_COLORS = [
  "bg-chart-1",
  "bg-chart-2",
  "bg-chart-3",
  "bg-chart-4",
  "bg-chart-5",
  "bg-chart-6",
  "bg-chart-7",
  "bg-chart-8",
  "bg-chart-9",
];

export default function TokensDemo() {
  return (
    <div className="flex w-full max-w-xl flex-col gap-6 rounded-3xl border border-border-button-default bg-background-full p-6">
      <div className="flex flex-col gap-1">
        <p className="text-title-2-medium text-text-primary">A shared visual language</p>
        <p className="text-body-regular text-text-secondary">
          Surfaces, text, and data colors adapt to the selected theme.
        </p>
      </div>
      <div className="grid grid-cols-3 gap-3">
        {SURFACES.map((surface) => (
          <div
            key={surface.label}
            className={cx(
              surface.className,
              "flex h-24 items-end rounded-2xl border border-border-button-default p-3",
            )}
          >
            <span className="text-body-2-medium text-text-primary">{surface.label}</span>
          </div>
        ))}
      </div>
      <div className="flex flex-wrap gap-x-6 gap-y-2 text-body-medium">
        <span className="text-text-primary">Primary text</span>
        <span className="text-text-secondary">Secondary text</span>
        <span className="text-text-tertiary">Supporting text</span>
      </div>
      <div className="flex flex-col gap-2">
        <p className="text-body-2-medium text-text-secondary">Chart palette</p>
        <div className="grid grid-cols-9 gap-2">
          {CHART_COLORS.map((className, index) => (
            <span
              key={className}
              role="img"
              aria-label={`Chart color ${index + 1}`}
              className={cx(className, "aspect-square rounded-lg")}
            />
          ))}
        </div>
      </div>
    </div>
  );
}

/* ---------------------------------------------------------------------------
 * @zeus/tokens — the single stylesheet an app imports.
 *
 *   import "@zeus/tokens";
 *
 * Pulls in Tailwind v4, the design tokens, the type scale, and the preflight
 * and component layer. Dark mode is class-based: put `.dark` on <html> and
 * every semantic token flips. Components reference semantic tokens
 * (`bg-background-secondary-default`, `text-text-primary`) and never `dark:`.
 * ------------------------------------------------------------------------- */

@import "tailwindcss";
@import "./theme.css";
@import "./typography.css";

@custom-variant dark (&:where(.dark, .dark *));

@import "./preflight.css";
