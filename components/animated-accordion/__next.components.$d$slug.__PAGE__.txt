import { useId, useRef, useState, type ReactNode } from "react";
import { gsap } from "gsap";
import { useGSAP } from "@gsap/react";
import { SplitText } from "gsap/dist/SplitText";

gsap.registerPlugin(useGSAP, SplitText);

export interface AnimatedAccordionItem {
  id: string;
  title: ReactNode;
  content: ReactNode;
  disabled?: boolean;
}
export interface AnimatedAccordionProps {
  items: AnimatedAccordionItem[];
  allowMultiple?: boolean;
  defaultOpenIds?: string[];
  openIds?: string[];
  onOpenChange?: (ids: string[]) => void;
  duration?: number;
  ease?: string;
  iconMode?: "rotate" | "fade" | "both";
  iconRotation?: number;
  enableStagger?: boolean;
  staggerDuration?: number;
  staggerDelay?: number;
  className?: string;
}

const css = `.z-aa{width:100%;color:var(--color-text-primary,#0c0c0c)}.z-aa-item{border-bottom:1px solid var(--color-border-secondary,#e6e6e6)}.z-aa-item:last-child{border-bottom:0}.z-aa-trigger{display:flex;width:100%;padding:20px 0;justify-content:space-between;gap:24px;align-items:center;border:0;background:none;text-align:left;color:inherit;cursor:pointer;font:500 16px/1.4 inherit;font-size:16px;font-weight:500}.z-aa-trigger:disabled{opacity:.4;cursor:default}.z-aa-trigger:focus-visible{outline:2px solid #a4fb81;outline-offset:4px}.z-aa-trigger svg{width:16px;height:16px;flex-shrink:0}.z-aa-panel{overflow:hidden}.z-aa-text{padding:0 0 20px;font-size:14px;line-height:1.6;color:var(--color-text-secondary,#737373)}`;

function Panel({
  item,
  open,
  toggle,
  duration = 0.8,
  ease = "expo.inOut",
  iconMode = "both",
  iconRotation = -180,
  enableStagger = false,
  staggerDuration = 0.6,
  staggerDelay = 0.15,
}: { item: AnimatedAccordionItem; open: boolean; toggle: () => void } & AnimatedAccordionProps) {
  const id = useId();
  const panel = useRef<HTMLDivElement>(null),
    icon = useRef<SVGSVGElement>(null),
    bar = useRef<SVGPathElement>(null),
    text = useRef<HTMLDivElement>(null);
  const mounted = useRef(false);
  useGSAP(
    () => {
      const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
      const d = !mounted.current || reduced ? 0 : Math.max(0, duration);
      mounted.current = true;
      let split: SplitText | undefined;
      gsap.to(panel.current, { height: open ? "auto" : 0, duration: d, ease, overwrite: true });
      gsap.to(icon.current, {
        rotation: open && iconMode !== "fade" ? iconRotation : 0,
        duration: d,
        ease,
        overwrite: true,
      });
      gsap.to(bar.current, {
        opacity: open && iconMode !== "rotate" ? 0 : 1,
        duration: d * 0.5,
        overwrite: true,
      });
      if (open && enableStagger && !reduced && text.current) {
        split = SplitText.create(text.current, { type: "lines", mask: "lines" });
        gsap.from(split.lines, {
          yPercent: 110,
          duration: staggerDuration,
          stagger: staggerDelay,
          ease: "expo.out",
        });
      }
      return () => split?.revert();
    },
    {
      scope: panel,
      dependencies: [
        open,
        duration,
        ease,
        iconMode,
        iconRotation,
        enableStagger,
        staggerDuration,
        staggerDelay,
      ],
      revertOnUpdate: true,
    },
  );
  return (
    <div className="z-aa-item">
      <button
        className="z-aa-trigger"
        type="button"
        id={`${id}-trigger`}
        aria-controls={`${id}-panel`}
        aria-expanded={open}
        disabled={item.disabled}
        onClick={toggle}
      >
        {item.title}
        <svg ref={icon} viewBox="0 0 16 16" fill="none" aria-hidden="true">
          <path d="M2 8h12" stroke="currentColor" strokeWidth="1.5" />
          <path ref={bar} d="M8 2v12" stroke="currentColor" strokeWidth="1.5" />
        </svg>
      </button>
      <div
        ref={panel}
        className="z-aa-panel"
        id={`${id}-panel`}
        role="region"
        aria-labelledby={`${id}-trigger`}
        aria-hidden={!open}
        inert={!open}
        style={{ height: open ? "auto" : 0 }}
      >
        <div ref={text} className="z-aa-text">
          {item.content}
        </div>
      </div>
    </div>
  );
}

export function AnimatedAccordion({
  items,
  allowMultiple = false,
  defaultOpenIds = [],
  openIds,
  onOpenChange,
  className = "",
  ...motion
}: AnimatedAccordionProps) {
  const [internal, setInternal] = useState(() =>
    allowMultiple ? defaultOpenIds : defaultOpenIds.slice(0, 1),
  );
  const open = openIds ?? internal;
  const toggle = (id: string) => {
    const next = open.includes(id)
      ? open.filter((key) => key !== id)
      : allowMultiple
        ? [...open, id]
        : [id];
    if (openIds === undefined) setInternal(next);
    onOpenChange?.(next);
  };
  return (
    <>
      <style>{css}</style>
      <div className={`z-aa ${className}`}>
        {items.map((item) => (
          <Panel
            {...motion}
            items={items}
            key={item.id}
            item={item}
            open={open.includes(item.id)}
            toggle={() => toggle(item.id)}
          />
        ))}
      </div>
    </>
  );
}
