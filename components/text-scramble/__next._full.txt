import { useRef } from "react";
import {
  typeUnits,
  typeInteractive,
  useTypeMotion,
  type TypographyMotionProps,
} from "./typography-shared";
import { typeGraphemes, typeStaggerProgress } from "./typography-math";
export interface TextScrambleProps extends TypographyMotionProps {
  characters?: string;
  scrambleSpeed?: number;
  stagger?: number;
}
export function TextScramble({
  text = "OUTPACE YOUR LIMITS",
  characters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789",
  scrambleSpeed = 30,
  stagger = 0.5,
  ...props
}: TextScrambleProps) {
  const root = useRef<HTMLSpanElement>(null);
  const target = typeGraphemes(text),
    pool = typeGraphemes(characters || "ABCDEFGHIJKLMNOPQRSTUVWXYZ");
  useTypeMotion(
    root,
    props,
    (p) => {
      typeUnits(root.current!).forEach((unit, i) => {
        const q = typeStaggerProgress(p, i, target.length, stagger);
        const char = target[i] ?? "";
        unit.textContent =
          q >= 1 || /\s/.test(char)
            ? char
            : pool[
                Math.abs(
                  Math.floor(
                    Math.sin((Math.floor(p * Math.max(1, scrambleSpeed) * 20) + 1) * (i + 7)) *
                      10000,
                  ),
                ) % pool.length
              ]!;
      });
    },
    `${text}:${characters}:${scrambleSpeed}:${stagger}`,
  );
  return (
    <span
      ref={root}
      className={props.className}
      style={{ display: "inline-block", ...props.style }}
      aria-label={text}
      data-typography="text-scramble"
      {...typeInteractive(props.trigger)}
    >
      <span aria-hidden="true">
        {target.map((char, i) => (
          <span
            key={i}
            style={{ position: "relative", display: "inline-block", whiteSpace: "pre" }}
          >
            <span style={{ visibility: "hidden" }}>{char}</span>
            <span data-type-unit="" style={{ position: "absolute", inset: 0 }}>
              {char}
            </span>
          </span>
        ))}
      </span>
    </span>
  );
}

/** Grapheme splitting keeps accents and composed emoji in one animated unit. */
export function typeGraphemes(text: string): string[] {
  return typeof Intl.Segmenter === "function"
    ? Array.from(
        new Intl.Segmenter(undefined, { granularity: "grapheme" }).segment(text),
        (item) => item.segment,
      )
    : Array.from(text);
}

/** Retain the nearest unused copy of each glyph, including repeated characters. */
export function typeMatchGlyphs(previous: TypeGlyphIdentity[], text: string): TypeGlyphIdentity[] {
  const used = new Set<number>();
  let nextId = Math.max(-1, ...previous.map((glyph) => glyph.id)) + 1;
  return typeGraphemes(text).map((char, index) => {
    const matches = previous
      .map((glyph, oldIndex) => ({ glyph, oldIndex }))
      .filter(({ glyph }) => glyph.char === char && !used.has(glyph.id))
      .sort((a, b) => Math.abs(a.oldIndex - index) - Math.abs(b.oldIndex - index));
    const match = matches[0]?.glyph;
    if (match) {
      used.add(match.id);
      return match;
    }
    return { char, id: nextId++ };
  });
}

  draw: (progress: number, reduced: boolean) => void,
  contentKey = "",
) {

  controlled.current = options.progress;
  const currentRender = useRef<((progress: number) => void) | null>(null);
  const {

    once = true,
    replayKey,
    progress,
    scrollDistance,
  } = options;

    const element = root.current;
    if (!element) return;

      frame = 0,
      ready = false,
      active = true,
      playing = false,
      elapsed = 0,
      last = 0,
      completed = false;

    const ownsRoot = () => !disposed && root.current === element;
    const scrollParent = typeScrollParent(element);
    const render = (value: number) => {
      if (!ready || !ownsRoot()) return;
      const p = media.matches ? 1 : typeClamp(value);
      latest.current.draw(p, media.matches);
      element.dataset.progress = String(p);
      latest.current.onProgress?.(p);
      if (p < 1 && !once) completed = false;
      if (p >= 1 && !completed) {
        completed = true;
        latest.current.onComplete?.();
      }
    };
    currentRender.current = render;
    const scroll = () => {
      if (controlled.current !== undefined) {
        render(controlled.current);
        return;

        bounds = scrollParent?.getBoundingClientRect();

      const top = rect.top - (bounds?.top ?? 0);
      render((height * 0.92 - top) / Math.max(80, scrollDistance ?? height * 0.65));
    };

        render(controlled.current);
        return;

      render(p);

        !ownsRoot() ||
        controlled.current !== undefined ||

        return;
      cancelAnimationFrame(frame);
      elapsed = 0;
      last = 0;
      completed = false;
      playing = true;

      }

        start();
      }
    };

      cancelAnimationFrame(frame);
      frame = 0;
      last = 0;
      if (!ready || !ownsRoot()) return;
      if (media.matches) {
        render(1);
        return;
      }
      if (controlled.current !== undefined) {
        render(controlled.current);
        return;
      }
      if (trigger === "scroll") scroll();
      else if (playing && active && !document.hidden) frame = requestAnimationFrame(tick);
    };
    const observer =
      typeof IntersectionObserver !== "undefined"
        ? new IntersectionObserver((entries) => {
            active = entries[0]?.isIntersecting ?? true;
            sync();
          })
        : null;
    observer?.observe(element);
    media.addEventListener("change", sync);
    document.addEventListener("visibilitychange", sync);
    if (trigger === "hover") {
      element.addEventListener("pointerenter", start);
      element.addEventListener("focus", start);
    }
    if (trigger === "click") {
      element.addEventListener("click", start);
      element.addEventListener("keydown", keydown);
    }
    if (trigger === "scroll") {
      window.addEventListener("scroll", scroll, { passive: true });
      scrollParent?.addEventListener("scroll", scroll, { passive: true });
      window.addEventListener("resize", scroll);
    }
    (document.fonts?.ready ?? Promise.resolve()).then(() => {
      if (!ownsRoot()) return;
      ready = true;
      if (controlled.current !== undefined) render(controlled.current);
      else if (trigger === "load") start();
      else if (trigger === "scroll") scroll();
      else render(1);

      disposed = true;

      cancelAnimationFrame(frame);

      element.removeEventListener("focus", start);
      element.removeEventListener("click", start);
      element.removeEventListener("keydown", keydown);
      window.removeEventListener("scroll", scroll);
      scrollParent?.removeEventListener("scroll", scroll);
      window.removeEventListener("resize", scroll);
    };
  }, [root, trigger, duration, delay, once, replayKey, scrollDistance, contentKey]);

    if (progress !== undefined) currentRender.current?.(progress);
  }, [progress]);
