          y: shape.y,

          height: shape.height,

            x: center - 4 * liquid.neck,
            y: top,
            width: 8 * liquid.neck,
            height,
            opacity: liquid.neck,
            force3D: true,
            autoRound: false,
          });
        }
      };
      const startGoo = () => {
        gsap.set(plate.current, {
          x: anchor.x + 2,
          y: anchor.y + 2,
          width: anchor.width - 4,
          height: anchor.height - 4,

          button.removeEventListener("focus", enter);
          button.removeEventListener("blur", leave);
          button.removeEventListener("keydown", key);
          button.removeAttribute("aria-describedby");

        if (active && !hidden) show(active, true);
      };
      const resize = new ResizeObserver(reposition);
      resize.observe(el);
      el.addEventListener("pointerleave", leave);
      window.addEventListener("resize", reposition);

        tween?.kill();

        itemConfiguration,

        travelDuration,
