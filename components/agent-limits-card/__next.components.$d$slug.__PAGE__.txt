/** 1_000_000 → "1M", 820_000 → "820k", 69_900 → "69.9k". */
export function formatTokens(tokens: number): string {
  const abs = Math.abs(tokens);
  if (abs >= 1_000_000) return String(Math.round((tokens / 1_000_000) * 10) / 10) + "M";
  if (abs >= 1_000) return String(Math.round((tokens / 1_000) * 10) / 10) + "k";
  return String(tokens);
}
const share = (part: number, whole: number) => whole > 0 ? (part / whole) * 100 : 0;
const clamp = (fraction: number) => Math.min(Math.max(fraction, 0), 1);
const widthPercent = (part: number, whole: number) =>
  String(Math.round(clamp(whole > 0 ? part / whole : 0) * 1_000_000) / 10_000) + "%";
const palette = [6, 5, 3, 8, 7, 4, 1, 2, 9];
const segmentColor = (segment: AgentContextSegment, index: number) =>
  segment.color ?? (segment.deferred ? "var(--color-chart-cursor)" : "var(--color-chart-" + palette[index % palette.length] + ")");
// Measured from the live preview: height and opacity use the same curve.
const ease = [0.22, 1, 0.36, 1] as const;

  return (

        <RiArrowRightSLine aria-hidden className={cx("size-4 shrink-0 text-text-tertiary transition-transform duration-200 ease-out motion-reduce:transition-none", open && "rotate-90")} />
        <span className="min-w-0 flex-1 truncate text-body-regular text-text-secondary">{group.label}</span>
        <span className="text-body-regular text-text-tertiary tabular-nums">{formatTokens(group.tokens)}</span>
        <span className="w-14 text-right text-body-regular text-text-tertiary tabular-nums">{group.items.length}</span>
      </button>
      <AnimatePresence initial={false}>
        {open && (
          <motion.div id={panelId} initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }}
            exit={{ height: 0, opacity: 0 }} transition={{ duration: reducedMotion ? 0 : 0.26, ease }} className="overflow-hidden">
            <ul className="flex flex-col pb-1 pl-[22px]">
              {group.items.map(item => (
                <li key={item.label} className="flex items-center gap-2 py-1">

                  <span className="text-body-2-regular text-text-tertiary tabular-nums">{formatTokens(item.tokens)}</span>
                  <span className="w-14" aria-hidden />
                </li>
              ))}
            </ul>
          </motion.div>
        )}
      </AnimatePresence>
    </li>
  );
}

export function AgentLimitsCard({ context, plan, planHref, limits = [], defaultExpanded = false, expanded, onExpandedChange, className }: AgentLimitsCardProps) {
  const [internalExpanded, setInternalExpanded] = useState(defaultExpanded);
  // Keep group state outside the exiting panel so reopening restores it.
  const [openGroups, setOpenGroups] = useState<Record<string, boolean>>({});
  const open = expanded ?? internalExpanded;
  const reducedMotion = useReducedMotion() === true;
  const panelId = useId();
  const counted = context.segments.map((segment, index) => ({ segment, index })).filter(({ segment }) => !segment.deferred);
  const used = counted.reduce((sum, { segment }) => sum + segment.tokens, 0);
  const free = Math.max(context.max - used, 0);
  function toggle() {
    if (expanded === undefined) setInternalExpanded(!open);
    onExpandedChange?.(!open);
  }
  return (
    <section className={cx("flex w-full min-w-0 flex-col rounded-2xl bg-background-secondary-default px-4 pt-2.5 pb-4", className)}>
      <button type="button" aria-expanded={open} aria-controls={open ? panelId : undefined} onClick={toggle}
        className="group -mx-2 flex cursor-pointer items-center justify-between gap-3 rounded-2lg px-2 py-1 text-left outline-none transition-colors duration-150 hover:bg-background-secondary-hover focus-visible:ring-2 focus-visible:ring-border-focus-ring motion-reduce:transition-none">
        <span className="text-body-medium text-text-secondary">Context window</span>
        <span className="flex items-center gap-2">
          <span className="text-body-medium whitespace-nowrap text-text-secondary tabular-nums">
            {formatTokens(used)} / {formatTokens(context.max)}{" "}
            <span className="text-text-primary">({Math.round(share(used, context.max))}%)</span>
          </span>
          <RiArrowDownSLine aria-hidden className={cx("size-4 shrink-0 text-text-tertiary transition-transform duration-200 ease-out group-hover:text-text-secondary motion-reduce:transition-none", open && "rotate-180")} />
        </span>
      </button>
      <div role="progressbar" aria-label="Context window usage"
        aria-valuenow={Math.round(clamp(context.max > 0 ? used / context.max : 0) * 100)} aria-valuemin={0} aria-valuemax={100}
        className="mt-1.5 flex h-1.5 w-full gap-px overflow-hidden rounded-full bg-chart-track">
        {counted.map(({ segment, index }) => (
          <div key={segment.label} className="h-full shrink-0 transition-[width] duration-500 ease-out motion-reduce:transition-none"
            style={{ width: widthPercent(segment.tokens, context.max), backgroundColor: segmentColor(segment, index) }}
            title={segment.label + " · " + formatTokens(segment.tokens)} />
        ))}
      </div>
      <AnimatePresence initial={false}>
        {open && (
          <motion.div id={panelId} initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }}
            exit={{ height: 0, opacity: 0 }} transition={{ duration: reducedMotion ? 0 : 0.32, ease }} className="-mx-2 overflow-hidden px-2">
            <div className="flex flex-col pt-3">
              <ul className="flex flex-col">
                {context.segments.map((segment, index) => (
                  <li key={segment.label} className="flex items-center gap-2 py-[5px]">
                    <span aria-hidden className="size-2.5 shrink-0 rounded-[3px]" style={{ backgroundColor: segmentColor(segment, index) }} />
                    <span className="min-w-0 flex-1 truncate text-body-regular text-text-primary">{segment.label}</span>
                    <span className="text-body-regular text-text-tertiary tabular-nums">{formatTokens(segment.tokens)}</span>
                    <span className="w-14 text-right text-body-medium text-text-primary tabular-nums">{segment.deferred ? "—" : share(segment.tokens, context.max).toFixed(1) + "%"}</span>
                  </li>
                ))}
                <li className="flex items-center gap-2 py-[5px]">
                  <span aria-hidden className="size-2.5 shrink-0 rounded-[3px] bg-chart-track" />
                  <span className="min-w-0 flex-1 truncate text-body-regular text-text-primary">Free space</span>
                  <span className="text-body-regular text-text-tertiary tabular-nums">{formatTokens(free)}</span>
                  <span className="w-14 text-right text-body-medium text-text-primary tabular-nums">{share(free, context.max).toFixed(1)}%</span>
                </li>
              </ul>
              {context.groups && context.groups.length > 0 && (
                <ul className="mt-1.5 flex flex-col">
                  {context.groups.map(group => (
                    <GroupRow key={group.label} group={group} open={openGroups[group.label] === true}
                      onToggle={() => setOpenGroups(current => ({ ...current, [group.label]: !current[group.label] }))}
                      reducedMotion={reducedMotion} />
                  ))}
                </ul>
              )}
            </div>
          </motion.div>
        )}
      </AnimatePresence>
      {limits.length > 0 && (
        <>
          <div className="my-3 h-px w-full bg-separator-border-strong" />
          <div className="flex items-center justify-between gap-3 py-1">
            <span className="text-body-medium text-text-secondary">Plan usage limits{plan ? " · " + plan : ""}</span>
            {planHref && (
              <a href={planHref} aria-label="Manage plan"
                className="flex size-6 items-center justify-center rounded-md text-text-tertiary outline-none transition-colors duration-150 hover:bg-background-secondary-hover hover:text-text-secondary focus-visible:ring-2 focus-visible:ring-border-focus-ring motion-reduce:transition-none">
                <RiArrowRightLine aria-hidden className="size-4" />
              </a>

          </div>
          <ul className="flex flex-col gap-3 pt-1">

                <li key={limit.label} className="flex flex-col gap-1.5">
                  <div className="flex items-baseline justify-between gap-3">
                    <span className="min-w-0 truncate text-body-medium text-text-primary">{limit.label}</span>
                    <span className="flex shrink-0 items-baseline gap-2">
                      <span className="text-body-regular text-text-tertiary">{limit.resets}</span>

                    </span>
                  </div>

                    className="flex h-1.5 w-full gap-px overflow-hidden rounded-full bg-chart-track">
                    <div className="h-full rounded-full transition-[width] duration-500 ease-out motion-reduce:transition-none"

                  </div>
                </li>
              );
            })}
          </ul>
        </>
      )}
    </section>
  );
}
