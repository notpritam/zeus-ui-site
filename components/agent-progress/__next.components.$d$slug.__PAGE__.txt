import { AnimatePresence, motion, useIsPresent } from "motion/react";
import { type ReactNode, useEffect, useId, useLayoutEffect, useRef, useState } from "react";
import { useHydratedReducedMotion } from "../../hooks/use-hydrated-reduced-motion";
import { cx } from "../../utils/cx";

export interface AgentProgressProps {
  steps?: string[];
  /** Delay before timed work begins; entrance animation runs during this interval. */
  startDelay?: number;
  stepDuration?: number;
  completionDelay?: number;
  onFinished?: () => void;
  run?: boolean;
  defaultMinimized?: boolean;
  className?: string;
}
export const AGENT_PROGRESS_STEPS = [
  "Read project files",
  "Update light mode tokens",
  "Implement dark mode tokens",
  "Register the theme toggle",
  "Run lint and production build",
];
const EASE = [0.22, 1, 0.36, 1] as const;

function PresenceContents({ children }: { children: (present: boolean) => ReactNode }) {
  return children(useIsPresent());
}

function ProgressCircle({
  progress,
  done = false,
  waiting = false,
  size = 14,
}: {
  progress: number;
  done?: boolean;
  waiting?: boolean;
  size?: number;
}) {
  const reduce = useHydratedReducedMotion() === true;
  const radius = size === 16 ? 6.75 : 5.75;
  const ring = (
    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} fill="none" aria-hidden>
      <circle
        cx={size / 2}
        cy={size / 2}
        r={radius}
        stroke="var(--color-border-button-default)"
        strokeWidth={size === 16 ? 2.5 : 1.5}
      />
      <circle
        data-progress-ring={size === 16 || undefined}
        data-step-progress={size === 14 || undefined}
        cx={size / 2}
        cy={size / 2}
        r={radius}
        pathLength="1"
        stroke="var(--color-agent-progress-ring)"
        strokeWidth={size === 16 ? 2.5 : 1.5}
        strokeLinecap="round"
        strokeDasharray={`${Math.max(0, Math.min(progress, 1))} 1`}
        transform={`rotate(-90 ${size / 2} ${size / 2})`}
      />
    </svg>
  );
  if (size === 16) return ring;
  return (
    <span aria-hidden className="relative z-10 flex size-3.5 shrink-0 items-center justify-center">
      <AnimatePresence initial={false}>
        {done ? (
          <motion.span
            key="done"
            initial={reduce ? false : { opacity: 0 }}
            animate={{ opacity: 1 }}
            transition={{ duration: reduce ? 0 : 0.34, ease: EASE }}
            className="absolute inset-[-0.5px]"
          >
            <svg viewBox="0 0 14 14" className="size-[15px]" aria-hidden>
              <circle cx="7" cy="7" r="7" fill="var(--color-background-quaternary-default)" />
              <path
                d="M4 7.5 5.646 9.146a.5.5 0 0 0 .708 0L10 5.5"
                fill="none"
                stroke="var(--color-foreground-icon-secondary)"
                strokeWidth="1.5"
                strokeLinecap="round"
              />
            </svg>
          </motion.span>
        ) : waiting ? (
          <motion.span
            key="waiting"
            initial={reduce ? false : { opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            transition={{ duration: reduce ? 0 : 0.3, ease: [0.25, 0.1, 0.35, 1] }}
            className="absolute inset-[-0.5px]"
          >
            <svg viewBox="0 0 15 15" className="size-[15px]" aria-hidden>
              <circle
                cx="7.5"
                cy="7.5"
                r="7"
                fill="none"
                stroke="var(--color-background-quaternary-default)"
                strokeDasharray="2 2"
              />
            </svg>
          </motion.span>
        ) : (
          <motion.span
            key="active"
            initial={reduce ? false : { opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            transition={{ duration: reduce ? 0 : 0.25, ease: "easeOut" }}
            className="absolute inset-0"
          >
            {ring}
          </motion.span>
        )}
      </AnimatePresence>
    </span>
  );
}

/** The timer belongs to the card, so minimizing never resets or pauses work. */
export function AgentProgress({
  steps = AGENT_PROGRESS_STEPS,
  startDelay = 1100,
  stepDuration = 2400,
  completionDelay = 1000,
  onFinished,
  run = true,
  defaultMinimized = false,
  className,
}: AgentProgressProps = {}) {
  const [minimized, setMinimized] = useState(defaultMinimized);
  const [interacted, setInteracted] = useState(false);
  const [elapsed, setElapsed] = useState(0);
  const elapsedRef = useRef(0);
  const onFinishedRef = useRef(onFinished);
  const finishedRef = useRef(false);
  const reduce = useHydratedReducedMotion() === true;
  const panelId = useId();
  const compactRef = useRef<HTMLButtonElement>(null);
  const expandedRef = useRef<HTMLButtonElement>(null);
  const restoreFocus = useRef(false);
  const delayElapsedRef = useRef(0);
  const duration = Math.max(1, stepDuration);
  const totalDuration = steps.length * duration;
  const completed = Math.min(steps.length, Math.floor(elapsed / duration));
  const finished = completed >= steps.length;
  const left = steps.length - completed;
  useEffect(() => {
    onFinishedRef.current = onFinished;
  }, [onFinished]);

  useEffect(() => {
    if (!run || elapsedRef.current >= totalDuration) return;
    let previous = Date.now();
    const id = window.setInterval(() => {
      const now = Date.now();
      const delta = now - previous;
      const delayRemaining = Math.max(0, startDelay - delayElapsedRef.current);
      delayElapsedRef.current += Math.min(delta, delayRemaining);
      elapsedRef.current = Math.min(
        totalDuration,
        elapsedRef.current + Math.max(0, delta - delayRemaining),
      );
      previous = now;
      setElapsed(elapsedRef.current);
      if (elapsedRef.current >= totalDuration) window.clearInterval(id);
    }, 25);
    return () => window.clearInterval(id);
  }, [run, totalDuration, startDelay]);

  useLayoutEffect(() => {
    if (!restoreFocus.current) return;
    const button = minimized ? compactRef.current : expandedRef.current;
    if (button) {
      button.focus({ preventScroll: true });
      restoreFocus.current = false;
    }
  }, [minimized]);
  const toggle = () => {
    restoreFocus.current = true;
    setInteracted(true);
    setMinimized((value) => !value);
  };
  const statusText = finished
    ? "All steps completed"
    : `${left} ${left === 1 ? "step" : "steps"} left`;
  const status = (
    <span
      aria-hidden
      className="relative inline-flex min-w-0 overflow-hidden text-body-medium text-text-secondary"
    >
      <AnimatePresence mode="popLayout" initial={false}>
        <motion.span
          key={statusText}
          initial={reduce ? false : { opacity: 0, filter: "blur(3px)" }}
          animate={{ opacity: 1, filter: "blur(0px)" }}
          exit={{ opacity: 0, filter: "blur(3px)" }}
          transition={{ duration: reduce ? 0 : 0.26, ease: EASE }}
          className="block whitespace-nowrap"
        >
          {statusText}
        </motion.span>
      </AnimatePresence>
    </span>
  );

  useEffect(() => {
    if (!finished || finishedRef.current) return;
    const id = window.setTimeout(
      () => {
        finishedRef.current = true;
        onFinishedRef.current?.();
      },
      Math.max(0, completionDelay),
    );
    return () => window.clearTimeout(id);
  }, [finished, completionDelay]);

  const expandedHeight = 51 + Math.max(0, steps.length * 38 - 6);

  return (
    <motion.section
      key={reduce ? "reduced" : "animated"}
      data-testid="agent-progress"
      initial={reduce ? false : { height: 2, opacity: 0, filter: "blur(8px)", y: -12 }}
      animate={{
        height: minimized ? 44 : expandedHeight,
        opacity: 1,
        filter: "blur(0px)",
        y: 0,
      }}
      transition={{
        height: { duration: reduce ? 0 : !interacted ? 0.67 : minimized ? 0.42 : 0.24, ease: EASE },
        opacity: { duration: reduce ? 0 : 0.35, ease: EASE },
        filter: { duration: reduce ? 0 : 0.4, ease: EASE },
        y: { duration: reduce ? 0 : 0.52, ease: EASE },
      }}
      className={cx(
        "relative w-[341px] max-w-full overflow-hidden rounded-2xl border border-border-button-default bg-background-primary-default shadow-xs dark:border-background-primary-default",
        className,
      )}
    >
      <span role="status" className="sr-only" aria-live="polite" aria-atomic="true">
        {statusText}
      </span>
      <AnimatePresence initial={false}>
        {!finished && (
          <motion.span
            key="overall-ring"
            initial={false}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            transition={{ duration: reduce ? 0 : 0.25, ease: EASE }}
            className="pointer-events-none absolute top-[14px] left-[14px] z-20 flex size-4 items-center justify-center"
          >
            <ProgressCircle progress={totalDuration ? elapsed / totalDuration : 0} size={16} />
          </motion.span>
        )}
      </AnimatePresence>
      <AnimatePresence>
        {minimized && (
          <PresenceContents key="compact">
            {(present) => (
              <motion.button
                type="button"
                ref={compactRef}
                data-testid="agent-progress-minimized"
                aria-label="Expand steps"
                aria-expanded={false}
                aria-controls={panelId}
                aria-hidden={!present || undefined}
                inert={!present}
                onClick={toggle}
                initial={reduce ? false : { opacity: 0, filter: "blur(3px)" }}
                animate={{ opacity: 1, filter: "blur(0px)" }}
                exit={{ opacity: 0, filter: "blur(3px)" }}
                transition={{ duration: reduce ? 0 : 0.2, delay: reduce ? 0 : 0.12, ease: EASE }}
                className={cx(
                  "group absolute inset-0 flex cursor-pointer items-center gap-2 px-2.5 py-1.5 text-left outline-none focus-visible:ring-2 focus-visible:ring-inset focus-visible:ring-border-focus-ring",
                  !present && "pointer-events-none",
                )}
              >
                <span className="flex min-w-0 shrink-0 items-center pl-1">
                  <motion.span
                    aria-hidden
                    initial={false}
                    animate={{ width: finished ? 0 : 24, opacity: finished ? 0 : 1 }}
                    transition={{ duration: reduce ? 0 : 0.4, ease: EASE }}
                    className="h-4 shrink-0 overflow-hidden"
                  />
                  {status}
                </span>
                {!finished && (
                  <span className="flex min-w-0 flex-1 items-center gap-2 overflow-hidden py-1.5 pr-6 [mask-image:linear-gradient(to_right,#000_0%,#000_100%)] transition-[mask-image] duration-300 group-hover:[mask-image:linear-gradient(to_right,#000_0%,#000_68%,transparent_94%)]">
                    <svg aria-hidden viewBox="0 0 14 14" className="size-3.5 shrink-0">
                      <path
                        d="M7.47 2.47a.75.75 0 0 1 1.06 0l4.177 4.176a.5.5 0 0 1 0 .708L8.53 11.53a.75.75 0 0 1-1.06-1.06l2.72-2.72H2a.75.75 0 0 1 0-1.5h8.19L7.47 3.53a.75.75 0 0 1 0-1.06Z"
                        fill="var(--color-foreground-icon-secondary)"
                      />
                    </svg>
                    <span className="min-w-0 flex-1 overflow-hidden text-body-medium">
                      <span
                        className={cx("block truncate", !reduce && "agent-progress-loading-text")}
                      >
                        {steps[completed]}
                      </span>
                    </span>
                  </span>
                )}
                <span className="absolute top-1/2 right-2.5 size-5 -translate-y-1/2 opacity-0 transition-opacity duration-200 group-hover:opacity-100 group-focus-visible:opacity-100">
                  <svg aria-hidden viewBox="0 0 20 20" className="size-5">
                    <rect
                      x="1"
                      y="1"
                      width="18"
                      height="18"
                      rx="4"
                      fill="var(--color-background-quaternary-default)"
                    />
                    <path
                      d="M7.553 6.109a.75.75 0 0 1 .75-.75h5.907a.5.5 0 0 1 .5.5v5.906a.75.75 0 0 1-1.5 0V7.919l-5.79 5.791a.75.75 0 1 1-1.061-1.06l5.79-5.791H8.303a.75.75 0 0 1-.75-.75Z"
                      fill="var(--color-foreground-icon-secondary)"
                    />
                  </svg>
                </span>
              </motion.button>
            )}
          </PresenceContents>
        )}
      </AnimatePresence>
      <div
        id={panelId}
        aria-hidden={minimized || undefined}
        inert={minimized}
        className={minimized ? "pointer-events-none" : undefined}
      >
        <AnimatePresence>
          {!minimized && (
            <motion.div
              key="expanded"
              data-testid="agent-progress-expanded"
              initial={reduce ? false : { opacity: 0, filter: "blur(3px)" }}
              animate={{ opacity: 1, filter: "blur(0px)" }}
              exit={{
                opacity: 0,
                filter: "blur(3px)",
                transition: { duration: reduce ? 0 : 0.22, ease: EASE },
              }}
              transition={{ duration: reduce ? 0 : interacted ? 0.16 : 0.22, ease: EASE }}
              style={{ height: expandedHeight }}
              className="absolute inset-x-0 top-0 px-2.5 pt-3 pb-2.5"
            >
              <div className="flex h-5 items-center pl-1">
                <motion.span
                  aria-hidden
                  initial={false}
                  animate={{ width: finished ? 0 : 24, opacity: finished ? 0 : 1 }}
                  transition={{ duration: reduce ? 0 : 0.4, ease: EASE }}
                  className="h-4 shrink-0 overflow-hidden"
                />
                <span className="flex min-w-0 flex-1">{status}</span>
                <button
                  type="button"
                  ref={expandedRef}
                  aria-label="Minimize steps"
                  aria-expanded
                  aria-controls={panelId}
                  onClick={toggle}
                  className="relative size-5 cursor-pointer rounded-sm outline-none transition-opacity duration-200 hover:opacity-80 focus-visible:ring-2 focus-visible:ring-border-focus-ring"
                >
                  <span className="absolute inset-px rounded-sm bg-background-quaternary-default" />
                  <span className="absolute top-[13px] left-1 h-0.5 w-3 rounded-[3px] bg-foreground-icon-secondary" />
                </button>
              </div>
              <ol className="mt-[9px] flex flex-col gap-1.5">
                {steps.map((step, index) => {
                  const done = index < completed;
                  const active = index === completed && !finished;
                  return (
                    <motion.li
                      key={`${index}-${step}-${reduce}`}
                      initial={reduce ? false : { opacity: 0, y: -4, filter: "blur(6px)" }}
                      animate={{ opacity: 1, y: 0, filter: "blur(0px)" }}
                      transition={{
                        duration: reduce ? 0 : interacted ? 0.24 : 0.45,
                        delay: reduce ? 0 : interacted ? index * 0.035 : 0.65 + index * 0.16,
                        ease: EASE,
                      }}
                      className="h-8"
                    >
                      <motion.div
                        animate={{ paddingLeft: active ? 9 : 4, paddingRight: active ? 13 : 4 }}
                        transition={{ duration: reduce ? 0 : 0.3, ease: EASE }}
                        className="relative flex h-full items-center gap-2 rounded-full"
                      >
                        {active && (

                            aria-hidden
                            layoutId={reduce ? undefined : `${panelId}-active-step`}

                              reduce

                        <ProgressCircle
