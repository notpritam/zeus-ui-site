Installation

Add calendar to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.1.0.tgz zeus-ui add calendar

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — app-shell, avatar, button, cx, date-picker, dropdown, icon-button, input, notification-center, segmented-control, select, tokens, use-dismiss-on-outside-press. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

calendar-demo.tsx
"use client";
import { Calendar } from "@/components/application/calendar/calendar";
export default function CalendarDemo() {
  return <Calendar />;
}

const isoDate = (date: Date) =>
  `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}-${String(date.getDate()).padStart(2, "0")}`;
function parseDate(value: string) {
  const [year = 2026, month = 8, day = 1] = value.split("-").map(Number);
  return new Date(year, month - 1, day, 12);
}

function EventDetails({
  event,
  onEdit,
}: {
  event: CalendarEvent;
  onEdit?: CalendarProps["onEventEdit"];
}) {
  const date = parseDate(event.date);
  const timeParts = event.start?.match(/^(\d{2}):(\d{2})$/);
  const endMinutes =
    timeParts && event.duration
      ? (Number(timeParts[1]) * 60 + Number(timeParts[2]) + event.duration) % (24 * 60)
      : null;
  const endTime =
    endMinutes === null
      ? null
      : `${String(Math.floor(endMinutes / 60)).padStart(2, "0")}:${String(endMinutes % 60).padStart(2, "0")}`;
  const timezone = new Intl.DateTimeFormat("en", { timeZoneName: "shortOffset" })
    .formatToParts(date)
    .find((part) => part.type === "timeZoneName")
    ?.value.replace(/^GMT$/, "GMT+0");
  const row =
    "flex h-9 w-full shrink-0 items-center gap-2.5 rounded-2lg bg-background-secondary-default py-2 pr-1.5 pl-2";
  const lead = "flex min-w-0 flex-1 items-center gap-1.5";
  const badge =
    "shrink-0 rounded-sm bg-background-tertiary-default px-1 py-1 text-caption-1-medium text-text-secondary";
  return (
    <div className="flex w-full flex-col gap-2.5">
      <div className="flex w-full flex-col gap-px rounded-2lg bg-background-secondary-default px-2.5 py-2">
        <h3 className="text-headline-medium wrap-anywhere text-text-primary">{event.title}</h3>
        <p className="text-body-medium text-text-secondary">
          {date.toLocaleDateString("en-US", { weekday: "short", month: "short", day: "numeric" })}
        </p>
      </div>
      {event.meetingUrl && (
        <div className={row}>
          <div className={lead}>
            <img
              src="/brand/google_meet.png"
              alt=""
              width={20}
              height={20}
              className="size-5 shrink-0"
            />
            <span className="text-body-2-medium whitespace-nowrap text-text-primary">
              Google Meet
            </span>
          </div>
          <div className="flex shrink-0 items-center gap-1.5">
            <span className={badge}>{event.meetingUrl.split("/").pop()}</span>
            <ButtonLink
              href={event.meetingUrl}
              target="_blank"
              rel="noreferrer"
              size="xs"
              variant="primary"
            >
              Join
            </ButtonLink>
          </div>
        </div>
      )}
      <div className={row}>
        <div className={lead}>
          <RiTimeLine className="size-[18px] shrink-0 text-foreground-icon-secondary" />
          <span className="flex items-center gap-1.5 text-body-2-medium whitespace-nowrap text-text-primary">
            <span>{event.start ?? "All day"}</span>
            {endTime && (
              <>
                <RiArrowRightLine className="size-5 shrink-0 text-foreground-icon-secondary" />
                <span>{endTime}</span>
              </>
            )}
          </span>
        </div>
        {event.duration && (
          <span className={badge}>
            {event.duration >= 60 ? `${event.duration / 60}h` : `${event.duration}m`}
          </span>
        )}
      </div>
      <div className={row}>
        <div className={lead}>
          <RiGlobalLine className="size-[18px] shrink-0 text-foreground-icon-secondary" />
          <span className="flex items-center gap-1 text-body-2-medium whitespace-nowrap">
            <span className="text-text-secondary">{timezone}</span>
            {event.location && <span className="text-text-primary">{event.location}</span>}
          </span>
        </div>
        <Button
          size="xs"
          variant="secondary"
          iconOnly
          leadingIcon={RiCornerDownLeftLine}
          aria-label="Edit timezone"
          className="text-foreground-icon-secondary"
          onClick={() => onEdit?.(event, "timezone")}
        />
      </div>
      {event.participants?.length ? (
        <div className="flex w-full flex-col gap-0.5 rounded-2lg bg-background-secondary-default py-2 pr-1.5 pl-2">
          <div className="flex w-full items-center gap-2.5">

              leadingIcon={RiGroupLine}
              aria-label="Edit participants"
              className="text-foreground-icon-secondary"
              onClick={() => onEdit?.(event, "participants")}
            />
          </div>
          <div className="flex w-full flex-col">
            {event.participants.map((person, index) => (
              <div key={person.email} className="flex w-full items-center gap-2 rounded-2lg py-1.5">
                <Avatar
                  size="xs"
                  initials={person.name.charAt(0)}
                  color={index === 1 ? "blue" : "neutral"}
                />
                <span className="truncate text-body-2-medium text-text-primary">
                  {person.email}
                </span>
              </div>
            ))}

function EventPill({
  event,
  onSelect,
  onEdit,
  compact = false,
}: {
  event: CalendarEvent;
  onSelect?: (event: CalendarEvent) => void;
  onEdit?: CalendarProps["onEventEdit"];
  compact?: boolean;
}) {
  const [open, setOpen] = useState(false);
  const reduce = useReducedMotion() === true;
  const buttonRef = useRef<HTMLButtonElement>(null);
  const anchorRef = useRef<HTMLElement>(null);
  const popoverRef = useRef<HTMLElement>(null);
  useDismissOnOutsidePress(open, () => setOpen(false), [buttonRef, popoverRef]);
  return (
    <DialogTrigger isOpen={open} onOpenChange={setOpen}>
      <AriaButton
        ref={buttonRef}
        onPress={() => {
          anchorRef.current =
            buttonRef.current?.closest<HTMLElement>('[role="gridcell"]') ?? buttonRef.current;
          onSelect?.(event);
        }}
        aria-label={`${event.title}${event.start ? ` ${event.start}` : ""}`}
        className={cx(
          "flex w-full min-w-0 cursor-pointer items-center justify-between gap-0.5 rounded-sm px-1 py-0.5 outline-none transition-[filter] duration-150 hover:brightness-95 focus-visible:ring-2 focus-visible:ring-border-focus-ring sm:gap-1 sm:rounded-md sm:px-1.5",
          eventTone[event.color ?? "lime"],
        )}
      >
        <span className="truncate text-[10px] leading-3 sm:text-body-2-medium">{event.title}</span>
        {event.start && (
          <span
            className={cx(
              "hidden shrink-0 opacity-70 sm:inline sm:text-caption-1-medium",
              eventTimeTone[event.color ?? "lime"],
            )}
          >
            {event.start}
          </span>
        )}
      </AriaButton>
      {compact &&
        createPortal(
          <AnimatePresence>
            {open && (
              <motion.div
                aria-hidden="true"
                initial={{ opacity: 0 }}
                animate={{ opacity: 1 }}
                exit={{ opacity: 0 }}
                transition={{ duration: reduce ? 0 : 0.3, ease: "easeOut" }}
                className="pointer-events-none fixed inset-0 z-[99999] bg-black/10 dark:bg-transparent"
              />
            )}
          </AnimatePresence>,
          document.body,
        )}
      <Popover
        ref={popoverRef}
        triggerRef={anchorRef}
        isNonModal
        placement="right top"
        offset={6}
        className="w-[302px] max-w-[calc(100vw-32px)] rounded-[20px] border border-border-button-default bg-background-primary-default p-2.5 outline-none shadow-[0px_1px_2px_0px_rgba(0,0,0,0.04),0px_7px_8px_0px_rgba(0,0,0,0.04)] transition duration-150 ease-out sm:data-[entering]:scale-90 sm:data-[entering]:opacity-0 sm:data-[entering]:blur-[4px] sm:data-[exiting]:scale-90 sm:data-[exiting]:opacity-0 sm:data-[exiting]:blur-[4px] max-sm:fixed! max-sm:inset-x-0! max-sm:top-auto! max-sm:bottom-0! max-sm:m-0! max-sm:max-h-[85dvh]! max-sm:w-full max-sm:max-w-none max-sm:overflow-y-auto max-sm:rounded-[24px] max-sm:rounded-b-none max-sm:border-x-0 max-sm:border-b-0 max-sm:pb-[calc(10px+env(safe-area-inset-bottom))] max-sm:duration-300 max-sm:data-[entering]:translate-y-full max-sm:data-[exiting]:translate-y-full"
      >
        <Dialog aria-label={`${event.title} details`} className="outline-none">
          <EventDetails event={event} onEdit={onEdit} />
        </Dialog>
      </Popover>
    </DialogTrigger>
  );
}

              <p className="pl-2 text-body-medium text-text-secondary">{group.account}</p>
              <div className="flex w-full flex-col gap-1">
                {group.calendars.map((label, index) => {
                  const key = `${group.account}-${label}`;
                  return (
                    <button
                      key={key}
                      type="button"
                      onClick={() => {
                        setOpen(false);
                        onCalendarSelect?.(group.account, label);
                      }}
                      className="flex w-full cursor-pointer items-center gap-2 rounded-2lg px-2 py-1.5 outline-none transition-colors hover:bg-background-primary-hover focus-visible:bg-background-primary-hover"
                    >
                      <span
                        className={cx(
                          "flex size-5 shrink-0 items-center justify-center rounded-md",
                          group.tones[index],
                        )}
                      >
                        <RiRssFill className="size-3" />
                      </span>
                      <span className="truncate text-body-medium text-text-primary">{label}</span>
                    </button>
                  );
                })}
              </div>
            </div>
          </div>
        ))}
        <Button
          size="small"
          variant="secondary"
          leadingIcon={RiAddLine}
          className="w-full"
          onClick={() => {
            setOpen(false);
            onAddAccount?.();
          }}
        >
          Add new account
        </Button>
      </DropdownPopover>
    </Dropdown>
  );
}

function NewEvent({ date, onAdd }: { date: Date; onAdd: (event: CalendarEvent) => void }) {
  const [open, setOpen] = useState(false);
  const [title, setTitle] = useState("");
  const [day, setDay] = useState<CalendarDate | null>(null);
  const [time, setTime] = useState("09:00");
  const [color, setColor] = useState<CalendarEventColor>("blue");
  const id = useId();
  function submit(event: FormEvent) {
    event.preventDefault();
    if (!title.trim()) return;
    onAdd({
      id: `event-${globalThis.crypto.randomUUID()}`,
      title: title.trim(),
      date: day?.toString() ?? isoDate(date),
      start: time || undefined,
      color,
      duration: 60,
    });
    setOpen(false);
    setTitle("");
    setDay(null);
  }
  return (
    <>
      <Button onClick={() => setOpen(true)} leadingIcon={RiAddLine} variant="primary" size="medium">
        New event
      </Button>
      <ModalOverlay
        isOpen={open}
        onOpenChange={setOpen}
        isDismissable
        className="fixed inset-0 z-50 flex items-center justify-center bg-background-full/60 p-4 backdrop-blur-sm"
      >
        <Modal className="w-full max-w-[400px] rounded-3xl border border-border-button-default bg-background-primary-default p-5 shadow-dropdown">
          <Dialog aria-label="New event" className="outline-none">
            <div className="mb-5 flex items-center justify-between">
              <h2 className="text-title-3-semibold text-text-primary">New event</h2>
              <IconButton
                icon={RiCloseLine}
                aria-label="Close new event"
                size="small"
                onClick={() => setOpen(false)}
              />
            </div>
            <form onSubmit={submit} className="flex flex-col gap-4">
              <TextField aria-label="Event name" value={title} onChange={setTitle} isRequired>
                <span className="text-body-2-medium text-text-secondary">Event name</span>
                <InputBase id={id} placeholder="What’s happening?" autoFocus />
              </TextField>
              <div className="flex items-end gap-3">
                <div className="flex flex-1 flex-col gap-1">
                  <span className="text-body-2-medium text-text-secondary">Date</span>
                  <DatePicker
                    aria-label="Event date"
                    value={
                      day ??
                      new CalendarDate(date.getFullYear(), date.getMonth() + 1, date.getDate())
                    }
                    onChange={setDay}
                  />
                </div>
                <TextField
                  aria-label="Start time"
                  value={time}
                  onChange={setTime}
                  className="max-w-[120px]"
                >
                  <span className="text-body-2-medium text-text-secondary">Time</span>
                  <InputBase type="time" />
                </TextField>
              </div>
              <div className="flex flex-col gap-2">
                <span className="text-body-2-medium text-text-secondary">Color</span>
                <div className="flex gap-2">
                  {COLORS.map((tone) => (
                    <button
                      key={tone}
                      type="button"
                      aria-label={`${tone} event color`}
                      aria-pressed={color === tone}
                      onClick={() => setColor(tone)}
                      className={cx(
                        "flex size-7 items-center justify-center rounded-full outline-none focus-visible:ring-2 focus-visible:ring-border-focus-ring",
                        eventTone[tone],
                      )}
                    >
                      {color === tone && <RiCheckLine className="size-4" />}
                    </button>
                  ))}
                </div>
              </div>
              <div className="mt-2 flex justify-end gap-2">
                <Button variant="secondary" onClick={() => setOpen(false)}>
                  Cancel
                </Button>
                <Button type="submit" variant="primary" disabled={!title.trim()}>
                  Create event
                </Button>
              </div>
            </form>
          </Dialog>
        </Modal>
      </ModalOverlay>
    </>
  );
}

const compactCalendarQuery = "(max-width: 639px)";
function subscribeCompactCalendar(onChange: () => void) {
  const query = window.matchMedia(compactCalendarQuery);
  query.addEventListener("change", onChange);
  return () => query.removeEventListener("change", onChange);
}
const getCompactCalendar = () => window.matchMedia(compactCalendarQuery).matches;
const getServerCompactCalendar = () => false;

export function Calendar({
  date: controlledDate,
  onDateChange,
  events: controlledEvents,
  onEventsChange,
  onEventSelect,
  notifications = CALENDAR_NOTIFICATIONS,
  onAddAccount,
  onCalendarSelect,
  onEventEdit,
  className,
}: CalendarProps = {}) {
  const [internalDate, setInternalDate] = useState(() => new Date(2026, 7, 1, 12));
  const [internalEvents, setInternalEvents] = useState(CALENDAR_EVENTS);
  const [pickerOpen, setPickerOpen] = useState(false);
  const monthTrigger = useRef<HTMLButtonElement>(null);
  const compact = useSyncExternalStore(
    subscribeCompactCalendar,
    getCompactCalendar,
    getServerCompactCalendar,
  );
  const date = controlledDate ?? internalDate;
  const events = controlledEvents ?? internalEvents;
  const year = date.getFullYear();
  const month = date.getMonth();
  const label = date.toLocaleDateString("en-US", { month: "long", year: "numeric" });
  const first = new Date(year, month, 1, 12);
  const cells = 42;
  const days = Array.from(
    { length: cells },
    (_, index) => new Date(year, month, index - first.getDay() + 1, 12),
  );
  const eventsByDate = new Map<string, CalendarEvent[]>();
  for (const event of events) {
    const list = eventsByDate.get(event.date) ?? [];
    list.push(event);
    eventsByDate.set(event.date, list);
  }
  function changeDate(next: Date) {
    if (!controlledDate) setInternalDate(next);
    onDateChange?.(next);
  }
  function addEvent(event: CalendarEvent) {
    const next = [...events, event];
    if (!controlledEvents) setInternalEvents(next);
    onEventsChange?.(next);
  }
  return (
    <section
      className={cx("flex w-full min-w-0 flex-col gap-2.5", className)}
      aria-label="Calendar"
    >
      <header className="flex flex-wrap items-end justify-between gap-2 px-3 pt-3 sm:px-0 sm:pt-0">
        <h1 className="whitespace-nowrap px-1 text-title-2-medium text-text-primary">{label}</h1>
        <div className="flex w-full flex-nowrap items-start gap-2.5 sm:w-auto">
          <NotificationBell
            notifications={notifications}
            popoverClassName="w-[430px] max-w-[calc(100vw-32px)] origin-top-right transition duration-150 ease-out data-[entering]:scale-95 data-[entering]:opacity-0 data-[entering]:blur-[2px] data-[exiting]:scale-95 data-[exiting]:opacity-0 data-[exiting]:blur-[2px]"
          />
          <CalendarSubscriptions onAddAccount={onAddAccount} onCalendarSelect={onCalendarSelect} />
          <div className="relative h-9 min-w-0 flex-1 sm:w-[320px] sm:flex-none">
            <div className="absolute inset-x-0 top-0 flex items-center rounded-2lg border border-border-button-default bg-background-primary-default p-2 shadow-dropdown dark:border-transparent dark:shadow-none">
              <button
                type="button"
                aria-label="Previous month"
                onClick={() => changeDate(new Date(year, month - 1, 1, 12))}
                className="flex size-4 shrink-0 items-center justify-center rounded-[3px] outline-none hover:bg-background-secondary-hover focus-visible:ring-2 focus-visible:ring-border-focus-ring"
              >
                <MonthChevron direction="previous" />
              </button>
              <button
                ref={monthTrigger}
                type="button"
                onClick={() => setPickerOpen(!pickerOpen)}
                aria-label={`Jump to date, ${label}`}
                aria-expanded={pickerOpen}
                className="min-w-0 flex-1 truncate text-center text-body-medium text-text-primary outline-none focus-visible:ring-2 focus-visible:ring-border-focus-ring"
              >
                <span className="sm:hidden">
                  {date.toLocaleDateString("en-US", { month: "short" })}
                </span>
                <span className="hidden sm:inline">{label}</span>
              </button>
              <button
                type="button"
                aria-label="Next month"
                onClick={() => changeDate(new Date(year, month + 1, 1, 12))}
                className="flex size-4 shrink-0 items-center justify-center rounded-[3px] outline-none hover:bg-background-secondary-hover focus-visible:ring-2 focus-visible:ring-border-focus-ring"
              >
                <MonthChevron direction="next" />
              </button>
            </div>
            <DatePicker
              aria-label="Jump to date"
              triggerRef={monthTrigger}
              isOpen={pickerOpen}
              onOpenChange={setPickerOpen}
              value={new CalendarDate(year, month + 1, date.getDate())}
              onChange={(value) => {
                if (value) changeDate(new Date(value.year, value.month - 1, value.day, 12));
              }}
            />
          </div>
          <NewEvent date={date} onAdd={addEvent} />
        </div>
      </header>
      <div className="w-full overflow-hidden rounded-b-3xl bg-background-secondary-default sm:overflow-visible sm:rounded-3xl sm:p-3">
        <div className="flex h-full min-h-0 w-full flex-col overflow-hidden border-y border-separator-border-strong dark:border-separator-border sm:h-auto sm:gap-2 sm:overflow-visible sm:border-0">
          <div className="grid grid-cols-7 border-b border-separator-border-strong dark:border-separator-border sm:gap-2 sm:border-0">
            {DAY_NAMES.map((day) => (
              <span
                key={day}
                className="flex items-center justify-center border-r border-separator-border-strong bg-background-secondary-default px-0.5 py-1 text-center text-[11px] leading-4 text-text-secondary last:border-r-0 dark:border-separator-border sm:rounded-xl sm:border-0 sm:px-2.5 sm:py-[5px] sm:text-body-2-regular"
              >
                {day}
              </span>
            ))}
          </div>
          <div
            role="grid"
            aria-label={label}
            className="grid min-h-0 flex-1 grid-cols-7 grid-rows-[repeat(6,minmax(0,1fr))] sm:flex-none sm:grid-rows-[repeat(6,minmax(0,auto))] sm:gap-2"
          >
            {days.map((day) => {
              const key = isoDate(day);
              const dayEvents = eventsByDate.get(key) ?? [];
              const inMonth = day.getMonth() === month;
              return (
                <div
                  key={key}
                  role="gridcell"
                  aria-label={day.toLocaleDateString("en-US", { month: "long", day: "numeric" })}
                  className="relative h-full min-w-0 border-r border-b border-separator-border-strong last:border-b-0 nth-[7n]:border-r-0 nth-[n+36]:border-b-0 dark:border-separator-border sm:border-0"
                >
                  <div
                    className={cx(
                      "relative flex h-full min-h-[72px] min-w-0 flex-col overflow-hidden sm:min-h-[94px] sm:rounded-xl lg:min-h-[105px] xl:min-h-[128px] 2xl:min-h-[164px]",
                      inMonth
                        ? "bg-background-primary-default dark:bg-background-secondary-default sm:shadow-card sm:dark:bg-background-primary-default"
                        : "bg-background-secondary-default sm:bg-background-tertiary-default",
                    )}
                  >
                    <button
                      type="button"
                      aria-label={`Select ${key}`}
                      onClick={() => changeDate(day)}
                      className={cx(
                        "w-full pt-1.5 pl-1.5 text-left text-[11px] leading-4 font-medium outline-none focus-visible:ring-2 focus-visible:ring-inset focus-visible:ring-border-focus-ring sm:pt-2 sm:pl-2.5 sm:text-body-2-medium",
                        inMonth ? "text-text-primary" : "text-text-secondary",
                      )}
                    >
                      {day.getDate()}
                    </button>
                    <div className="mt-auto flex min-w-0 flex-col gap-0.5 px-1 pb-1 sm:gap-[5px] sm:px-2 sm:pb-2">
                      {dayEvents.length > 4 && (
                        <Dropdown>
                          <DropdownTrigger className="w-full truncate rounded-sm text-left text-[10px] leading-3 font-medium text-text-secondary sm:text-body-2-medium">
                            +{dayEvents.length - 4} more
                          </DropdownTrigger>
                          <DropdownPopover aria-label={`Events on ${key}`} className="w-[280px]">
                            <DropdownGroup
                              label={day.toLocaleDateString("en-US", {
                                month: "long",
                                day: "numeric",
                              })}
                            >
                              {dayEvents.map((event) => (
                                <EventPill
                                  key={event.id}
                                  event={event}
                                  onSelect={onEventSelect}
                                  onEdit={onEventEdit}
                                  compact={compact}
                                />
                              ))}
                            </DropdownGroup>
                          </DropdownPopover>
                        </Dropdown>
                      )}
                      {dayEvents.slice(0, 4).map((event) => (
                        <EventPill
                          key={event.id}
                          event={event}
                          onSelect={onEventSelect}
                          onEdit={onEventEdit}
                          compact={compact}
                        />
                      ))}
                    </div>
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      </div>
    </section>
  );
}
