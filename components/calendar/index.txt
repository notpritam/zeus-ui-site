const isoDate = (date: Date) =>
  `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}-${String(date.getDate()).padStart(2, "0")}`;
function parseDate(value: string) {
  const [year = 2026, month = 8, day = 1] = value.split("-").map(Number);
  return new Date(year, month - 1, day, 12);
}

function EventDetails({
  event,
  onEdit,
}: {
  event: CalendarEvent;
  onEdit?: CalendarProps["onEventEdit"];
}) {
  const date = parseDate(event.date);
  const timeParts = event.start?.match(/^(\d{2}):(\d{2})$/);
  const endMinutes =
    timeParts && event.duration
      ? (Number(timeParts[1]) * 60 + Number(timeParts[2]) + event.duration) % (24 * 60)
      : null;
  const endTime =
    endMinutes === null
      ? null
      : `${String(Math.floor(endMinutes / 60)).padStart(2, "0")}:${String(endMinutes % 60).padStart(2, "0")}`;
  const timezone = new Intl.DateTimeFormat("en", { timeZoneName: "shortOffset" })
    .formatToParts(date)
    .find((part) => part.type === "timeZoneName")
    ?.value.replace(/^GMT$/, "GMT+0");
  const row =
    "flex h-9 w-full shrink-0 items-center gap-2.5 rounded-2lg bg-background-secondary-default py-2 pr-1.5 pl-2";
  const lead = "flex min-w-0 flex-1 items-center gap-1.5";
  const badge =
    "shrink-0 rounded-sm bg-background-tertiary-default px-1 py-1 text-caption-1-medium text-text-secondary";
  return (
    <div className="flex w-full flex-col gap-2.5">
      <div className="flex w-full flex-col gap-px rounded-2lg bg-background-secondary-default px-2.5 py-2">
        <h3 className="text-headline-medium wrap-anywhere text-text-primary">{event.title}</h3>
        <p className="text-body-medium text-text-secondary">
          {date.toLocaleDateString("en-US", { weekday: "short", month: "short", day: "numeric" })}
        </p>
      </div>
      {event.meetingUrl && (
        <div className={row}>
          <div className={lead}>
            <img
              src="/brand/google_meet.png"
              alt=""
              width={20}
              height={20}
              className="size-5 shrink-0"
            />
            <span className="text-body-2-medium whitespace-nowrap text-text-primary">
              Google Meet
            </span>
          </div>
          <div className="flex shrink-0 items-center gap-1.5">
            <span className={badge}>{event.meetingUrl.split("/").pop()}</span>
            <ButtonLink
              href={event.meetingUrl}
              target="_blank"
              rel="noreferrer"
              size="xs"
              variant="primary"
            >
              Join
            </ButtonLink>
          </div>
        </div>
      )}
      <div className={row}>
        <div className={lead}>
          <RiTimeLine className="size-[18px] shrink-0 text-foreground-icon-secondary" />
          <span className="flex items-center gap-1.5 text-body-2-medium whitespace-nowrap text-text-primary">
            <span>{event.start ?? "All day"}</span>
            {endTime && (
              <>
                <RiArrowRightLine className="size-5 shrink-0 text-foreground-icon-secondary" />
                <span>{endTime}</span>
              </>
            )}
          </span>
        </div>
        {event.duration && (
          <span className={badge}>
            {event.duration >= 60 ? `${event.duration / 60}h` : `${event.duration}m`}
          </span>
        )}
      </div>
      <div className={row}>
        <div className={lead}>
          <RiGlobalLine className="size-[18px] shrink-0 text-foreground-icon-secondary" />
          <span className="flex items-center gap-1 text-body-2-medium whitespace-nowrap">
            <span className="text-text-secondary">{timezone}</span>
            {event.location && <span className="text-text-primary">{event.location}</span>}
          </span>
        </div>
        <Button
          size="xs"
          variant="secondary"
          iconOnly
          leadingIcon={RiCornerDownLeftLine}
          aria-label="Edit timezone"
          className="text-foreground-icon-secondary"
          onClick={() => onEdit?.(event, "timezone")}
        />
      </div>
      {event.participants?.length ? (
        <div className="flex w-full flex-col gap-0.5 rounded-2lg bg-background-secondary-default py-2 pr-1.5 pl-2">
          <div className="flex w-full items-center gap-2.5">
            <div className={lead}>
              <RiGlobalLine className="size-[18px] shrink-0 text-foreground-icon-secondary" />
              <span className="text-body-2-medium whitespace-nowrap text-text-secondary">
                Participants

            </div>
            <Button
              size="xs"
              variant="secondary"
              iconOnly
              leadingIcon={RiGroupLine}
              aria-label="Edit participants"
              className="text-foreground-icon-secondary"
              onClick={() => onEdit?.(event, "participants")}
            />
          </div>
          <div className="flex w-full flex-col">
            {event.participants.map((person, index) => (
              <div key={person.email} className="flex w-full items-center gap-2 rounded-2lg py-1.5">
                <Avatar
                  size="xs"
                  initials={person.name.charAt(0)}
                  color={index === 1 ? "blue" : "neutral"}
                />

    </div>
  );
}

function EventPill({
  event,
  onSelect,
  onEdit,
  compact = false,
}: {
  event: CalendarEvent;
  onSelect?: (event: CalendarEvent) => void;
  onEdit?: CalendarProps["onEventEdit"];
  compact?: boolean;
}) {
  const [open, setOpen] = useState(false);
  const reduce = useReducedMotion() === true;
  const buttonRef = useRef<HTMLButtonElement>(null);
  const anchorRef = useRef<HTMLElement>(null);
  const popoverRef = useRef<HTMLElement>(null);
  useDismissOnOutsidePress(open, () => setOpen(false), [buttonRef, popoverRef]);
  return (
    <DialogTrigger isOpen={open} onOpenChange={setOpen}>
      <AriaButton
        ref={buttonRef}
        onPress={() => {
          anchorRef.current =
            buttonRef.current?.closest<HTMLElement>('[role="gridcell"]') ?? buttonRef.current;
          onSelect?.(event);
        }}
        aria-label={`${event.title}${event.start ? ` ${event.start}` : ""}`}
        className={cx(
          "flex w-full min-w-0 cursor-pointer items-center justify-between gap-0.5 rounded-sm px-1 py-0.5 outline-none transition-[filter] duration-150 hover:brightness-95 focus-visible:ring-2 focus-visible:ring-border-focus-ring sm:gap-1 sm:rounded-md sm:px-1.5",
          eventTone[event.color ?? "lime"],
        )}
      >
        <span className="truncate text-[10px] leading-3 sm:text-body-2-medium">{event.title}</span>
        {event.start && (
          <span
            className={cx(
              "hidden shrink-0 opacity-70 sm:inline sm:text-caption-1-medium",
              eventTimeTone[event.color ?? "lime"],
            )}
          >
            {event.start}
          </span>
        )}
      </AriaButton>
      {compact &&
        createPortal(
          <AnimatePresence>
            {open && (
              <motion.div
                aria-hidden="true"
                initial={{ opacity: 0 }}
                animate={{ opacity: 1 }}
                exit={{ opacity: 0 }}
                transition={{ duration: reduce ? 0 : 0.3, ease: "easeOut" }}
                className="pointer-events-none fixed inset-0 z-[99999] bg-black/10 dark:bg-transparent"
              />
            )}
          </AnimatePresence>,
          document.body,
        )}
      <Popover
        ref={popoverRef}
        triggerRef={anchorRef}
        isNonModal
        placement="right top"
        offset={6}
        className="w-[302px] max-w-[calc(100vw-32px)] rounded-[20px] border border-border-button-default bg-background-primary-default p-2.5 outline-none shadow-[0px_1px_2px_0px_rgba(0,0,0,0.04),0px_7px_8px_0px_rgba(0,0,0,0.04)] transition duration-150 ease-out sm:data-[entering]:scale-90 sm:data-[entering]:opacity-0 sm:data-[entering]:blur-[4px] sm:data-[exiting]:scale-90 sm:data-[exiting]:opacity-0 sm:data-[exiting]:blur-[4px] max-sm:fixed! max-sm:inset-x-0! max-sm:top-auto! max-sm:bottom-0! max-sm:m-0! max-sm:max-h-[85dvh]! max-sm:w-full max-sm:max-w-none max-sm:overflow-y-auto max-sm:rounded-[24px] max-sm:rounded-b-none max-sm:border-x-0 max-sm:border-b-0 max-sm:pb-[calc(10px+env(safe-area-inset-bottom))] max-sm:duration-300 max-sm:data-[entering]:translate-y-full max-sm:data-[exiting]:translate-y-full"
      >
        <Dialog aria-label={`${event.title} details`} className="outline-none">
          <EventDetails event={event} onEdit={onEdit} />
        </Dialog>

function NewEvent({ date, onAdd }: { date: Date; onAdd: (event: CalendarEvent) => void }) {
  const [open, setOpen] = useState(false);
  const [title, setTitle] = useState("");
  const [day, setDay] = useState<CalendarDate | null>(null);
  const [time, setTime] = useState("09:00");
  const [color, setColor] = useState<CalendarEventColor>("blue");
  const id = useId();
  function submit(event: FormEvent) {
    event.preventDefault();
    if (!title.trim()) return;
    onAdd({
      id: `event-${globalThis.crypto.randomUUID()}`,
      title: title.trim(),
      date: day?.toString() ?? isoDate(date),
      start: time || undefined,
      color,
      duration: 60,
    });
    setOpen(false);
    setTitle("");
    setDay(null);
  }
  return (
    <>
      <Button onClick={() => setOpen(true)} leadingIcon={RiAddLine} variant="primary" size="medium">
        New event
      </Button>
      <ModalOverlay
        isOpen={open}
        onOpenChange={setOpen}
        isDismissable
        className="fixed inset-0 z-50 flex items-center justify-center bg-background-full/60 p-4 backdrop-blur-sm"
      >
        <Modal className="w-full max-w-[400px] rounded-3xl border border-border-button-default bg-background-primary-default p-5 shadow-dropdown">
          <Dialog aria-label="New event" className="outline-none">
            <div className="mb-5 flex items-center justify-between">
              <h2 className="text-title-3-semibold text-text-primary">New event</h2>
              <IconButton
                icon={RiCloseLine}
                aria-label="Close new event"
                size="small"
                onClick={() => setOpen(false)}
              />
            </div>
            <form onSubmit={submit} className="flex flex-col gap-4">
              <TextField aria-label="Event name" value={title} onChange={setTitle} isRequired>
                <span className="text-body-2-medium text-text-secondary">Event name</span>
                <InputBase id={id} placeholder="What’s happening?" autoFocus />
              </TextField>
              <div className="flex items-end gap-3">
                <div className="flex flex-1 flex-col gap-1">
                  <span className="text-body-2-medium text-text-secondary">Date</span>
                  <DatePicker
                    aria-label="Event date"
                    value={
                      day ??
                      new CalendarDate(date.getFullYear(), date.getMonth() + 1, date.getDate())
                    }
                    onChange={setDay}
                  />
                </div>
                <TextField
                  aria-label="Start time"
                  value={time}
                  onChange={setTime}
                  className="max-w-[120px]"
                >
                  <span className="text-body-2-medium text-text-secondary">Time</span>
                  <InputBase type="time" />
                </TextField>
              </div>
              <div className="flex flex-col gap-2">
                <span className="text-body-2-medium text-text-secondary">Color</span>
                <div className="flex gap-2">
                  {COLORS.map((tone) => (
                    <button
                      key={tone}
                      type="button"
                      aria-label={`${tone} event color`}
                      aria-pressed={color === tone}
                      onClick={() => setColor(tone)}
                      className={cx(
                        "flex size-7 items-center justify-center rounded-full outline-none focus-visible:ring-2 focus-visible:ring-border-focus-ring",
                        eventTone[tone],
                      )}
                    >
                      {color === tone && <RiCheckLine className="size-4" />}
                    </button>
                  ))}
                </div>
              </div>
              <div className="mt-2 flex justify-end gap-2">
                <Button variant="secondary" onClick={() => setOpen(false)}>
                  Cancel
                </Button>
                <Button type="submit" variant="primary" disabled={!title.trim()}>
                  Create event
                </Button>
              </div>
            </form>
          </Dialog>
        </Modal>
      </ModalOverlay>
    </>
  );
}
