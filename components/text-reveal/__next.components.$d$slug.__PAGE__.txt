import { useRef, type ReactNode } from "react";
import { gsap } from "gsap";
import { useGSAP } from "@gsap/react";
import { ScrollTrigger } from "gsap/dist/ScrollTrigger";
import { SplitText } from "gsap/dist/SplitText";

gsap.registerPlugin(useGSAP, ScrollTrigger, SplitText);

export interface TextRevealProps {
  children: ReactNode;
  className?: string;
  type?: "lines" | "words" | "chars";
  start?: string;
  end?: string;
  scrub?: boolean | number;
  duration?: number;
  stagger?: number;
  ease?: string;
  yPercent?: number;
  markers?: boolean;
  onComplete?: () => void;
}

export function TextReveal({
  children,
  className,
  type = "words",
  start = "top 80%",
  end,
  scrub = false,
  duration = 0.8,
  stagger = 0.08,
  ease = "expo.out",
  yPercent = 110,
  markers = false,
  onComplete,
}: TextRevealProps) {
  const root = useRef<HTMLDivElement>(null);
  useGSAP(
    (_, contextSafe) => {
      const el = root.current;
      if (!el) return;
      let disposed = false;
      let split: SplitText | undefined;
      let animation: gsap.core.Tween | undefined;
      const mq = gsap.matchMedia();
      const init = contextSafe!(() => {
        if (disposed) return;
        mq.add("(prefers-reduced-motion: no-preference)", () => {
          split = SplitText.create(el, {
            type: type === "lines" ? "lines" : `lines,${type}`,
            mask: "lines",
            autoSplit: true,
            onSplit(instance) {
              animation = gsap.from(instance[type], {
                yPercent,
                duration,
                stagger,
                ease,
                scrollTrigger: { trigger: el, start, end, scrub, once: !scrub, markers },
                onComplete,
              });
              return animation;
            },
          });
          return () => {
            animation?.scrollTrigger?.kill();
            animation?.kill();
            split?.revert();
          };
        });
      });
      (document.fonts?.ready ?? Promise.resolve()).then(init);
      return () => {
        disposed = true;
        mq.revert();
      };
    },
    {
      scope: root,
      dependencies: [
        children,
        type,
        start,
        end,
        scrub,
        duration,
        stagger,
        ease,
        yPercent,
        markers,
        onComplete,
      ],
      revertOnUpdate: true,
    },
  );
  return (
    <div ref={root} className={className} data-zeus-text-reveal>
      {children}
    </div>
  );
}
