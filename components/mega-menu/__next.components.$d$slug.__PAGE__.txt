import {
  type CSSProperties,
  type ReactNode,
  useEffect,
  useId,
  useLayoutEffect,
  useRef,
  useState,
} from "react";
import { MenuInternalStyles } from "./menu-shared";
export interface MegaMenuLink {
  id?: string;
  label: string;
  href: string;
}
export interface MegaMenuGroup {
  id: string;
  label: string;
  columns: readonly { heading: string; links: readonly MegaMenuLink[] }[];
  feature?: { image: string; alt?: string; label?: string; title: string; href: string };
}
export interface MegaMenuProps {
  groups: readonly MegaMenuGroup[];
  brand?: ReactNode;
  brandHref?: string;
  actions?: readonly MegaMenuLink[];
  activeGroup?: string | null;
  defaultActiveGroup?: string | null;
  onActiveGroupChange?: (id: string | null) => void;
  onNavigate?: (link: MegaMenuLink) => void;
  duration?: number;
  stagger?: number;
  openDelay?: number;
  closeDelay?: number;
  background?: string;
  color?: string;
  backdrop?: boolean;
  className?: string;
  style?: CSSProperties;
}
export function MegaMenu({
  groups,
  brand = "Studio",
  brandHref = "#",
  actions = [],
  activeGroup,
  defaultActiveGroup = null,
  onActiveGroupChange,
  onNavigate,
  duration = 0.45,
  stagger = 0.035,
  openDelay = 100,
  closeDelay = 180,
  background = "#fff",
  color = "#161616",
  backdrop = true,
  className,
  style,
}: MegaMenuProps) {
  const [local, setLocal] = useState<string | null>(defaultActiveGroup),
    [mobileOpen, setMobileOpen] = useState(false),
    [mobile, setMobile] = useState(false),
    [height, setHeight] = useState(0);
  const active = activeGroup === undefined ? local : activeGroup;
  const current = groups.find((group) => group.id === active);
  const root = useRef<HTMLElement>(null),
    panel = useRef<HTMLDivElement>(null),
    trigger = useRef<HTMLButtonElement | null>(null),
    toggle = useRef<HTMLButtonElement>(null),
    timer = useRef<ReturnType<typeof setTimeout> | undefined>(undefined),
    focusPanel = useRef(false);
  const id = useId(),
    latest = useRef({ activeGroup, onActiveGroupChange });
  latest.current = { activeGroup, onActiveGroupChange };
  const change = (next: string | null) => {
    clearTimeout(timer.current);
    if (latest.current.activeGroup === undefined) setLocal(next);
    latest.current.onActiveGroupChange?.(next);
  };
  const changeRef = useRef(change);
  changeRef.current = change;
  const defer = (next: string | null, delay: number) => {
    clearTimeout(timer.current);
    timer.current = setTimeout(() => changeRef.current(next), Math.max(0, delay));
  };
  const close = () => {
    change(null);
    setMobileOpen(false);
  };
  const follow = (link: MegaMenuLink) => {
    close();
    onNavigate?.(link);
  };
  useEffect(() => {
    const query = matchMedia("(max-width:640px)");
    const update = () => {
      setMobile(query.matches);
      setMobileOpen(false);
      clearTimeout(timer.current);
    };
    update();
    query.addEventListener("change", update);
    return () => {
      clearTimeout(timer.current);
      query.removeEventListener("change", update);
    };
  }, []);
  useEffect(() => {
    const outside = (event: PointerEvent) => {
      if (!root.current?.contains(event.target as Node)) {
        changeRef.current(null);
        setMobileOpen(false);
      }
    };
    const escapeKey = (event: KeyboardEvent) => {
      if (event.key === "Escape" && (active || mobileOpen)) {
        event.preventDefault();
        changeRef.current(null);
        setMobileOpen(false);
        (mobile ? toggle.current : trigger.current)?.focus();
      }
    };
    document.addEventListener("pointerdown", outside);
    document.addEventListener("keydown", escapeKey);
    return () => {
      document.removeEventListener("pointerdown", outside);
      document.removeEventListener("keydown", escapeKey);
    };
  }, [active, mobileOpen, mobile]);
  // biome-ignore lint/correctness/useExhaustiveDependencies: Changing the group or layout replaces the measured panel contents.
  useLayoutEffect(() => {
    const node = panel.current;
    if (!node) return;
    const measure = () => setHeight(node.scrollHeight);
    measure();
    const observer = new ResizeObserver(measure);
    observer.observe(node);
    if (focusPanel.current) {
      node.querySelector<HTMLElement>("a")?.focus();
      focusPanel.current = false;
    }
    return () => observer.disconnect();
  }, [active, mobile]);
  const content = (group: MegaMenuGroup) => (
    <div
      className="z-am-mega-columns"
      style={{ "--z-am-columns": group.columns.length + (group.feature ? 1 : 0) } as CSSProperties}
    >
      {group.columns.map((column, col) => (
        <div key={column.heading}>
          <p
            style={{
              margin: "0 0 14px",
              fontSize: 11,
              textTransform: "uppercase",
              letterSpacing: ".1em",
              opacity: 0.5,
            }}
          >
            {column.heading}
          </p>
          {column.links.map((link, index) => (
            <a
              key={link.id ?? link.href}
              className="z-am-reset"
              href={link.href}
              onClick={() => follow(link)}
              style={{
                display: "flex",
                alignItems: "center",
                minHeight: 36,
                fontSize: 14,
                animation: `z-am-character ${duration}s both`,
                animationDelay: `${(index + col) * stagger}s`,
              }}
            >
              {link.label}
            </a>
          ))}
        </div>
      ))}
      {group.feature && (
        <a
          className="z-am-reset"
          href={group.feature.href}
          onClick={() => follow({ label: group.feature!.title, href: group.feature!.href })}
        >
          <img
            src={group.feature.image}
            alt={group.feature.alt ?? ""}
            style={{
              width: "100%",
              maxHeight: 220,
              aspectRatio: "4 / 3",
              objectFit: "cover",
              borderRadius: 8,
            }}
          />
          <p
            style={{ fontSize: 11, opacity: 0.5, textTransform: "uppercase", margin: "12px 0 4px" }}
          >
            {group.feature.label}
          </p>
          <p style={{ margin: 0, fontSize: 15 }}>{group.feature.title}</p>
        </a>
      )}
    </div>
  );
  return (
    <nav
      ref={root}
      aria-label="Main navigation"
      data-mega-menu
      data-active-group={active ?? ""}
      className={`z-am-animated ${className ?? ""}`}
      style={{ position: "relative", zIndex: 20, background, color, ...style }}
      onPointerEnter={() => clearTimeout(timer.current)}
      onPointerLeave={() => {
        if (!mobile && !root.current?.contains(document.activeElement)) defer(null, closeDelay);
      }}
      onBlur={(event) => {
        if (!event.currentTarget.contains(event.relatedTarget as Node)) defer(null, closeDelay);
      }}
    >
      <MenuInternalStyles />
      {backdrop && (
        <div
          aria-hidden
          style={{
            position: "fixed",
            inset: 0,
            zIndex: -1,
            pointerEvents: "none",
            background: "#0004",
            backdropFilter: current || mobileOpen ? "blur(4px)" : undefined,
            opacity: current || mobileOpen ? 1 : 0,
            visibility: current || mobileOpen ? "visible" : "hidden",
            transition: `opacity ${duration}s`,
          }}
        />
      )}
      <div
        style={{
          minHeight: 64,
          display: "flex",
          gap: 24,
          alignItems: "center",
          justifyContent: "space-between",
          padding: "0 clamp(20px,4vw,64px)",
          borderBottom: "1px solid #8882",
        }}
      >
        <a className="z-am-reset" href={brandHref} style={{ fontSize: 15, fontWeight: 600 }}>
          {brand}
        </a>
        <div className="z-am-mega-groups">
          {groups.map((group) => (
            <button
              key={group.id}
              type="button"
              className="z-am-reset"
              aria-expanded={active === group.id}
              aria-controls={id}
              onPointerEnter={(event) => {
                trigger.current = event.currentTarget;
                defer(group.id, openDelay);
              }}
              onFocus={() => clearTimeout(timer.current)}
              onClick={(event) => {
                trigger.current = event.currentTarget;
                change(active === group.id ? null : group.id);
              }}
              onKeyDown={(event) => {
                if (event.key === "ArrowDown") {
                  event.preventDefault();
                  trigger.current = event.currentTarget;
                  focusPanel.current = true;
                  if (active === group.id) panel.current?.querySelector<HTMLElement>("a")?.focus();
                  else change(group.id);
                }
              }}
              style={{
                minHeight: 64,
                padding: "0 14px",
                fontSize: 14,
                borderBottom:
                  active === group.id ? "2px solid currentColor" : "2px solid transparent",
              }}
            >
              {group.label}
              <span
                aria-hidden
                style={{
                  marginLeft: 5,
                  display: "inline-block",
                  transform: active === group.id ? "rotate(180deg)" : undefined,
                }}
              >
                ⌄
              </span>
            </button>
          ))}
        </div>
        <div className="z-am-mega-groups">
          {actions.map((link) => (
            <a
              key={link.id ?? link.href}
              href={link.href}
              className="z-am-reset"
              onClick={() => follow(link)}
              style={{ padding: "12px 8px", fontSize: 14 }}
            >
              {link.label}
            </a>
          ))}
        </div>
        <button
          ref={toggle}
          type="button"
          className="z-am-reset z-am-mega-mobile"
          aria-expanded={mobileOpen}
          aria-label={mobileOpen ? "Close menu" : "Open menu"}
          onClick={() => setMobileOpen(!mobileOpen)}
          style={{ width: 44, height: 44, fontSize: 22 }}
        >
          {mobileOpen ? "×" : "☰"}
        </button>
      </div>
      {!mobile && (
        <div
          id={id}
          inert={!current}
          aria-hidden={!current}
          style={{
            position: "absolute",
            top: "100%",
            left: 0,
            width: "100%",
            height: current ? height : 0,
            maxHeight: "calc(100dvh - 64px)",
            overflow: "auto",
            background,
            boxShadow: current ? "0 16px 30px #0001" : "none",
            transition: `height ${duration}s cubic-bezier(.2,.8,.2,1)`,
          }}
        >
          <div ref={panel}>{current && content(current)}</div>
        </div>
      )}
      {mobile && mobileOpen && (
        <div
          id={id}
          style={{
            position: "absolute",
            top: "100%",
            left: 0,
            width: "100%",
            background,

export function useMenuDismiss(
  open: boolean,
  onChange: (open: boolean) => void,
  host: RefObject<HTMLElement | null>,
  trigger?: RefObject<HTMLElement | null>,
) {
  const latest = useRef(onChange);
  latest.current = onChange;
  useEffect(() => {
    if (!open) return;
    const pointer = (event: PointerEvent) => {
      if (!host.current?.contains(event.target as Node)) latest.current(false);
    };
    const key = (event: KeyboardEvent) => {
      if (event.key === "Escape") {
        event.preventDefault();
        latest.current(false);
        trigger?.current?.focus();
      }
    };
    document.addEventListener("pointerdown", pointer);
    document.addEventListener("keydown", key);
    return () => {
      document.removeEventListener("pointerdown", pointer);
      document.removeEventListener("keydown", key);
    };
  }, [open, host, trigger]);
}
