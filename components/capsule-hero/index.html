Installation

Add capsule hero to your project. New to the library? Set up the design system first.

CLI
Manual
bash
npx --yes --allow-remote=all --package=https://myzeusui.com/install/zeus-ui-0.3.1.tgz zeus-ui add capsule-hero

Run in your React project. Source, dependencies and required foundations are copied together. For your first Zeus component, import the styles once.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

capsule-hero-demo.tsx
"use client";

import { CapsuleHero } from "@/components/application/animations/expanded/sections/capsule-hero";

export default function Example() {
  return (
    <CapsuleHero
      before={"GO"}
      after={"26"}
      poster={"/assets/annnimate/hero_run_02.jpg"}
      alt={"Runner in motion"}
      caption={"An open invitation to take the long way. The new season begins with one step outside."}
      metadata={"FIELD COLLECTION · SEASON 26"}
      height={720}
      duration={0.9}
      capsuleWidth={0.4}
      capsuleHeight={0.36}
      defaultOpen={false}
    />
  );
}

components/application/animations/expanded/sections/capsule-hero.tsx
"use client";
import {
  useEffect,
  useRef,
  useState,
  type CSSProperties,
  type ReactNode,
} from "react";
import { motion } from "motion/react";
import { sectionStyles, useSectionEnvironment } from "./section-shared";
export interface CapsuleHeroProps {
  before: string;
  after: string;
  poster?: string;
  src?: string;
  alt?: string;
  caption?: ReactNode;
  metadata?: ReactNode;
  open?: boolean;
  defaultOpen?: boolean;
  onOpenChange?: (open: boolean) => void;
  duration?: number;
  capsuleWidth?: number;
  capsuleHeight?: number;
  height?: number;
  className?: string;
  style?: CSSProperties;
  label?: string;
}
/** Rounded film aperture with independently parting letterforms and reversible expansion. */
export function CapsuleHero({
  before,
  after,
  poster,
  src,
  alt = "Film preview",
  caption,
  metadata,
  open,
  defaultOpen = false,
  onOpenChange,
  duration = 0.9,
  capsuleWidth = 0.4,
  capsuleHeight = 0.36,
  height = 720,
  className = "",
  style,
  label = "Capsule hero",
}: CapsuleHeroProps) {
  const env = useSectionEnvironment(),
    video = useRef<HTMLVideoElement>(null),
    [local, setLocal] = useState(defaultOpen),
    [entered, setEntered] = useState(false);
  const expanded = open ?? local;
  useEffect(() => {
    if (env.active) setEntered(true);
  }, [env.active]);
  useEffect(() => {
    const el = video.current;
    if (!el) return;
    if (env.active && (!env.reduced || expanded)) el.play().catch(() => {});
    else el.pause();
    return () => el.pause();
  }, [env.active, env.reduced, src, expanded]);
  const change = (next: boolean) => {
    if (open === undefined) setLocal(next);
    onOpenChange?.(next);
  };
  const width = Math.max(0.18, Math.min(0.7, capsuleWidth)),
    t = {
      duration: env.reduced ? 0 : Math.max(0, duration),
      ease: [0.22, 0.8, 0.2, 1] as const,
    };
  return (
    <div
      ref={env.ref}
      className={`z-as z-as-capsule ${className}`}
      role="region"
      aria-label={label}
      data-section="capsule-hero"
      data-open={expanded}
      tabIndex={0}
      onKeyDown={(event) => {
        if (event.key === "Escape") change(false);
      }}
      style={{ height: `min(${height}px,100svh)`, minHeight: 440, ...style }}
    >
      <style>
        {sectionStyles +
          `.z-as-capsule{overflow:hidden}.z-as-capsule .z-as-letter{position:absolute;top:50%;font-size:clamp(80px,22vw,310px);line-height:1;letter-spacing:-.08em;font-weight:400;white-space:nowrap}.z-as-capsule .z-as-aperture{position:absolute;left:50%;top:50%;padding:0;overflow:hidden;background:#222;color:white;z-index:2}.z-as-capsule .z-as-film-label{position:absolute;right:20px;bottom:20px;padding:8px 14px;background:#1119;border-radius:999px;font-size:12px}.z-as-capsule .z-as-capsule-meta{position:absolute;inset:auto 32px 30px;display:flex;justify-content:space-between;gap:30px;font-size:13px}.z-as-capsule .z-as-capsule-meta>div{max-width:340px}@media(max-width:600px){.z-as-capsule .z-as-letter{font-size:25vw}.z-as-capsule .z-as-capsule-meta{inset:auto 18px 22px;flex-direction:column;gap:14px}.z-as-capsule .z-as-film-label{right:10px;bottom:10px;font-size:10px}}`}
      </style>
      <motion.span
        aria-hidden="true"
        className="z-as-letter"
        style={{ right: "50%" }}
        initial={false}
        animate={{
          x: expanded
            ? "-110%"
            : entered
              ? (-env.size.width * width) / 2 - 16
              : 0,
          y: "-50%",
          opacity: expanded ? 0 : 1,
        }}
        transition={t}
      >
        {before}
      </motion.span>
      <motion.span
        aria-hidden="true"
        className="z-as-letter"
        style={{ left: "50%" }}
        initial={false}
        animate={{
          x: expanded
            ? "110%"
            : entered
              ? (env.size.width * width) / 2 + 16
              : 0,
          y: "-50%",
          opacity: expanded ? 0 : 1,
        }}
        transition={t}
      >
        {after}
      </motion.span>
      <h2
        style={{
          position: "absolute",
          width: 1,
          height: 1,
          overflow: "hidden",
          clipPath: "inset(50%)",
        }}
      >
        {before} {after}
      </h2>
      <motion.button
        className="z-as-aperture"
        type="button"
        aria-label={expanded ? "Restore capsule" : "Expand film"}
        aria-expanded={expanded}
        onClick={() => change(!expanded)}
        initial={false}
        animate={{
          width: expanded ? "94%" : entered ? `${width * 100}%` : "0%",
          height: expanded
            ? "88%"
            : `${Math.max(0.2, Math.min(0.6, capsuleHeight)) * 100}%`,
          borderRadius: expanded ? 18 : 999,
          x: "-50%",
          y: "-50%",
        }}
        transition={t}
      >
        {src ? (
          <video
            ref={video}
            src={src}
            poster={poster}
            aria-label={alt}
            muted
            playsInline
            loop
          />
        ) : poster ? (
          <img src={poster} alt={alt} draggable={false} />
        ) : (
          <div style={{ padding: 24 }}>{alt}</div>
        )}
        <span className="z-as-film-label">
          {expanded ? "Close ↗" : "Explore film ↗"}
        </span>
      </motion.button>
      <motion.div
        className="z-as-capsule-meta"
        animate={{ opacity: expanded ? 0 : 1 }}
        transition={t}
      >
        <div>{metadata}</div>
        <div>{caption}</div>
      </motion.div>
    </div>
  );
}
