Installation

Add logo to your project. New to the library? Set up the design system first.

CLI
Manual
Download the CLI bundle

Save the bundle in your project folder, then run this command there. It includes the registry, tokens and example images.

bash
npx --yes --package=./zeus-ui-0.2.2.tgz zeus-ui add logo

Downloads this site’s CLI, copies the source into your project and pulls in everything it needs — cx, tokens. The component source is yours; imported npm packages are installed as needed.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

logo-demo.tsx
"use client";

import { Logo } from "@/components/foundations/brand/logo";

export default function LogoDemo() {
  return (
    <div className="flex items-center gap-8">
      <Logo />
    </div>
  );
}

Source

1 file. Source code you can read, customize and keep in your project.

components/foundations/brand/logo.tsx
import { cx } from "../../utils/cx";

/** Zeus mark, self-contained so SDK consumers do not need a site asset or image loader. */
export function Logo({ size = 32, mono = false, className, decorative = false }: {
  size?: number;
  mono?: boolean;
  className?: string;
  decorative?: boolean;
}) {
  return (
    <svg width={size} height={size} viewBox="0 0 32 32" fill="none"
      role={decorative ? "presentation" : "img"} aria-label={decorative ? undefined : "Zeus UI"}
      aria-hidden={decorative || undefined}
      className={cx("shrink-0", mono ? "text-text-primary" : "text-accent-500", className)}>
      <rect x="2" y="2" width="28" height="28" rx="7" fill="currentColor" />
      <path d="M10 9h14l-8 9h6L8 26l5-10H8l2-7Z" fill="var(--color-background-full, white)" />
    </svg>
  );
}

/** Retained for applications that distinguish a Pro badge in their own interface. */
export function ProLogo({ size = 36, className, decorative = false }: {
  size?: number;
  className?: string;
  decorative?: boolean;
  priority?: boolean;
}) {
  return <Logo size={size} className={className} decorative={decorative} mono />;
}
