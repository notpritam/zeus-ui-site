Installation

Add activity rings to your project. New to the library? Set up the design system first.

CLI
Manual
bash
npx --yes --allow-remote=all --package=https://myzeusui.com/install/zeus-ui-0.3.0.tgz zeus-ui add activity-rings-card

Run in your React project. Source, dependencies and required foundations are copied together. For your first Zeus component, import the styles once.

Usage

The complete example running in the preview. Adjust the props and content for your interface.

activity-rings-card-demo.tsx
"use client";

import { ActivityRingsCard } from "@/components/application/charts/activity-rings-card";

export default function ActivityRingsCardDemo() {
  return (
    <div className="w-full max-w-md">
      <ActivityRingsCard />
    </div>
  );
}

components/application/charts/activity-rings-card.tsx
"use client";
import { motion } from "motion/react";
import { useState } from "react";
import { useHydratedReducedMotion } from "../../hooks/use-hydrated-reduced-motion";
import { cx } from "../../utils/cx";
import { StatTiles, safeRatio } from "./pro-chart";
export interface ActivityRing {
  label: string;
  value: number;
  goal: number;
  unit?: string;
  tone: number;
  color?: string;
  displayValue?: string;
  progress?: number;
}
export interface ActivityRingsCardProps {
  rings?: ActivityRing[];
  title?: string;
  className?: string;
  height?: number | "auto";
  selectedDate?: Date;
}
export const ACTIVITY_RINGS: ActivityRing[] = [
  { label: "Move", value: 1592, goal: 1941, unit: "kcal", tone: 3, progress: 0.82 },
  {
    label: "Exercise",
    value: 105,
    goal: 175,
    unit: "min",
    tone: 2,
    displayValue: "1h 45m",
    progress: 0.6,
  },
  { label: "Running", value: 5.2, goal: 6.93, unit: "km", tone: 4, progress: 0.75 },
];
export function ActivityRings({
  rings = ACTIVITY_RINGS,
  active,
  onHover,
  className,
  animate = true,
}: {
  rings?: ActivityRing[];
  active?: number | null;
  onHover?: (index: number | null) => void;
  animate?: boolean;
  className?: string;
}) {
  const reduced = useHydratedReducedMotion(),
    step = rings.length > 1 ? Math.min(24, 64 / (rings.length - 1)) : 24,
    stroke = Math.min(18, step * 0.75);
  return (
    <svg
      viewBox="0 0 200 200"
      className={cx("h-full max-h-[210px] w-full overflow-visible", className)}
      role="img"
      aria-label={rings
        .map((r) => `${r.label}: ${r.displayValue ?? `${r.value} ${r.unit ?? ""}`}`)
        .join(", ")}
    >
      {rings.map((r, i) => {
        const radius = 82 - i * step,
          color = r.color ?? `var(--color-chart-${r.tone})`,
          progress = safeRatio(r.progress ?? r.value / r.goal, 1) * 100;
        return (
          <g
            key={r.label}
            transform="rotate(-90 100 100)"
            onMouseEnter={() => onHover?.(i)}
            onMouseLeave={() => onHover?.(null)}
            opacity={active != null && active !== i ? 0.25 : 1}
            className="transition-opacity duration-200 motion-reduce:transition-none"
          >
            <circle
              cx={100}
              cy={100}
              r={radius}
              fill="none"
              stroke={color}
              strokeWidth={stroke}
              opacity={0.16}
            />
            <motion.circle
              key={reduced ? "reduced" : "animated"}
              cx={100}
              cy={100}
              r={radius}
              pathLength={100}
              fill="none"
              stroke={color}
              strokeWidth={stroke}
              strokeLinecap="round"
              initial={reduced || !animate ? false : { strokeDasharray: "0 100" }}
              animate={{ strokeDasharray: `${progress} ${100 - progress}` }}
              transition={{ duration: reduced || !animate ? 0 : 0.65, ease: "easeOut" }}
            />
          </g>
        );
      })}
    </svg>
  );
}
export function ActivityRingsCard({
  rings = ACTIVITY_RINGS,
  title = "Activity",
  className,
  height = 330,
  selectedDate,
}: ActivityRingsCardProps = {}) {
  const [active, setActive] = useState<number | null>(null);
  const derived =
    selectedDate && rings === ACTIVITY_RINGS
      ? rings.map((r, i) => ({
          ...r,
          value: i === 0 ? 816 + (selectedDate.getDate() - 10) * 17 : r.value,
          displayValue: i === 0 ? undefined : r.displayValue,
          progress: i === 0 ? 0.29 + (selectedDate.getDate() % 8) / 20 : r.progress,
        }))
      : rings;
  return (
    <section
      className={cx(
        "flex w-full min-w-0 flex-col gap-4 rounded-[20px] bg-background-secondary-default p-2.5",
        className,
      )}
      style={{ height: height === "auto" ? undefined : height }}
    >
      <div className="flex flex-col gap-[11px]">
        <p className="px-1.5 pt-1.5 text-body-medium text-text-secondary">{title}</p>
        <div className="px-2 pb-1">
          <StatTiles
            items={derived.map((r) => ({
              label: r.label,
              value: r.displayValue ?? `${r.value.toLocaleString("en-US")} ${r.unit ?? ""}`,
              color: r.color ?? `var(--color-chart-${r.tone})`,
            }))}
            active={active}
            onHover={setActive}
          />
        </div>
      </div>
      <div className="flex min-h-0 w-full flex-1 items-center justify-center">
        <ActivityRings rings={derived} active={active} onHover={setActive} />
      </div>
    </section>
  );
}
