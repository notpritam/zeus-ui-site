export function ButtonSurface({
  variant,
  variables,
  children,
  className = "",
  style,
  disabled = false,
  href,
  target,
  rel,
  download,
  type = "button",
  name,
  value,
  form,
  formAction,
  formMethod,
  formNoValidate,
  formTarget,
  elementRef,
  onPress,
  onClick,
  onEngage,
  onDisengage,
  onTrack,
  onPointerEnter,
  onPointerLeave,
  onPointerMove,
  onPointerDown,
  onPointerUp,
  onPointerCancel,
  onFocus,
  onBlur,
  duration: _duration,
  easing: _easing,
  background: _background,
  fill: _fill,
  color: _color,
  hoverColor: _hoverColor,
  radius: _radius,
  ...native
}: SurfaceProps) {
  const [active, setActive] = useState(false);
  const hover = useRef(false),
    focus = useRef(false);
  const engage = (element: ActionElement, event?: PointerEvent<ActionElement>) => {
    if (disabled) return;
    setActive(true);
    onEngage?.(element, event);
  };
  const disengage = (element: ActionElement) => {
    if (hover.current || focus.current) return;
    setActive(false);
    onDisengage?.(element);
  };
  const shared = {
    ...native,
    ref: elementRef,
    className: `z-anb-action z-anb-${variant} ${className}`,
    style: { ...variables, ...style },
    "data-active": active && !disabled,
    onClick: (event: MouseEvent<ActionElement>) => {
      if (disabled) {
        event.preventDefault();
        return;
      }
      onClick?.(event);
      if (!event.defaultPrevented) onPress?.(event);
    },
    onPointerEnter: (event: PointerEvent<ActionElement>) => {
      onPointerEnter?.(event);
      if (event.pointerType !== "touch") {
        hover.current = true;
        engage(event.currentTarget, event);
      }
    },
    onPointerLeave: (event: PointerEvent<ActionElement>) => {
      onPointerLeave?.(event);
      hover.current = false;
      disengage(event.currentTarget);
    },
    onPointerMove: (event: PointerEvent<ActionElement>) => {
      onPointerMove?.(event);
      if (!disabled && event.pointerType !== "touch") onTrack?.(event.currentTarget, event);
    },
    onPointerDown: (event: PointerEvent<ActionElement>) => {
      onPointerDown?.(event);
      if (event.pointerType === "touch") engage(event.currentTarget, event);
    },
    onPointerUp: (event: PointerEvent<ActionElement>) => {
      onPointerUp?.(event);
      if (event.pointerType === "touch") disengage(event.currentTarget);
    },
    onPointerCancel: (event: PointerEvent<ActionElement>) => {
      onPointerCancel?.(event);
      hover.current = false;
      disengage(event.currentTarget);
    },
    onFocus: (event: FocusEvent<ActionElement>) => {
      onFocus?.(event);
      // Pointer clicks retain their native focus without pinning hover motion.
      focus.current = event.currentTarget.matches(":focus-visible");
      if (focus.current) engage(event.currentTarget);
    },
    onBlur: (event: FocusEvent<ActionElement>) => {
      onBlur?.(event);
      focus.current = false;
      disengage(event.currentTarget);
    },
  };
  return (
    <>
      <style>{buttonCss}</style>
      {href !== undefined ? (
        <a
          {...shared}
          ref={elementRef as Ref<HTMLAnchorElement>}
          href={disabled ? undefined : href}
          aria-disabled={disabled || undefined}
          tabIndex={disabled ? -1 : native.tabIndex}
          target={target}
          rel={rel ?? (target === "_blank" ? "noopener noreferrer" : undefined)}
          download={download}
        >
          {children}
        </a>
      ) : (
        <button
          {...shared}
          ref={elementRef as Ref<HTMLButtonElement>}
          disabled={disabled}
          type={type}
          name={name}
          value={value}
          form={form}
          formAction={formAction}
          formMethod={formMethod}
          formNoValidate={formNoValidate}
          formTarget={formTarget}
        >
          {children}
        </button>
      )}
    </>
  );
}
