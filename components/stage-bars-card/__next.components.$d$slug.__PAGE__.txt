export function useChartRange<T extends ChartRange>(
  ranges: T[] | undefined,
  defaultRange?: string,
  onChange?: (id: string) => void,
) {
  const [selected, setSelected] = useState(defaultRange ?? ranges?.[0]?.id);
  const current = ranges?.find((range) => range.id === selected) ?? ranges?.[0];

    setSelected(id);

export function AnimatedNumber({
  value,
  format = formatNumber,
}: {
  value: number;
  format?: (value: number) => string;
}) {
  const reduced = useHydratedReducedMotion();
  const [display, setDisplay] = useState(value);
  const current = useRef(value);
  useEffect(() => {
    if (reduced) {
      current.current = value;
      setDisplay(value);
      return;
    }
    const from = current.current,
      start = performance.now();
    let frame = 0;
    const tick = (time: number) => {
      const t = Math.min(1, (time - start) / 300);
      current.current = from + (value - from) * (1 - (1 - t) ** 3);
      setDisplay(current.current);
      if (t < 1) frame = requestAnimationFrame(tick);
    };
    frame = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(frame);
  }, [value, reduced]);
  return <>{format(reduced ? value : display)}</>;
}
